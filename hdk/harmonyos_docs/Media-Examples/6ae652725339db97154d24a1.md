## 创建对象，初始化并注册callback

```screen
mHwAudioKit = new HwAudioKit(this, callBack);
mHwAudioKit.initialize();
// 创建KaraokeFeature：
mHwAudioKaraokeFeatureKit = mHwAudioKit.createFeature(HwAudioKit.FeatureType. HWAUDIO_FEATURE_KARAOKE);
// 创建回调
IAudioKitCallback callBack = new IAudioKitCallback() {
@Override
public void onResult(int i) {
if (i == 0) {
// HwAudioKit初始化成功,使用其接口,例如：
mHwAudioKit .getSupportedFeatures(); //返回list集合
mHwAudioKit .isFeatureSupported(FeatureType type); //返回true或false；true表示支持，false表示产品不支持查询的功能
}
if (i == 2) {
// 未安装Engine服务，不支持kit
}
if (i == 1000) {
// HwAudioKaraokeFeatureKit初始化成功, 可使用其接口, 例如：
// 查询Karaoke是否支持, 返回true或false；true表示支持，false表示产品不支持:
int isSupport = mHwAudioKaraokeFeatureKit.isKaraokeFeatureSupport();
// 打开或关闭耳返, 返回int数值 0表示成功，1806表示产品不支持K歌，1805表示未插耳机:
boolean   = mHwAudioKaraokeFeatureKit.enableKaraokeFeature(enable);
// 设置音效模式及音量, 0表示成功，1806表示产品不支持K歌，1805表示未插耳机，1807表示参数值错误
int success = mHwAudioKaraokeFeatureKit.setParameter(HwAudioKaraokeFeatureKit.ParameName.CMD_SET_AUDIO_EFFECT_MODE_BASE, value);
// 获取时延信息, -1表示获取失败
int latency = mHwAudioKaraokeFeatureKit.getKaraokeLatency();
  }
 }
}
```
