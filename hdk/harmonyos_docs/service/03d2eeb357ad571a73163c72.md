# chips（话题）推荐

Agent chips推荐场景下，小艺感知用户当前所在App，向Agent请求推荐内容，由Agent返回适合当前场景的操作建议。该场景为一次性任务，contextId与会话保持一致，连续请求复用同一contextId。

![](https://contentcenter-vali-drcn.dbankcdn.cn/pvt_2/DeveloperAlliance_scene_100_1/8c/v3/ivMMQtkFQHuKwTaj4sXysA/zh-cn_image_0000002670263969.png?HW-CC-KV=V1&HW-CC-Date=20260924T081231Z&HW-CC-Expire=31536000000&HW-CC-Sign=6F1D111F3F4FD7A83B230235BC08D2D50957FB839C3B24492E7FB7AF5341A0FE "点击放大")

小艺发起推荐请求（小艺发送至Agent）：

```codeblock
{
   "jsonrpc":"2.0",
   "id":"req-001",
   "method":"PerceptionSuggest",
   "params":{
     "message":{
       "messageId":"msg-001",
       "contextId":"ctx-001",
       "role":"ROLE_USER",
       "parts":[
         {
           "data":{
             "perceptionContext":{
               "foregroundApp":"com.example.app",
               "browserContexts":[],
               "currentScreenContent":{},
               "intentLabels":[]
             }
           }
         }
       ]
     },
     "metadata":{
       "traceId":"trace-7f8a2b3c-0001",
       "agentId":"agent-example-001",
       "appVersion":"12.0.1"
     }
   }
 }
```

Agent返回推荐内容（Agent发送至小艺）：

```codeblock
{
   "jsonrpc":"2.0",
   "id":"req-001",
   "result":{
     "artifactUpdate":{
       "taskId":"task-001",
       "contextId":"ctx-001",
       "artifact":{
         "artifactId":"art-suggestions-001",
         "parts":[
           {
             "data":{
               "suggestionCandidates":[
                 {
                   "text":"查看最新订单",
                   "reply":"查看最新订单"
                 },
                 {
                   "text":"联系客服",
                   "jump":{
                     "deepLink":{
                       "url":"example://service",
                       "appName":"示例应用",
                       "appPackage":"com.example.app"
                     }
                   }
                 }
               ]
             }
           }
         ]
       },
       "append":false,
       "lastChunk":true
     },
     "metadata":{
       "traceId":"trace-7f8a2b3c-0001",
       "agentId":"agent-example-001",
       "agentVersion":"1.2.0"
     }
   }
 }
```

内容为连续的两帧：

```codeblock
{
  "jsonrpc":"2.0",
  "id":"req-001",
  "result":{
    "statusUpdate":{
      "taskId":"task-001",
      "contextId":"ctx-001",
      "status":{
        "state":"TASK_STATE_COMPLETED",
        "timestamp":"2026-03-24T10:00:02.000Z"
      }
    },
    "metadata":{
      "traceId":"trace-7f8a2b3c-0001",
      "agentId":"agent-example-001",
      "agentVersion":"1.2.0"
    }
  }
 }
```
