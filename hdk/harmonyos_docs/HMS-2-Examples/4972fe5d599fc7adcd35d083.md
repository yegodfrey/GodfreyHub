        36.         }
        37.         for (int i = 26, j = 0; i <= 51; i++, j++) {
        38.             lookUpBase64Alphabet[i] = (char) ('a' + j);
        39.         }
        40.         for (int i = 52, j = 0; i <= 61; i++, j++) {
        41.             lookUpBase64Alphabet[i] = (char) ('0' + j);
        42.         }
        43.         lookUpBase64Alphabet[62] = (char) '+';
        44.         lookUpBase64Alphabet[63] = (char) '/';
        45.     }
        46.     private static boolean isWhiteSpace(char octect) {
        47.         return (octect == 0x20 || octect == 0xd || octect == 0xa || octect == 0x9);
        48.     }
        49.     private static boolean isPad(char octect) {
        50.         return (octect == PAD);
        51.     }
        52.     private static boolean isData(char octect) {
        53.         return (octect < BASELENGTH && base64Alphabet[octect] != -1);
        54.     }
        55.     /**
        56.      * Encodes hex octects into Base64
        57.      *
        58.      * @param binaryData Array containing binaryData
        59.      * @return Encoded Base64 array
        60.      */
        61.     public static String encode(byte[] binaryData) {
        62.         if (binaryData == null) {
        63.             return null;
        64.         }
        65.         int lengthDataBits = binaryData.length * EIGHTBIT;
        66.         if (lengthDataBits == 0) {
        67.             return "";
        68.         }
        69.         int fewerThan24bits = lengthDataBits % TWENTYFOURBITGROUP;
        70.         int numberTriplets = lengthDataBits / TWENTYFOURBITGROUP;
        71.         int numberQuartet = fewerThan24bits != 0 ? numberTriplets + 1 : numberTriplets;
        72.         char encodedData[] = null;
        73.         encodedData = new char[numberQuartet * 4];
        74.         byte k = 0, l = 0, b1 = 0, b2 = 0, b3 = 0;
        75.         int encodedIndex = 0;
        76.         int dataIndex = 0;
        77.         if (fDebug) {
        78.             System.out.println("number of triplets = " + numberTriplets);
        79.         }
        80.         for (int i = 0; i < numberTriplets; i++) {
        81.             b1 = binaryData[dataIndex++];
        82.             b2 = binaryData[dataIndex++];
        83.             b3 = binaryData[dataIndex++];
        84.             if (fDebug) {
        85.                 System.out.println("b1= " + b1 + ", b2= " + b2 + ", b3= " + b3);
        86.             }
        87.             l = (byte) (b2 & 0x0f);
        88.             k = (byte) (b1 & 0x03);
        89.             byte val1 = ((b1 & SIGN) == 0) ? (byte) (b1 >> 2) : (byte) ((b1) >> 2 ^ 0xc0);
        90.             byte val2 = ((b2 & SIGN) == 0) ? (byte) (b2 >> 4) : (byte) ((b2) >> 4 ^ 0xf0);
        91.             byte val3 = ((b3 & SIGN) == 0) ? (byte) (b3 >> 6) : (byte) ((b3) >> 6 ^ 0xfc);
        92.             if (fDebug) {
        93.                 System.out.println("val2 = " + val2);
        94.                 System.out.println("k4   = " + (k << 4));
        95.                 System.out.println("vak  = " + (val2 | (k << 4)));
        96.             }
        97.             encodedData[encodedIndex++] = lookUpBase64Alphabet[val1];
        98.             encodedData[encodedIndex++] = lookUpBase64Alphabet[val2 | (k << 4)];
        99.             encodedData[encodedIndex++] = lookUpBase64Alphabet[(l << 2) | val3];
        100.             encodedData[encodedIndex++] = lookUpBase64Alphabet[b3 & 0x3f];
        101.         }
        102.         // form integral number of 6-bit groups
        103.         if (fewerThan24bits == EIGHTBIT) {
        104.             b1 = binaryData[dataIndex];
        105.             k = (byte) (b1 & 0x03);
        106.             if (fDebug) {
        107.                 System.out.println("b1=" + b1);
        108.                 System.out.println("b1<<2 = " + (b1 >> 2));
        109.             }
        110.             byte val1 = ((b1 & SIGN) == 0) ? (byte) (b1 >> 2) : (byte) ((b1) >> 2 ^ 0xc0);
        111.             encodedData[encodedIndex++] = lookUpBase64Alphabet[val1];
        112.             encodedData[encodedIndex++] = lookUpBase64Alphabet[k << 4];
        113.             encodedData[encodedIndex++] = PAD;
        114.             encodedData[encodedIndex++] = PAD;
        115.         } else if (fewerThan24bits == SIXTEENBIT) {
        116.             b1 = binaryData[dataIndex];
        117.             b2 = binaryData[dataIndex + 1];
        118.             l = (byte) (b2 & 0x0f);
        119.             k = (byte) (b1 & 0x03);
        120.             byte val1 = ((b1 & SIGN) == 0) ? (byte) (b1 >> 2) : (byte) ((b1) >> 2 ^ 0xc0);
        121.             byte val2 = ((b2 & SIGN) == 0) ? (byte) (b2 >> 4) : (byte) ((b2) >> 4 ^ 0xf0);
        122.             encodedData[encodedIndex++] = lookUpBase64Alphabet[val1];
        123.             encodedData[encodedIndex++] = lookUpBase64Alphabet[val2 | (k << 4)];
        124.             encodedData[encodedIndex++] = lookUpBase64Alphabet[l << 2];
        125.             encodedData[encodedIndex++] = PAD;
        126.         }
        127.         return new String(encodedData);
        128.     }
        129.     /**
        130.      * Decodes Base64 data into octects
        131.      *
        132.      * @param encoded string containing Base64 data
        133.      * @return Array containind decoded data.
        134.      */
        135.     public static byte[] decode(String encoded) {
        136.         if (encoded == null) {
        137.             return null;
        138.         }
        139.         char[] base64Data = encoded.toCharArray();
        140.         // remove white spaces
        141.         int len = removeWhiteSpace(base64Data);
        142.         if (len % FOURBYTE != 0) {
        143.             return null;// should be divisible by four
        144.         }
        145.         int numberQuadruple = (len / FOURBYTE);
        146.         if (numberQuadruple == 0) {
        147.             return new byte[0];
        148.         }
        149.         byte decodedData[] = null;
        150.         byte b1 = 0, b2 = 0, b3 = 0, b4 = 0;
        151.         char d1 = 0, d2 = 0, d3 = 0, d4 = 0;
        152.         int i = 0;
        153.         int encodedIndex = 0;
        154.         int dataIndex = 0;
        155.         decodedData = new byte[(numberQuadruple) * 3];
        156.         for (; i < numberQuadruple - 1; i++) {
        157.             if (!isData((d1 = base64Data[dataIndex++])) || !isData((d2 = base64Data[dataIndex++]))
        158.                     || !isData((d3 = base64Data[dataIndex++])) || !isData((d4 = base64Data[dataIndex++]))) {
        159.                 return null;
        160.             }
        161.             // if found "no data" just return null
        162.             b1 = base64Alphabet[d1];
        163.             b2 = base64Alphabet[d2];
        164.             b3 = base64Alphabet[d3];
        165.             b4 = base64Alphabet[d4];
        166.             decodedData[encodedIndex++] = (byte) (b1 << 2 | b2 >> 4);
        167.             decodedData[encodedIndex++] = (byte) (((b2 & 0xf) << 4) | ((b3 >> 2) & 0xf));
        168.             decodedData[encodedIndex++] = (byte) (b3 << 6 | b4);
        169.         }
        170.         if (!isData((d1 = base64Data[dataIndex++])) || !isData((d2 = base64Data[dataIndex++]))) {
        171.             return null;// if found "no data" just return null
        172.         }
        173.         b1 = base64Alphabet[d1];
        174.         b2 = base64Alphabet[d2];
        175.         d3 = base64Data[dataIndex++];
        176.         d4 = base64Data[dataIndex++];
        177.         if (!isData((d3)) || !isData((d4))) {// Check if they are PAD characters
        178.             if (isPad(d3) && isPad(d4)) {
        179.                 if ((b2 & 0xf) != 0)// last 4 bits should be zero
        180.                 {
        181.                     return null;
        182.                 }
        183.                 byte[] tmp = new byte[i * 3 + 1];
        184.                 System.arraycopy(decodedData, 0, tmp, 0, i * 3);
        185.                 tmp[encodedIndex] = (byte) (b1 << 2 | b2 >> 4);
        186.                 return tmp;
        187.             } else if (!isPad(d3) && isPad(d4)) {
        188.                 b3 = base64Alphabet[d3];
        189.                 if ((b3 & 0x3) != 0)// last 2 bits should be zero
        190.                 {
        191.                     return null;
        192.                 }
        193.                 byte[] tmp = new byte[i * 3 + 2];
        194.                 System.arraycopy(decodedData, 0, tmp, 0, i * 3);
        195.                 tmp[encodedIndex++] = (byte) (b1 << 2 | b2 >> 4);
        196.                 tmp[encodedIndex] = (byte) (((b2 & 0xf) << 4) | ((b3 >> 2) & 0xf));
        197.                 return tmp;
        198.             } else {
        199.                 return null;
        200.             }
        201.         } else { // No PAD e.g 3cQl
        202.             b3 = base64Alphabet[d3];
        203.             b4 = base64Alphabet[d4];
        204.             decodedData[encodedIndex++] = (byte) (b1 << 2 | b2 >> 4);
        205.             decodedData[encodedIndex++] = (byte) (((b2 & 0xf) << 4) | ((b3 >> 2) & 0xf));
        206.             decodedData[encodedIndex++] = (byte) (b3 << 6 | b4);
        207.         }
        208.         return decodedData;
        209.     }
        210.     /**
        211.      * remove WhiteSpace from MIME containing encoded Base64 data.
        212.      *
        213.      * @param data the byte array of base64 data (with WS)
        214.      * @return the new length
        215.      */
        216.     private static int removeWhiteSpace(char[] data) {
        217.         if (data == null) {
        218.             return 0;
        219.         }
        220.         // count characters that's not whitespace
        221.         int newSize = 0;
        222.         int len = data.length;
        223.         for (int i = 0; i < len; i++) {
        224.             if (!isWhiteSpace(data[i])) {
        225.                 data[newSize++] = data[i];
        226.             }
        227.         }
        228.         return newSize;
        229.     }
