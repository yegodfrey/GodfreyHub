        249.     private void checkPay() {
        250.         // 取出所有未确认订单 | Remove all unacknowledged orders
        251.         SharedPreferences sp = getSharedPreferences(UNCHECK_PAYREQUESTID_FILE, 0);
        252.         MapallUnCheckedPays = sp.getAll();
        253.         Set setAllUnChecked = allUnCheckedPays.entrySet();
        254.         boolean hasUnConfirmedPayRecorder = false;
        255.         // 循环查询 | Circular Query
        256.         for (Map.Entryele :setAllUnChecked){
        257.             if (ele == null) {
        258.                 continue;
        259.             }
        260.             Object valueObj = ele.getValue();
        261.             if (valueObj != null && valueObj instanceof Boolean) {
        262.                 Boolean valueBoolean = (Boolean) valueObj;
        263.                 if (!valueBoolean) {
        264.                     String reqId = ele.getKey();
        265.                     getPayDetail(reqId);
        266.                     hasUnConfirmedPayRecorder = true;
        267.                 }
        268.             }
        269.         }
        270.         if (!hasUnConfirmedPayRecorder) {
        271.             showLog("checkPay: no pay to check");
        272.         }
        273.     }

        341.             public void onResult(int retCode, OrderResult checkPayResult) {
        342.                 showLog("checkPay: requId=" + reqId + "  retCode=" + retCode);
        343.                 if (checkPayResult != null && checkPayResult.getReturnCode() == retCode) {

        346.                         boolean checkRst = PaySignUtil.checkSign(checkPayResult, pay_pub_key);
        347.                         if (checkRst) {

        349.                             showLog("checkPay: Pay successfully, distribution of goods");
        350.                         } else {
        351.                             // 验签失败，当支付失败处理 | Verification fails when payment fails to deal with
        352.                             showLog("checkPay: Failed to verify signature, pay failed");
        353.                         }
        354.                         // 不需要再查询 | No more queries
        355.                         removeCacheRequestId(checkPayResult.getRequestId());
        356.                     } else if (retCode == PayStatusCodes.ORDER_STATUS_HANDLING
        357.                             || retCode == PayStatusCodes.ORDER_STATUS_UNTREATED
        358.                             || retCode == PayStatusCodes.PAY_STATE_TIME_OUT) {
        359.                         // 未处理完，需要重新查询。如30分钟后再次查询。超过24小时当支付失败处理 | Not finished processing, you need to requery. such as 30 minutes after the query again. More than 24 hours when payment fails to handle
        360.                         showLog("checkPay: Pay failed. errorCode=" + retCode + "  errMsg=" + checkPayResult.getReturnDesc());
        361.                     } else if (retCode == PayStatusCodes.PAY_STATE_NET_ERROR) {
        362.                         // 网络失败，需要重新查询 | Network failure, need to Requery
        363.                         showLog("checkPay: A network problem caused the payment to fail. errorCode=" + retCode + "  errMsg=" + checkPayResult.getReturnDesc());
        364.                     } else {
        365.                         // 支付失败，不需要再查询 | Payment failed, no more queries required
        366.                         showLog("checkPay: Pay failed. errorCode=" + retCode + "  errMsg=" + checkPayResult.getReturnDesc());
        367.                         removeCacheRequestId(reqId);
        368.                     }
        369.                 } else {
        370.                     // 没有结果回来，需要重新查询。如30分钟后再次查询。超过24小时当支付失败处理 | No results back, you need to requery. such as 30 minutes after the query again. More than 24 hours when payment fails to handle
        371.                     showLog("checkPay: Pay failed. errorCode=" + retCode);
        372.                 }
        373.             }
        374.         });
