        230.     public static boolean doCheck(String content, String sign, String publicKey, String signtype) {            
        231.        try {            
        232.            KeyFactory keyFactory = KeyFactory.getInstance("RSA");            
        233.            byte[] encodedKey = decoder.decode(publicKey);            
        234.            PublicKey pubKey = keyFactory.generatePublic(new X509EncodedKeySpec(encodedKey));            
        235.            java.security.Signature signature = null;            
        236.            if ("RSA256".equals(signtype)) {            
        237.               signature = java.security.Signature.getInstance(SIGN_ALGORITHMS256);            
        238.            } else {            
        239.               signature = java.security.Signature.getInstance(SIGN_ALGORITHMS);            
        240.            }            
        241.            signature.initVerify(pubKey);            
        242.            signature.update(content.getBytes("utf-8"));            
        243.            boolean bverify = signature.verify(decoder.decode(sign));            
        244.            return bverify;            
        245.        } catch (Exception e) {            
        246.            e.printStackTrace();            
        247.        }            
        248.        return false;            
        249.     }
