请求头需要使用AccessToken进行鉴权，这里使用开放平台鉴权提供的服务接口获取AccessToken。服务接口提供两种模式来获取AccessToken：授权码模式和客户端密码模式，本接口使用客户端密码模式。具体可参考[开放平台鉴权](https://developer.huawei.com/consumer/cn/doc/30101)。  
  
最佳实践：不要在每次调用服务端接口之前都重新申请应用AT，频繁的申请可能触发服务拒绝，导致在一定时间内无法申请新的应用AT。每个申请的AT均有一个有效期返回，在有效期内该AT可以反复使用。建议在访问服务端接口并且返回的http结果码为401时，才重新申请应用AT。

  


获取AT的JAVA示例代码：
         
         复制
         
              1. public class AtDemo {
              2.     // TODO: The values of (clientId, clientSecret, tokenUrl) should be replaced with the actual one.
              3.     private static final String clientSecret = "appsecret"; //your client secret
              4.     // !!! the client id is your APP ID
              5.     private static final String clientId = "1234567"; //your app id
              6.     private static final String tokenUrl = "https://oauth-login.cloud.huawei.com/oauth2/v2/token"; //token url to get the authorization
              7.     /**
              8.     * The accessToken.
              9.     */
              10.     private static String accessToken;
              11.     /**
              12.     * Gets App Level AccessToken.
              13.     *
              14.     * @return the App Level AccessToken
              15.     * @throws Exception the exception
              16.     */
              17.     public static String getAppAT() throws Exception {
              18.         //fetch accessToken
              19.         String grant_type = "client_credentials";
              20.         String msgBody = MessageFormat.format("grant_type={0}&client_secret={1}&client_id={2}", grant_type,URLEncoder.encode(clientSecret, "UTF-8"), clientId);
              21.         String response = httpPost(tokenUrl, "application/x-www-form-urlencoded; charset=UTF-8", msgBody, 5000, 5000, null);
              22.         JSONObject obj = JSONObject.parseObject(response);
              23.         accessToken = obj.getString("access_token");
              24.         // TODO: display the accessToken in console, you can remove it
              25.         System.out.println(accessToken);
              26.         return accessToken;
              27.     }
              28.     /**
              29.     * Build Authorization in Header
              30.     *
              31.     * @param appAt appAt
              32.     * @return headers
              33.     */
              34.     public static Map<String, String> buildAuthorization(String appAt) {
              35.         String oriString = MessageFormat.format("APPAT:{0}", appAt);
              36.         String authorization =
              37.         MessageFormat.format("Basic {0}", Base64.encodeBase64String(oriString.getBytes(StandardCharsets.UTF_8)));
              38.         Map<String, String> headers = new HashMap<>();
              39.         headers.put("Authorization", authorization);
              40.         headers.put("Content-Type", "application/json; charset=UTF-8");
              41.         return headers;
              42.     }
              43.     /**
              44.     * Http post function.
              45.     *
              46.     * @param httpUrl the http url
              47.     * @param data the data
              48.     * @param connectTimeout the connect timeout
              49.     * @param readTimeout the read timeout
              50.     * @param headers the headers
              51.     * @return the response as string
              52.     * @throws IOException the io exception
              53.     */
              54.     public static String httpPost(String httpUrl, String contentType, String data, int connectTimeout, int readTimeout, Map<String, String> headers) throws IOException {
              55.         OutputStream output = null;
              56.         InputStream in = null;
              57.         HttpURLConnection urlConnection = null;
              58.         BufferedReader bufferedReader = null;
              59.         InputStreamReader inputStreamReader = null;
              60.         try {
              61.             URL url = new URL(httpUrl);
              62.             urlConnection = (HttpURLConnection)url.openConnection();
              63.             urlConnection.setRequestMethod("POST");
              64.             urlConnection.setDoOutput(true);
              65.             urlConnection.setDoInput(true);
              66.             urlConnection.setRequestProperty("Content-Type", contentType);
              67.             if (headers != null) {
              68.                 for (String key : headers.keySet()) {
              69.                     urlConnection.setRequestProperty(key, headers.get(key));
              70.                 }
              71.             }
              72.             urlConnection.setConnectTimeout(connectTimeout);
              73.             urlConnection.setReadTimeout(readTimeout);
              74.             urlConnection.connect();
              75.             // POST data
              76.             output = urlConnection.getOutputStream();
              77.             output.write(data.getBytes("UTF-8"));
              78.             output.flush();
              79.             // read response
              80.             if (urlConnection.getResponseCode() < 400) {
              81.                 in = urlConnection.getInputStream();
              82.             } else {
              83.                 in = urlConnection.getErrorStream();
              84.             }
              85.             inputStreamReader = new InputStreamReader(in, "UTF-8");
              86.             bufferedReader = new BufferedReader(inputStreamReader);
              87.             StringBuilder strBuf = new StringBuilder();
              88.             String str;
              89.             while ((str = bufferedReader.readLine()) != null) {
              90.                 strBuf.append(str);
              91.             }
              92.             return strBuf.toString();
              93.         } finally {
              94.             if (bufferedReader != null) {
              95.                 bufferedReader.close();
              96.             }
              97.             if (inputStreamReader != null) {
              98.                 inputStreamReader.close();
              99.             }
              100.             if (in != null) {
              101.                 in.close();
              102.             }
              103.             if (urlConnection != null) {
              104.                 urlConnection.disconnect();
              105.             }
              106.         }
              107.     }
              108. }
