# 获取HarmonyOS 5及以上应用包名、组件名、模块的方法

1、查找入口HAP包

当应用是APP包时，找到是PC入口的HAP包 ；当应用是HAP包时，需要满足HAP为入口HAP包。

PC入口HAP包判断条件：

* hapInfo.distro.moduleType为entry
* 不需要进行类型判断产品基于CCM配置参数const.bms.supportAppTypes判断支持安装类型
* HAP中存在Ability的SkillInfo的actions中包含action.system.home或ohos.want.action.home
* HAP中存在Ability的SkillInfo的entities中包含entity.system.home

<br />

2、使用工具获取HAP包中应用入口信息

通过<https://gitee.com/openharmony/developtools_packing_tool>工具

官方文档:[https://developer.huawei.com/consumer/cn/doc/harmonyos-guides/unpacking-tool#%E5%8C%85%E8%A7%A3%E6%9E%90%E6%8E%A5%E5%8F%A3](https://developer.huawei.com/consumer/cn/doc/harmonyos-guides/unpacking-tool#包解析接口)

解析HAP包，遍历getProfileInfos的结果，找到第一个hapInfo.distro.moduleType为entry的profileInfo

找到HAP包中ability skills中包含（action.system.home或ohos.want.action.home）和entity.system.home的abilityName和其所属module中的name

若没有moduleType为entry的profile信息，则这个hap不支持桌面图标启动

* 应用名称：UncompressResult.getLabel()
* 应用包名：profileInfo中的appInfo.bundleName
* 组件名称：profileInfo中的hapInfo.mainElement
* 模块名称：profileInfo中的hapInfo.distro.moduleName

  ```
  import java.util.List;
  import ohos.UncompressEntrance;
  import ohos.UncompressResult;
  import ohos.ProfileInfo;
  import ohos.PackInfo;
  import ohos.AbilityInfo;
  import ohos.SkillInfo;

  public class App {
      public static boolean hasMainAbility(List<SkillInfo> infos) {
          boolean hasMainAbility = false;
          for (int i = 0; i < infos.size(); i++) {
              SkillInfo info = infos.get(i);
              boolean hasActionSystemHome = info.actions.contains("action.system.home");
              boolean hasActionSystemHome2 = info.actions.contains("ohos.want.action.home");
              boolean hasEntitySystemHone = info.entities.contains("entity.system.home");
              System.out.println("skillInfo.actions: " + info.actions);
              System.out.println("skillInfo.entities: " + info.entities);
              if ((hasActionSystemHome || hasActionSystemHome2) && hasEntitySystemHone) {
                  hasMainAbility = true;
                  break;
              }
          }
          return hasMainAbility;
      }

      public static boolean isEntryHap(ProfileInfo profile) {
          String moduleType = profile.hapInfo.distro.moduleType;
          List<String> deviceType = profile.hapInfo.deviceType;
          List<AbilityInfo> abilityInfos = profile.hapInfo.abilities;
          boolean hasMainAbility = false;
          for(int j = 0; j < abilityInfos.size(); j++) {
              AbilityInfo abilityInfo = abilityInfos.get(j);
              List<SkillInfo> skillInfos = abilityInfo.skills;
              if (hasMainAbility(skillInfos)) {
                  hasMainAbility = true;
                  break;
              }
          }
          return "entry".equals(moduleType) && hasMainAbility;
      }

      public static void main(String[] args) {
          UncompressResult result = UncompressEntrance.parseHap("xxxx.hap");
          List<ProfileInfo> profileInfos = result.getProfileInfos();
          System.out.println("\nGet info by actions & entites:");
          for(int i = 0; i < profileInfos.size(); i++) {
              ProfileInfo profile = profileInfos.get(i);
              if (!App.isEntryHap(profile)) {
                  continue;
              }
              System.out.println("Found module:");
              System.out.println("hapInfo.distro.moduleType: " + profile.hapInfo.distro.moduleType);
              System.out.println("hapInfo.deviceType: " + profile.hapInfo.deviceType);

              System.out.println("\nIcon: " + result.getIcon());
              System.out.println("Label: " + result.getLabel());
              System.out.println("appInfo.bundleName: " + profile.appInfo.bundleName);
              System.out.println("hapInfo.mainElement: " + profile.hapInfo.mainElement);
              System.out.println("hapInfo.distro.moduleName: " + profile.hapInfo.distro.moduleName);
              break;
          }
      }
  }
  ```
