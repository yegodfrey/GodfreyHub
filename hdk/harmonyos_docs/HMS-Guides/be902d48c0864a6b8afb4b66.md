# 在Manifest指定权限

在调用车载感知能力时，开发者需要先在Manifest中指定相应的权限。
      
      1. <!-- 蓝牙权限 -->
        2. <uses-permission android:name="android.permission.BLUETOOTH" />
    
    

# 导入接口类

使用车载蓝牙感知能力，除需要导入情景感知服务的公共能力类外，还需要导入车载蓝牙相关的类。
      
      1. import com.huawei.hmf.tasks.OnFailureListener;
        2. import com.huawei.hmf.tasks.OnSuccessListener;
        3. import com.huawei.hms.kit.awareness.Awareness;
        4. //导入车载蓝牙快照相关类
        5. import com.huawei.hms.kit.awareness.capture.BluetoothStatusResponse;
        6. import com.huawei.hms.kit.awareness.status.BluetoothStatus;
        7. //导入车载蓝牙围栏相关类
        8. import com.huawei.hms.kit.awareness.barrier.AwarenessBarrier;
        9. import com.huawei.hms.kit.awareness.barrier.BarrierStatus;
        10. import com.huawei.hms.kit.awareness.barrier.BluetoothBarrier;
        11. import com.huawei.hms.kit.awareness.barrier.BarrierUpdateRequest;
    
    

# 能力开发

## Capture能力开发

1\. 获取Awareness Kit的“Capture Client”。

2\. 通过“Capture Client”调用车载蓝牙状态查询能力接口查询情景状态。

3\. 监听情景感知服务结果返回，进行应用的业务处理。
      
      1. int deviceType = 0; // Value 0 indicates a Bluetooth car stereo.
        2. Awareness.getCaptureClient(this).getBluetoothStatus(deviceType)
        3.         .addOnSuccessListener(new OnSuccessListener<BluetoothStatusResponse>() {
        4.             @Override
        5.             public void onSuccess(BluetoothStatusResponse bluetoothStatusResponse) {
        6.                 BluetoothStatus bluetoothStatus = bluetoothStatusResponse.getBluetoothStatus();
        7.                 int status = bluetoothStatus.getStatus();
        8.                 String stateStr = "The Bluetooth car stereo is " +
        9.                         (status == BluetoothStatus.CONNECTED ? "connected" : "disconnected");
        10.                 Log.i(TAG, stateStr);
        11.             }
        12.         })
        13.         .addOnFailureListener(new OnFailureListener() {
        14.             @Override
        15.             public void onFailure(Exception e) {
        16.                 Log.e(TAG, "get bluetooth status failed", e);
        17.             }
        18.         });
    
    

## Barrier能力开发

以下以开发车载蓝牙“connecting”的Barrier为例（车载蓝牙连接后触发Barrier）。

1\. 定义Barrier。
      
      1. final int deviceType = 0; // Value 0 indicates a Bluetooth car stereo.
        2. AwarenessBarrier connectingBarrier = BluetoothBarrier.connecting(deviceType);
    
    

2\. 定义Barrier状态改变时触发的“PendingIntent”，本示例中为发送一个广播，同时新建一个广播接收器用来接收这个广播。
      
      1. final String BARRIER_RECEIVER_ACTION = getApplication().getPackageName() + "BLUETOOTH_BARRIER_RECEIVER_ACTION";
        2. Intent intent = new Intent(BARRIER_RECEIVER_ACTION);
        3. PendingIntent pendingIntent = PendingIntent.getBroadcast(this, 1, intent, PendingIntent.FLAG_UPDATE_CURRENT);
        4. BluetoothBarrierReceiver barrierReceiver = new BluetoothBarrierReceiver();
        5. registerReceiver(barrierReceiver, new IntentFilter(BARRIER_RECEIVER_ACTION));
    
    

3\. 添加Barrier。
      
      1. String bluetoothBarrierLabel = "bluetooth connecting barrier";
        2. BarrierUpdateRequest.Builder builder = new BarrierUpdateRequest.Builder();
        3. BarrierUpdateRequest request = builder.addBarrier(bluetoothBarrierLabel, connectingBarrier,pendingIntent).build(); 
        4. Awareness.getBarrierClient(context).updateBarriers(request)
        5.         .addOnSuccessListener(new OnSuccessListener<Void>() {
        6.             @Override
        7.             public void onSuccess(Void aVoid) {
        8.                 Toast.makeText(getApplicationContext(), "add barrier success", Toast.LENGTH_SHORT).show();
        9.             }
        10.         })
        11.         .addOnFailureListener(new OnFailureListener() {
        12.             @Override
        13.             public void onFailure(Exception e) {
        14.                 Toast.makeText(getApplicationContext(), "add barrier failed", Toast.LENGTH_SHORT).show();
        15.                 Log.e(TAG, "add barrier failed", e);
        16.             }
        17.         });
    
    

4\. 定义广播接收器，用于监听Barrier事件，收到事件后进行应用的业务处理。
      
      1. class BluetoothBarrierReceiver extends BroadcastReceiver {
        2.     @Override
        3.     public void onReceive(Context context, Intent intent) {
        4.         BarrierStatus barrierStatus = BarrierStatus.extract(intent);
        5.         String label = barrierStatus.getBarrierLabel();
        6.         switch(barrierStatus.getPresentStatus()) {
        7.             case BarrierStatus.TRUE:
        8.                 Log.i(TAG, label + " status:true");
        9.                 break;
        10.             case BarrierStatus.FALSE:
        11.                 Log.i(TAG, label + " status:false");
        12.                 break;
        13.             case BarrierStatus.UNKNOWN:
        14.                 Log.i(TAG, label + " status:unknown");
        15.                 break;
        16.         }
        17.     }
        18. }
