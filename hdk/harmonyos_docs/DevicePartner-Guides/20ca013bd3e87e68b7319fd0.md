# 蓝牙情况

公共参数：

src -\> store -\> state.js

```
// 蓝牙相关
  bleOn: false, // 手机蓝牙是否开启
  devId: '',
  mac: '',
  connectStatus: 1, // 设备蓝牙连接状态 0-未连接 1-正在连接 2-已连接 3-断连中
  connectOk: 1, // 设备连接认证回调 0-未连接 1-正在连接 2-已连接
```

src -\> store -\> getters.js

```
// 都为true则连接蓝牙
  canConnect(state) {
    <!-- 只有蓝牙开启的情况，且获取到设备devId,mac才能连接设备 -->
    if (state.bleOn && state.devId && state.mac) {
      return true;
    }
    return false;
  },
  canControl(state) {
    return state.connectStatus === 2 && state.connectOk === 2;
  },
```

1. 初始获取蓝牙开关状态

   <br />

   onBluetoothAdapterStateChange

   ```
   // 监听蓝牙开关状态，手机上主动打开或关闭蓝牙模块会触发该函数
     onBluetoothAdapterStateChange({ state, dispatch }, payload = {}) {
       let { callback } = payload;
       console.hLog(`监听蓝牙开关状态 onBluetoothAdapterStateChange`)
       return generatePromise({
         method: 'onBluetoothAdapterStateChange',
         callback,
         onCallback: (res = {}) => {
           // 返回当前蓝牙的状态
           console.hLog(`当前蓝牙开关状态 ${res}`)
           if (res.available === undefined) {
             return;
           }
           state.bleOn = res.available;
           // 如果是关闭的情况，需要调用openBluetoothAdapter，请求打开蓝牙
           if (!res.available) {
             dispatch('openBluetoothAdapter');
           }
         }
       });
     }
   ```

3. 获取当前产品在云侧的缓存数据并保存

   <br />

   getCurrentRegisteredDevice

   从该接口中获取设备的相关信息，比如devId，mac，可用于下一步连接和订阅接口

   ```
   // 获取当前注册设备缓存信息
     getCurrentRegisteredDevice({ state, commit, dispatch }) {
       // 双连双控，安卓和鸿蒙的蓝牙接口是挂在在bleHilink对象上的，iOS是hilink对象
       let isIOS = state.type == "ios";
       let hilink = isIOS ? window.hilink : window.blehilin;
       if (!hilink || !hilink.getCurrentRegisteredDevice) {
         return new Promise((resolve) => {
           resolve(true);
         });
       }
       return generatePromise({
         method: "getCurrentRegisteredDevice",
       }).then((res = {}) => {
         console.hLog("设备信息 getCurrentRegisteredDevice:", res);
         // 获取到devId和mac后订阅设备消息
         state.devId = res.hilinkDevId;
         state.mac = res.deviceId;
         state.role = res.hilinkDeviceEntity.role;
         state.fwv = res.hilinkDeviceEntity.devInfo.fwv;
         state.hwv = res.hilinkDeviceEntity.devInfo.hwv;
         state.sn = res.hilinkDeviceEntity.devInfo.sn;
         state.model = res.hilinkDeviceEntity.devInfo.model;

         console.hLog("当前是否为共享用户:", res.hilinkDeviceEntity.role);
         if (res.hilinkDeviceEntity.role === "owner") {
           commit("UPDATE_STATE", {
             name: "isShare",
             value: false,
           });
         } else if (res.hilinkDeviceEntity.role === "family") {
           commit("UPDATE_STATE", {
             name: "isShare",
             value: true,
           });
         }

         // 订阅设备消息
         dispatch("subscribeBleEvent");
         if (res.hilinkDeviceEntity && res.hilinkDeviceEntity.services) {
           let result = res.hilinkDeviceEntity,
             services = result.services,
             devInfo = result.devInfo || {};
           // 更新设备id,名称
           commit("UPDATE_DEVICE_INFO", {
             devId: result.devId,
             devName: result.devName,
             otaVersion:
               (devInfo.fwv &&
                 devInfo.hwv &&
                 `${devInfo.fwv}(${devInfo.hwv})`) ||
               "",
           });
           console.hLog("devall", services);
           for (var i = 0; i < services.length; i++) {
             commit("UPDATE_CLOUD_DATA", services[i]);
           }
         }
       });
     }
   ```

4. 订阅蓝牙信息，更新H5与设备侧交互产生的数据变化，需特别注意回调数据的格式处理。若控制操作后出现文字描述与按钮状态不一致的情况，应优先检查数据格式subscribeBleEvent

   <br />

   该接口会返回两种消息：

   1）ConnectionStateChange 蓝牙连接状态（iOS上报的范围是1-4，安卓是0-3要特殊处理）。

   2）CharacteristicChanged 设备profile相关信息。

   ```
   // 订阅Ble消息
     subscribeBleEvent({ state, commit, dispatch }, payload = {}) {
       /**
        * 订阅蓝牙消息
        *
        * @param deviceId 设备id
        * @param bleMac 设备mac地址
        * @param resultCallback 回调
        */
       let { params = [state.devId, state.mac] } = payload;
       console.hLog("订阅:\n" + `devId:${params[0]}\n` + `mac:${params[1]}`);
       return generatePromise({
         method: "subscribeBleEvent",
         params,
         onCallback: (res = {}) => {
           console.hLog("上报:", JSON.stringify(res));
           if (res.type === "ConnectionStateChange") {
             switch (res.newStatus) {
               case 0: // 已断连
               case 1: // 连接中
               case 2: // 已连接
               case 3: // 断连中
               case 4: // iOS上报的4
                 state.connectStatus = res.newStatus;
                 if (state.type === "iOS" && res.newStatus === 4) {
                   state.connectStatus = 0;
                 }
                 if (
                   res.newStatus === 0 ||
                   (state.type === "iOS" && res.newStatus === 4)
                 ) {
                   state.connectOk = 0;
                   try {
                     window.connectBreakCallback();
                   } catch (error) {
                     console.log(error);
                   }
                 }
                 break;
               default:
                 break;
             }
           } else if (res.type === "CharacteristicChanged") {
             let content =
               res.content && res.content.vendor
                 ? res.content.vendor
                 : res.content;
             let sid = content && content.sid;
             let data = content && content.data;

             if (!content || !sid || !data) {
               return;
             }
             // 更新设备上报
             commit("commitHilink", content);
           }
         }
       });
     }
   ```

5. 连接蓝牙，更新当前连接状态

   <br />

   connectBle

   ```
   connectBle({ state, commit, dispatch }, payload = {}) {
       /**
        * 连接蓝牙
        *
        * @param deviceId 设备id
        * @param bleMac 设备蓝牙Mac地址
        * @param resultCallback 回调
        */
       commit("UPDATE_STATE", {
         name: "connectOk",
         value: 1, // 连接中
       });
       let { params = [state.devId, state.mac] } = payload;
       console.hLog("连接:\n" + `devId:${params[0]}\n` + `mac:${params[1]}`);

       return Promise.race([
         // 连接
         generatePromise({
           method: "connectBle",
           params,
           callback: `connectBleCallback${new Date().getTime()}`,
         }),
         // 倒计时
         dispatch("connectTimeout", 20000),
         // 连接中断
         dispatch("connectBreak"),
       ]).then((res = {}) => {
         console.hLog(`连接回调:${JSON.stringify(res)}`);
         commit("UPDATE_STATE", {
           name: "connectOk",
           value: res.errcode === 0 ? 2 : 0,
         });
         if (res.errcode === 0) {
           // 连接成功后重新连接的次数归零
           commit("UPDATE_STATE", {
             name: "reConnectCount",
             value: 0,
           });
         } else {
           // 连接失败 次数+1
           if (state.isReconnect) {
             commit("UPDATE_STATE", {
               name: "reConnectCount",
               value: ++state.reConnectCount,
             });
             console.hLog("重新连接失败的次数: ", state.reConnectCount);
             // 是否连接归零
             commit("UPDATE_STATE", {
               name: "isReconnect",
               value: false,
             });
           }
         }
       });
     }
   ```

6. 给设备发送控制命令

   <br />

   sendCommand

   ```
   // Ble通道发送命令
     sendCommand({ state, commit }, payload = {}) {
       /**
        * 修改蓝牙设备属性
        *
        * @param deviceId 设备id
        * @param bleMac 设备mac地址
        * @param serviceId sid
        * @param data 下发命令数据
        * @param resultCallback 回调
        */
       console.log(`payload = ${String(JSON.stringify(payload))}`);
       // console.log(`payload = ${JSON.stringify(payload)}`)
       if (state.controlling) {
         console.hLog("受控中");
         commit("UPDATE_STATE", {
           name: "NoControl",
           value: true,
         });
         Toast(i18n.t("controlling"));
         return;
       }
       commit("UPDATE_STATE", {
         name: "NoControl",
         value: false,
       });
       let {
         params = [state.devId, state.mac],
         data = "{}",
         toast = true,
       } = payload;
       let orignData = data;
       typeof data === "string" && (data = JSON.parse(data));
       let serviceId = data.sid;
       data =
         typeof data.data === "string"
           ? data.data
           : JSON.stringify(data.data);
       let method = "sendCommand";
       console.hLog(
         "method = ",
         method,
         "下发sid = ",
         serviceId,
         "data = ",
         data
       );
       let param =
         method === "sendCommand"
           ? [...params, serviceId, data]
           : [...params, JSON.stringify(orignData)];
       return generatePromise({
         method: method,
         params: param,
         afterHilinkCalled: () => {
           state.controlling = true;
         },
       }).then((res = {}) => {
         state.controlling = false;
         console.hLog("下发回调:", JSON.stringify(res));
         if (res.errcode !== 0 && toast) {
           Toast(i18n.t("control_fail"));
         }
         return res;
       });
     }
   ```
