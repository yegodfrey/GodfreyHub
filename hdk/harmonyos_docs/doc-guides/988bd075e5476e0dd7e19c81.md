js 文件示意如下：
    
    // index.js
     
    import mediaquery from '@system.mediaquery';
    export default {
      data: {
        title: "",
        todolist: [1, 2, 3, 4, 5],
        pageTab: ["标题", "页签", "页签", "页签", "页签", "页签", "页签", "页签"],
        csstab_text: 'tab_text',
        csstab_text1: 'tab_text1',
        index_text: 0,
    // 视频数据
        videoDate: [
          {videoSrc: "/common/video1.mp4", videoPoster: "pages/common/images/sshBody_Picture4.png"},
          {videoSrc: "/common/video1.mp4", videoPoster: "pages/common/images/sshBody_Picture5.png"},
          {videoSrc: "/common/video1.mp4", videoPoster: "pages/common/images/sshBody_Picture6.png"},
          {videoSrc: "/common/video1.mp4", videoPoster: "pages/common/images/sshBody_Picture7.png"}
        ],
    // 轮播数据
        swiperArr: [
          {img: 'pages/common/images/yBody_Picture1.png'},
          {img: 'pages/common/images/yBody_Picture6.png'},
          {img: 'pages/common/images/sBody_Picture2.png'},
        ],
        swiperArr1: [
          {img: 'pages/common/images/yBody_Picture1.png', img1: 'pages/common/images/yBody_Picture6.png'},
          {img: 'pages/common/images/sBody_Picture2.png', img1: 'pages/common/images/yBody_Picture1.png'},
        ],
        flag: true
      },
      onInit() {
        this.title = this.$t('strings.world');
      },
    // 媒体查询，动态判断flag类型，改变轮播布局
      onReady() {
        var mMediaQueryList = mediaquery.matchMedia('(max-width: 1200)');
        mMediaQueryList.addListener(this.maxWidthMatch);
      },
      maxWidthMatch(e) {
        if (e.matches) {
          this.flag = true
        } else {
          this.flag = false
        }
      },
      changeTab(ind, itm) {
        this.index_text = ind
      },
    }
