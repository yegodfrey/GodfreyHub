js 文件示意如下：
    
    // index.js
     
    import prompt from '@system.prompt';
    import mediaquery from '@system.mediaquery';
    export default {
      data: {
        title: "",
        todolist: [1, 2, 3, 4, 5, 6, 7, 8],
        pageTab: ["标题", "页签", "页签", "页签", "页签", "页签", "页签", "页签"],
        csstab_text: 'tab_text',
        csstab_text1: 'tab_text1',
        index_text: 0,
    // 轮播数据
        swiperArr: [
          {img: "pages/common/images/yBody_Picture1.png"},
          {img: "pages/common/images/yBody_Picture6.png"},
          {img: "pages/common/images/yBody_Picture.png"}
        ],
        swiperArr1: [
          {img: "pages/common/images/yBody_Picture1.png", img1: "pages/common/images/yBody_Picture6.png",},
          {img: "pages/common/images/yBody_Picture.png", img1: "pages/common/images/yBody_Picture1.png",}
        ],
        flag: true,
      },
    // 媒体查询
      onReady() {
        var mMediaQueryList = mediaquery.matchMedia('(max-width: 1200)');
        mMediaQueryList.addListener(this.maxWidthMatch);
      },
      maxWidthMatch(e) {
        if (e.matches) {
          this.flag = true
        } else {
          this.flag = false
        }
      },
    // 自定义弹窗
      showDialog(e) {
        this.$element('simpledialog').show()
      },
      changeTab(ind, itm) {
        this.index_text = ind
      },
      cancelSchedule(e) {
        this.$element('simpledialog').close()
        prompt.showToast({
          message: 'Successfully cancelled'
        })
      },
      setSchedule(e) {
        this.$element('simpledialog').close()
        prompt.showToast({
          message: 'Successfully confirmed'
        })
      },
    }
