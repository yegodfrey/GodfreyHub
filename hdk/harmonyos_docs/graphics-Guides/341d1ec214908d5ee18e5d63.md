# shader

## 配置说明

CG Kit渲染框架支持前向渲染和延迟渲染，延迟渲染所使用的shader请参见[延迟渲染管线](https://developer.huawei.com/consumer/cn/doc/development/graphics-Guides/deferred-rendering-0000001222790387)。CG Kit渲染框架会提供pbr Shader，包括顶点shader（vertex shader，简称vs）和像素shader（fragment shader，简称fs）的模板。顶点shader模板的名称为forward_pbr.vert，像素shader模板会根据光源信息进行区分，分别为directional_forward_pbr.frag（平行光），point_forward_pbr.frag（点光源）和spot_forward_pbr.frag（聚光灯），给用户提供了相关pbr的顶点shader和像素shader。

* 顶点shader片段，提供实现了获取世界坐标的函数方法。
* 像素shader片段，提供实现了获取纹理通道的相关函数。

一个完整的顶点shader包含顶点shader模板和顶点shader片段，一个完整的像素shader包含像素shader模板和像素shader片段。

CG Kit渲染框架提供的编译脚本将shader源码编译成可执行的二进制文件，在[cgmat](https://developer.huawei.com/consumer/cn/doc/development/graphics-Guides/cgmat-configuration-file-0000001106071566)材质文件中可配置为.vert或.frag的shader片段文件，也支持.spv的二进制文件。当cgmat中配置的为shader片段文件时，会根据添加的光源信息，自动与不同光源的模板进行拼接使用，针对不同光源可应用不同的shader文件；当cgmat中配置的为spv二进制文件时，不会进行shader的拼接，所有的光源均会应用同一个shader。

当前渲染框架仅支持预编译好的shader二进制文件（.spv格式），不支持实时编译shader。当前顶点shader的二进制文件和像素shader的二进制文件会根据用户配置的顶点shader片段、像素shader片段和模板拼接而成。如果cgmat不配置shader片段，系统会默认按照static.vert和basic_pbr.frag作为片段与模板进行拼接，skybox.cgmat配置示例如下。

```screen
"shader": [
    {
        "uri": "shaders/sky.frag",
        "type": "SHADER_STAGE_TYPE_FRAGMENT"
    },
    {
        "uri": "shaders/static.vert",
        "type": "SHADER_STAGE_TYPE_VERTEX"
    }
]
```

### 编写顶点shader

1. 顶点shader模板定义如下（forward_pbr.vert）。

   ```cpp
   #version 430 core

   // 定义顶点数据结构（location如下）。
   // 位置location必须设置为0
   layout(location=0) in vec3 inPosition;
   // 纹理坐标location必须设置为1
   layout(location=1) in vec2 inTexcoord0;
   // 法线location必须设置为2
   layout(location=2) in vec3 inNormal;
   // 切线location必须设置为3
   layout(location=3) in vec4 inTangent;

   // 定义全局常量数据结构（set、binding设置及数据结构与如下代码示例保持一致）。
   layout(set=0, binding=0) uniform GlobalUniform
   {
       mat4 view;
       mat4 projection;
       mat4 projectionOrtho;
       mat4 viewProjection;
       mat4 viewProjectionInv;
       mat4 viewProjectionOrtho;
       vec4 resolution;
       vec4 cameraPosition;
   }
   globalUniform;

   layout(set=0, binding=7) uniform ModelUniform
   {
       mat4 model;
   }
   modelUniform;

   layout (location = 0) out vec3 outPosition;
   layout (location = 1) out vec2 outTexcoord0;
   layout (location = 2) out vec3 outEyePosition;
   layout (location = 3) out mat3 outTangentBasis;

   void main()
   {
       vec4 worldPosition = getWorldPosition(inPosition);
       outPosition = worldPosition.xyz;
       outTexcoord0 = vec2(inTexcoord0.x, inTexcoord0.y);
       outEyePosition = globalUniform.cameraPosition.xyz;
       vec3 normal = getNormal(inNormal);
       // 传递切线空间基向量（用于法线映射）
       vec3 bitangent = cross(inTangent.xyz, normal) * inTangent.w;
       outTangentBasis = mat3(modelUniform.model) * mat3(inTangent.xyz, bitangent, normal);
       gl_Position = globalUniform.viewProjection * worldPosition;
   }
   ```

2. 顶点shader片段定义如下（static.vert）。

   ```screen
   vec4 getWorldPosition(vec3 position)
   {
       return modelUniform.model * vec4(position, 1.0);
   }

   vec3 getNormal(vec3 normal)
   {
       return normal;
   }
   ```

### 编写像素shader

1. 平行光像素shader模板定义如下（directional_forward_pbr.frag）。

   ```screen
   #version 430 core

   precision highp float;
   precision mediump sampler2D;

   const float PI = 3.141592;
   const float Epsilon = 0.00001;

   #define ENABLE_LIGHT

   // 光照数据结构（set、binding设置及数据结构与如下代码示例保持一致）
   layout (push_constant) uniform LightInfo{
       // xyz为颜色信息，w为强度信息
       vec4 color;
       // xyz为位置信息   
       vec4 position;
       // xyz为方向信息   
       vec4 direction;
       // x为类型信息
       vec4 typeAngle;
       // 衰减
       vec4 attenuation;
       // 环境
       vec4 ambient;
       // 漫反射
       vec4 diffuse;
       // 镜面
       vec4 specular;
   } lightInfo;

   // 所有电介质的恒定法线入射菲涅尔系数
   const vec3 Fdielectric = vec3(0.220916301);

   layout (location = 0) in vec3 inPosition;
   layout (location = 1) in vec2 inTexcoord0;
   layout (location = 2) in vec3 inEyePosition;
   layout (location = 3) in mat3 inTangentBasis;

   layout(location=0) out vec4 color;

   float ndfGGX(float nh, float roughness)
   {
       float alphaSq = roughness * roughness;
       float denom = (nh * nh) * (alphaSq - 1.0) + 1.0;
       return alphaSq / (PI * denom * denom);
   }

   float gaSchlickG1(float cosTheta, float k)
   {
       return cosTheta / (cosTheta * (1.0 - k) + k);
   }

   float gaSchlickGGX(float nl, float nv, float roughness)
   {
       float r = roughness + 1.0;
       // 将这种粗糙度重新映射用于分析光
       float k = (r * r) / 8.0;
       return gaSchlickG1(nl, k) * gaSchlickG1(nv, k);
   }

   vec3 fresnelSchlick(vec3 F0, float cosTheta)
   {
       return F0 + (vec3(1.0) - F0) * pow(1.0 - cosTheta, 5.0);
   }

   vec3 FresnelLerp(vec3 F0, vec3 F90, float cosA)
   {
       float t = pow(1 - cosA, 5.0);
       return mix(F0, F90, t);
   }

   float DisneyDiffuse(float NdotV, float NdotL, float LdotH, float perceptualRoughness)
   {
       float fd90 = 0.5 + 2 * LdotH * LdotH * perceptualRoughness;
       float lightScatter   = (1 + (fd90 - 1) * pow((1 - NdotL),5.0));
       float viewScatter    = (1 + (fd90 - 1) * pow((1 - NdotV),5.0));

       return lightScatter * viewScatter;
   }

   void main()
   {
       // 采样输入纹理以获取着色模型参数
       vec4 albedoTex = getAlbedoChannel();
       vec3 albedo = albedoTex.rgb;
       vec3 orm = getPbrChannel().rgb;
       float ao = orm.x;
       float perceptualRoughness = orm.y;
       float metalness = orm.z;
       float roughness = perceptualRoughness * perceptualRoughness;

       // 出射光方向（从世界空间像素位置到"眼睛"的矢量）
       vec3 eyePosition = inEyePosition;
       vec3 Lo = normalize(eyePosition - inPosition);

       // 获取当前像素点的法线并转换到世界空间
       vec3 N = normalize(2.0 * getNormalChannel().rgb - 1.0);
       N = normalize(inTangentBasis[0] * N.x + inTangentBasis[1] * N.y + inTangentBasis[2] * N.z);

       // 曲面法线和出射光方向之间的角度
       float nv = max(0.0, dot(N, Lo));

       // 镜面反射矢量
       vec3 Lr = reflect(-Lo, N);

       // 法向入射的菲涅尔反射率（对于金属，使用反照率颜色）
       vec3 F0 = mix(Fdielectric, albedo, metalness);

       // 灯的直接光照计算  
       vec3 directLighting = vec3(0);

       // 获取漫反射系数
       float kd = (1 - metalness) * 0.779083699;

   #ifdef ENABLE_LIGHT
       vec3 Li = vec3(0,0,1);

       // 光照强度加权后的光照颜色
       vec3 Lradiance = lightInfo.color.xyz * lightInfo.color.w;

       // 归一化的入射光方向
       Li = -normalize(lightInfo.direction.xyz);

       // 入射光与出射光的半角向量
       vec3 Lh = normalize(Li + Lo);

       // 计算法线与各种光向量的点积
       float nl = max(0.0, dot(N, Li));
       float nh = max(0.0, dot(N, Lh));
       float hl = max(0.0, dot(Lh, Li));

       // 迪士尼漫反射项
       float diffuseTerm = DisneyDiffuse(nv, nl, hl, perceptualRoughness);
       vec3 diffuseBRDF = kd * albedo * diffuseTerm;

       // 菲涅尔系数
       vec3 F = fresnelSchlick(F0, hl);
       // 法线分布函数
       float D = ndfGGX(nh, roughness);
       // 几何函数
       float G = gaSchlickGGX(nl, nv, roughness);

       // 库克-托伦斯镜面BRDF，漫反射项不按1/PI缩放，镜面反射项按PI倍放大
       float specularTerm = (D * G) * PI / max(Epsilon, 4.0 * nl * nv);
       specularTerm = sqrt(max(0.0001, specularTerm));
       vec3 specularBRDF = (F * specularTerm);

       directLighting += (diffuseBRDF + specularBRDF) * Lradiance * nl;
   #endif

       // 环境照明（IBL）
       vec3 ambientLighting;
       {
           // 采样法线方向的漫射辐照度
           vec3 irradiance = getEnvChannel(N, 10.0).rgb;

           // 计算环境照明的菲涅尔项。因为使用预过滤的立方体贴图，且辐照度来自多个方向，因此使用nv代替半角向量nh
           vec3 F = fresnelSchlick(F0, nv);

           // 辐照度贴图包含存在的辐射，假设BRDF，这里也不需要按1/PI缩放
           vec3 diffuseIBL = kd * albedo * irradiance;

           // 在正确的mipmap级别上采样预过滤的镜面反射环境 
           vec3 specularIrradiance = getEnvChannel(Lr, perceptualRoughness * 4.0).rgb;

           // 库克-托伦斯镜面BRDF的分和近似因子
           vec2 specularBRDF = getBRDFChannel(vec2(nv, roughness)).rg;

           float surfaceReduction = 1.0 - 0.28 * roughness * perceptualRoughness;
           float smoothness = 1 - perceptualRoughness;
           float grazingTerm = clamp((smoothness + (1 - kd)), 0.0, 1.0);

           // 镜面IBL贡献总额
           vec3 specularIBL =  surfaceReduction * specularIrradiance * FresnelLerp(F0, vec3(grazingTerm), nv);

           // 总环境照明贡献
           ambientLighting = diffuseIBL + specularIBL;
       }
       // 自发光颜色
       vec4 emission = getEmissionChannel();

       float alpha = 1 - kd + albedoTex.a * kd;
       // 最终像素点颜色
       color = vec4(directLighting + ambientLighting * ao + emission.rgb, alpha);
   }
   ```

2. 点光源像素shader模板定义如下（point_forward_pbr.frag）。

   {

       vec3 specularBRDF = (F * specularTerm);

   #endif

       // 最终像素点颜色
       float alpha = 1 - kd + albedoTex.a * kd;
       color = vec4(pointLighting, alpha);
   }

3. 聚光灯像素shader模板定义如下（spot_forward_pbr.frag）。

   ```screen
   #version 420 core

   precision highp float;
   precision mediump sampler2D;
   precision mediump samplerCube;

   const float PI = 3.141592;
   const float Epsilon = 0.00001;

   #define ENABLE_LIGHT

   // 光照数据结构（set、binding设置及数据结构与如下代码示例保持一致）

   void main()
   {
       // 采样输入纹理以获取着色模型参数
       vec4 albedoTex = getAlbedoChannel();
       vec3 albedo = albedoTex.rgb;
       vec3 orm = getPbrChannel().rgb;

       // 法向入射的菲涅尔反射率（对于金属，使用反照率颜色）
       vec3 F0 = mix(Fdielectric, albedo, metalness);

       // 聚光灯的光照计算  
       vec3 spotLighting = vec3(0);

       // 获取漫反射系数
       float kd = (1 - metalness) * 0.779083699;

   #ifdef ENABLE_LIGHT

       float theta = dot(Li, normalize(-lightInfo.direction.xyz));
       float epsilon = lightInfo.typeAngle.y - lightInfo.typeAngle.z;
       float intensity = clamp((theta - lightInfo.typeAngle.z) / epsilon, 0.0, 1.0);
       // 最终的光照强度
       Lradiance *= atten * intensity;

       // 入射光与出射光的半角向量
       vec3 Lh = normalize(Li + Lo);

       // 计算法线与各种光向量的点积
       float nl = max(0.0, dot(N, Li));
       float nh = max(0.0, dot(N, Lh));
       float hl = max(0.0, dot(Lh, Li));

       // 迪士尼漫反射项
       float diffuseTerm = DisneyDiffuse(nv, nl, hl, perceptualRoughness);
       vec3 diffuseBRDF = kd * albedo * diffuseTerm;

       // 菲涅尔系数
       vec3 F = fresnelSchlick(F0, hl);
       // 法线分布函数
       float D = ndfGGX(nh, roughness);
       // 几何函数
       float G = gaSchlickGGX(nl, nv, roughness);
       // 库克-托伦斯镜面BRDF，漫反射项不按1/PI缩放，镜面反射项按PI倍放大
       float specularTerm = (D * G) * PI / max(Epsilon, 4.0 * nl * nv);
       specularTerm = sqrt(max(0.0001, specularTerm));
       vec3 specularBRDF = (F * specularTerm);

       spotLighting += (diffuseBRDF + specularBRDF) * Lradiance * nl;
   #endif

       // 最终像素点颜色
       float alpha = 1 - kd + albedoTex.a * kd;
       color = vec4(spotLighting, alpha);
   }
   ```

## shader预编译离线脚本

CG Kit渲染框架目前仅支持预编译的shader二进制文件，所以用户编写的shader源码需要编译成二进制后使用，CG Kit提供了预编译脚本，供shader开发者参考。

### 环境准备

* 脚本执行支持的系统有Windows和Linux。
* 依赖Python环境，要求Python2或Python3。
* 依赖[glslang](https://github.com/KhronosGroup/glslang/releases/tag/master-tot)工具，用户根据不同的操作系统下载对应的执行文件。

### 脚本文件说明

1. Common.py

   该Python文件不能独立运行，提供公共函数的接口，定义支持的shaderStage类型。

   ```screen
   # coding=UTF-8
   import os
   import re

   # 打印日志
   def LOGINFO(info, *args):
       for arg in args:
           info += ' ' + str(arg)
       print (info)

   # 获取文件所在路径
   def GetDirPath(filePath):
       return os.path.normpath(os.path.dirname(filePath))

   # 获取完整文件路径（拼接路径和文件名）
   def GetFilePath(dir, filename):
       return os.path.normpath(os.path.join(dir, filename))

   # 获取文件名
   def GetFile(filePath):
       return os.path.split(filePath)[1]

   # 获取文件后缀
   def GetFileSuffix(file):
       return os.path.splitext(file)[-1][1:]

   # 获取文件名（无后缀）
   def GetFileName(file):
       return os.path.splitext(file)[0]

   # 获取文件的stage，单个文件返回后缀，多个文件检查是否为同一个stage
   def GetFileStage(file, *filePaths):
       resultStage = GetFileSuffix(file)
       if not filePaths:
           return resultStage

       for filePath in filePaths:
           if filePath != '':
               stage = GetFileSuffix(filePath)
               if resultStage != stage:
                   LOGINFO('Stages are different,', resultStage, stage)
                   return 'error'
       if resultStage notin shaderStages:
           LOGINFO('Input shader should end with .frag or .vert')
           return'error'
       return resultStage

   # 返回entryPoint所在文件行号，没找到则返回-1
   def SearchEntryPointInFile(filePath, entryPoint):
       searchString = r'\s?\(.*\)'
       try:
           with open(filePath, 'r') asfile:
               lineNumber = 0
               for line in file:
                   lineNumber += 1
                   match = re.search(searchString, line)
                   if match != None:
                       s = match.string[match.start():match.end()]
                       if entryPoint + s in line:
                           file.seek(0)
                           return lineNumber
               LOGINFO('Not find entryPoint \'' + entryPoint + '\' in file', filePath)
               return -1
       except:
           LOGINFO('Failed to read file,', filePath)
           return -2

   # 读取文件
   def ReadFile(filePath):
       try:
           with open(filePath, 'r') as file:
               return file.readlines()
       except:
           LOGINFO('Failed to read file', filePath)
           return []

   # 写入文件
   def WriteFile(content, filePath):
       try:
           with open(filePath, 'w+') as file:
               file.write(content)
               LOGINFO('Write file success', filePath)
               return True
       except:
           LOGINFO('Failed to write file', filePath)
           return False

   # 定义支持的shadeStage
   shaderStages = ['frag', 'vert']
   ```

2. ShaderConcatenate.py 该Python文件可以独立运行，支持自定义的shader文件拼接。通过命令行参数对两个文件拼接编译，必须输入一个参数为templateFile文件路径，其他命令参数为可选输入项。

   |可选输入项|说明|
   |:----|:-----------------------------|
   |-D|编译HLSL|
   |-e|指定glslang编译函数入口|
   |-h|打印帮助|
   |-s|shader函数实现文件（stage为vert或是frag）|

   * 针对拼接形式 输入第一个参数templateFile为"shaders/template.stage"，第二个参数shader函数实现文件为"shaders/shader.stage"。通过命令会将shader函数实现文件中的内容插入到template文件的main函数前，同时文件名进行拼接为shaders/template_shader.stage，最终编译得到的文件为shaders/template_shader.stage.spv。

     例如，参考命令行为python ShaderConcatenate.py "shaders/forward_pbr.frag" -s "shaders/sky.frag"。
   * 针对单个完整shader形式 输入templateFile为"shaders/templateAndShader.stage"会直接将该文件进行编译，最终编译得到的文件为templateAndShader.stage.spv。

   ```screen
   # coding=UTF-8
   import os
   import sys
   import getopt
   import platform
   import subprocess
   from Common import *

   isConcate = False
   concateSpvFile = ''
   concateShaderFile = ''

   # 打印帮助信息
   def PrintHelp():
       help = '''
   Usage: shaderConcatenate.py [templateFile] [option]...

   \'templateFile\' is the input template shader file, should end with .<stage>, where <stage> is:
      .vert  for a vertex shader
      .frag  for a fragment shader

   Options:
     -D         input is HLSL, for default will see input as GLSL
     -e <name>  specify <name> as the entry-point function name
                for default, will use \'main\' as entry-point function name
     -h         print this usage message
     -s <file>  use <file> as the shader function file
   '''
       LOGINFO(help)

   # 处理输入信息
   def HandleInput(opts, args):
       templateFilePath = ''
       shaderFilePath = ''
       entryPoint = ''
       isHLSL = False
       stage = ''

       if len(args) != 1:
           LOGINFO('Please input only one template file')
           PrintHelp()
           sys.exit(1)
       templateFilePath = args[0]

       # 根据输入命令处理信息
       for opt, arg in opts:
           if opt == '-h':
               PrintHelp()
               sys.exit()
           elif opt == '-D':
               isHLSL = True
           elif opt == '-e':
               entryPoint = arg
           elif opt == '-s':
               shaderFilePath = arg

       # 设置默认entryPoint
       if entryPoint == '':
           entryPoint = 'main'

       # 获取ShaderStage
       stage = GetFileStage(templateFilePath, shaderFilePath)
       if stage == 'error':
           PrintHelp()
           sys.exit(1)

       inputs = {
           'templateFilePath' : templateFilePath,
           'stage': stage,
           'entryPoint': entryPoint,
           'shaderFilePath' : shaderFilePath,
           'isHLSL' : isHLSL
       }

       return inputs

   # 拼接文件名
   def ConcateFileName(dir, suffix, *fileNames):
       concateName = ''
       for i in range(len(fileNames)):
           if i != len(fileNames) - 1:
               concateName += fileNames[i] + '_'
           else:
               concateName += fileNames[i]
       concateName += '.' + suffix
       return GetFilePath(dir, concateName)

   # 拼接文件路径，并组合成拼接文件名和输出spv文件名
   def ConcateFile(templateFilePath, shaderFilePath, stage):
       # 全局shader拼接文件名
       global concateShaderFile 
       # 全局输出spv文件名
       global concateSpvFile 
       # 是否为拼接结果
       global isConcate 
       dir = GetDirPath(templateFilePath)
       templateFileName = GetFileName(GetFile(templateFilePath))

       if shaderFilePath != '':
           shaderFileName = GetFileName(GetFile(shaderFilePath))

           concateShaderFile = ConcateFileName(dir, stage, templateFileName, shaderFileName)
           concateSpvFile = concateShaderFile + '.spv'
           isConcate = True
       else:
           concateShaderFile = ConcateFileName(dir, stage, templateFileName)
           concateSpvFile = concateShaderFile + '.spv'
           isConcate = False
       return

   # 拼接文件，将shader片段中的内容添加到template中entryPoint前，并输出到拼接文件
   def ConcateShader(templateFilePath, shaderFilePath, entryPoint):
       global concateShaderFile
       # 获取entryPoint函数入口位置
       mainPosition = SearchEntryPointInFile(templateFilePath, entryPoint)
       if mainPosition < 0:
           sys.exit(1)

       # 读取template文件
       templateLines = ReadFile(templateFilePath)
       if not templateFilePath:
           sys.exit(1)
       concateLines = templateLines

       # 读取shader片段文件
       if shaderFilePath != '':
           shaderLines = ReadFile(shaderFilePath)
           if not shaderLines:
               sys.exit(1)

           # 在template的main函数前插入shader片段中的内容，并储存拼接结果
           concateLines.insert(mainPosition - 1, ''.join(shaderLines))
           concatenate = ''.join(concateLines)
           if not WriteFile(concatenate, concateShaderFile):
               RemoveConcateFile()
               sys.exit(1)
       return

   '''
       构建glslang编译指令，并编译spv
       -S 区分stage
       -e 函数入口
       -o 输出文件路径名
       -V 输出spv
       （-D） 针对HLSL
   '''
   def BuildCommand(stage, entryPoint, isHLSL):
       # 全局shader拼接文件名
       global concateShaderFile 
       # 全局输出spv文件名
       global concateSpvFile 

       # 根据平台选择glslangValidator
       glslangCommand = ''
       if (platform.system() == 'Windows'):
           glslangCommand = 'glslangValidator.exe'
       elif (platform.system() == 'Linux'):
           glslangCommand = './glslangValidator'
       else:
           LOGINFO('Not supported system', platform.system())
           sys.exit(1)

       # 构建command命令
       command = [
           glslangCommand, concateShaderFile,
           '-S', stage,
           '-e', entryPoint,
           '-o', concateSpvFile,
           '-V']
       if isHLSL:
           command.append('-D')

       return command

   # 处理glslangValidator输出信息，如果有错误日志，以***开头打印
   def HandleOutput(output):
       result = False
       outputList = output.split('\n')
       fileName = outputList[0]
       if len(outputList) == 2:
           result = True
           return result, fileName, ''
       else:
           for i in range(len(outputList)):
               outputList[i] = '***' + outputList[i]
           return False, fileName, '\n'.join(outputList[1:-1])

   # 运行glslangValidator
   def RunGlslang(command):
       res = False
       try:
           output = subprocess.Popen(command, stdout=subprocess.PIPE).communicate()[0].decode()
           res, shaderFile, output = HandleOutput(output)
       except Exception as e:
           LOGINFO('Run glslangValidator failed')
           LOGINFO(e)
       finally:
           RemoveConcateFile()
       if res:
           LOGINFO('Compile success,', shaderFile)
       else:
           LOGINFO('Compile failed,', shaderFile)
           LOGINFO(output)
       return res

   # 删除拼接后的shader文件
   def RemoveConcateFile():
       # 全局shader拼接文件名
       global concateShaderFile 
       # 是否为拼接结果
       global isConcate 

       # 只删除拼接得到的shader文件，对于非拼接形式不删除文件
       if isConcate:
           if os.path.exists(concateShaderFile):
               os.remove(concateShaderFile)
               LOGINFO('Remove concateShaderFile success,', concateShaderFile)

   # 执行shader拼接接口函数
   def RunShaderConcatenate(templateFilePath, shaderFilePath, stage, entryPoint, isHLSL=False):
       shader = 'EMPTY'
       if shaderFilePath != '':
           shader = shaderFilePath

       # 获取文件路径，shaderStage拼接组合文件名作为最终的spv文件名
       ConcateFile(templateFilePath, shaderFilePath, stage)

       # 拼接shader
       ConcateShader(templateFilePath, shaderFilePath, entryPoint)
       LOGINFO('Template file:', templateFilePath)
       LOGINFO('Shader file:', shader)
       LOGINFO('Concate file name:', concateShaderFile)
       LOGINFO('Concate spriV file:', concateSpvFile)

       # 构建命令
       command = BuildCommand(stage, entryPoint, isHLSL)
       LOGINFO('Build command:', ' '.join(command))

       # 执行glslangValidator
       return RunGlslang(command)

   if __name__ == "__main__":
       try:
           opts, args = getopt.gnu_getopt(sys.argv[1:],'-D-h-e:-s:')
       except getopt.GetoptError:
           PrintHelp()
           sys.exit(1)

       # 处理输入信息
       inputs = HandleInput(opts, args)
       templateFilePath = inputs['templateFilePath']
       shaderFilePath = inputs['shaderFilePath']
       stage = inputs['stage']
       entryPoint = inputs['entryPoint']
       isHLSL = inputs['isHLSL']

       # 执行shader拼接编译
       res = RunShaderConcatenate(templateFilePath, shaderFilePath, stage, entryPoint, isHLSL)
       if res:
           sys.exit(0)
       else:
           sys.exit(2)
   ```


3. AutoConcat.py 该Python文件可独立运行，不支持命令行参数配置，自动使用shaderConcatenate.py的命令在shaders目录下寻找需要拼接的shader，并以目录作为最小单位进行shader拼接，对同一目录中的文件进行循环的匹配，不会跨目录匹配。根据目录和后缀名frag或者vert储存shader信息查找文件中的main，将有main函数的作为模板储存，其他的作为shader储存，其形成过程请参见[配置说明](#section61634615157)。

   构建命令时，脚本会遍历所有路径，查找所有template作为单个完整shader。对每一个template，遍历所有shader实现文件，将template与shader进行组合，最终会构建出如下组合形式。

   ```screen
   [
   ['shaders/directional_forward_pbr.frag', 'shaders/basic_pbr.frag'],
   ['shaders/point_forward_pbr.frag', 'shaders/basic_pbr.frag'],
   ['shaders/spot_forward_pbr.frag', 'shaders/basic_pbr.frag'],
   ['shaders/directional_forward_pbr.frag', 'shaders/sky.frag'],
   ['shaders/point_forward_pbr.frag', 'shaders/sky.frag'],
   ['shaders/spot_forward_pbr.frag', 'shaders/sky.frag'],
   ['shaders/forward_pbr.vert', shaders/static.vert']
   ]
   ```

   最终会编译出7个spv文件。AutoConcat.py源码如下。

   ```screen
   # coding=UTF-8
   import os
   import sys
   import getopt
   from Common import *
   from ShaderConcatenate import RunShaderConcatenate

   '''
   {
       'path':
           {
               'stage':
                   [
                       {'template':
                           [
                               'xxx.stage',
                               ...
                           ]
                       },
                       {'shader':
                           [
                               'yyy.stage',
                               ...
                           ]
                       }
                   ],
           },
       ...
   }
   '''
   # 遍历shaders目录，构建文件目录信息
   def GetFiles(shaderPath, entryPoint):
       filesDict = {}
       filesWalk = os.walk(shaderPath)
       # 遍历所有目录
       for root, _, files in filesWalk:
           for file in files:
               # 同一个目录中查找文件
               filePathRoot = os.path.normpath(root)
               if filePathRoot not in filesDict.keys():
                   filesDict[filePathRoot] = {}

               # 根据shaderStage构建文件信息
               stage = GetFileStage(file)
               if stage in shaderStages:
                   if stage not in filesDict[filePathRoot].keys():
                       filesDict[filePathRoot][stage] = {}
                       filesDict[filePathRoot][stage]['template'] = []
                       filesDict[filePathRoot][stage]['shader'] = []
                   # 获取完整文件路径
                   filePath = GetFilePath(filePathRoot, file)
                   # 搜索entryPoint，判定文件为template或是shader片段
                   mainPosition = SearchEntryPointInFile(filePath, entryPoint)
                   if mainPosition == -2:
                       sys.exit(1)
                   elif mainPosition == -1:
                       filesDict[filePathRoot][stage]['shader'].append(file)
                   else:
                       filesDict[filePathRoot][stage]['template'].append(file)
       return filesDict

   # 根据文件目录信息构建拼接组合
   def BuildCombines(filesDict):
       combines = []
       for dir, pathFiles in filesDict.items():
           for _, stageFiles in pathFiles.items():
               templateFiles = stageFiles['template']
               shaderFiles = stageFiles['shader']
               if len(templateFiles) == 0:
                   continue
               # 对每个template文件，遍历shader片段进行文件的拼接
               for template in templateFiles:
                   templateFile = GetFilePath(dir, template)
                   for shader in shaderFiles:
                       shaderFile = GetFilePath(dir, shader)
                       combines.append([templateFile, shaderFile])
       return combines

   if __name__ == "__main__":
       # 定义shaders目录和entryPoint
       shaderPath = 'shaders'
       entryPoint = 'main'

       # 获取文件目录并构建拼接组合
       filesDict = GetFiles(shaderPath, entryPoint)
       combines = BuildCombines(filesDict)

       # 执行shader拼接编译
       LOGINFO('\n--------Concate start--------\n')
       for templateFilePath, shaderFilePath in combines:
           stage = GetFileStage(templateFilePath, shaderFilePath)
           res = RunShaderConcatenate(templateFilePath, shaderFilePath, stage, entryPoint)
           if res:
               LOGINFO('ShaderConcatenate success,', templateFilePath, shaderFilePath, '\n')
           else:
               LOGINFO('ShaderConcatenate failed,', templateFilePath, shaderFilePath, '\n')

       LOGINFO('---------Concate end---------')
       sys.exit(0)
   ```
