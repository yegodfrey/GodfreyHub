支付接口调用 开发者请使用huaweipay.js的huaweiPay对象方法拉起支付页面。
    
    var orderInfo = {
            orderInfoStr: obj,
            isDistributed: type
        }
        var that = this
        huaweiPay.sub(ACTION_BUNDLE_NAME, orderInfo, function (data) {
            console.info("orderInfo: " + orderInfo)
            let res = JSON.parse(data)
            huaweiPay.unSub(ACTION_BUNDLE_NAME)
            if (res.returnCode == 0) {
                that.toast(res.data);
            } else {
                that.toast("支付失败")
            }
        })
