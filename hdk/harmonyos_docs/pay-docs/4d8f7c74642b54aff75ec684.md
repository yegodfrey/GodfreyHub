下载文件示例
    
    RestTemplate restTemplate = new RestTemplate(new HttpComponentsClientHttpRequestFactory(HttpClients.createSystem()));
    HttpHeaders header = new HttpHeaders();
    nspBaseResponse.getHeaders().forEach((key, value) -> {
             header.add(key, value);
    });
    HttpEntity<?> httpEntity = new HttpEntity(header);
    ResponseEntity<byte[]> responseEntity = restTemplate.exchange(nspBaseResponse.getUrl(), HttpMethod.GET, httpEntity, byte[].class, new Object[0]);
