异步通知结果验签
    
    log.info("Request uri:" + request.getRequestURI());
    // 1，读取消息体，校验格式，失败返回参数异常
    String reqBody = getRequestBody(request);
    if (StringUtils.isBlank(reqBody)) {
        log.error("RequestBody is empty");
        return BaseResponse.buildResp(ResultCodes.PARAM_EXCEPTION);
    }
    
    // 2, 消息验签
    String sign = request.getHeader("PayServSign");
    log.info("Request sign:" + sign);
    if (!verifySign(sign, reqBody)) {
        log.error("VerifySign failed");
        return BaseResponse.buildResp(ResultCodes.VERIFY_EXCEPTION);
    }
    
    // 3, 商户侧相关业务处理
    // 本处代码由商户自行实现，一般不抛出异常，内部处理支付结果通知
    try {
        TransResultCallbackNotify transOrderNotice =
            JSONObject.parseObject(reqBody, TransResultCallbackNotify.class);
        log.info("TransactionResultNotify request object convert success", transOrderNotice);
        doProcess(transOrderNotice);
    } catch (Exception e) {
        log.error("Business process failed", e);
    }
    // 4，处理完成，默认构造成功消息
    return BaseResponse.buildSuccessResp();
