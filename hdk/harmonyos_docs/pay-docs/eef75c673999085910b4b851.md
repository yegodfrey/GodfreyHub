配置build.gradle文件 在项目的build.gradle中，将libs目录作为依赖项。**注意** ：使用华为支付接口需要额外引入如下jar包。
    
    dependencies {
        // 添加下面的内容
        implementation fileTree(dir: "libs", include: ["*.aar"])
        // 其他依赖项
        implementation'com.google.code.gson:gson:2.8.6'
        implementation'com.mikesamuel:json-sanitizer:1.2.2'
        implementation'com.huawei.android.hms:security-ssl:1.1.5.308'
     }

在工程的build.gradle中添加华为开源仓库。
    
    maven { url 'https://developer.huawei.com/repo/' }
