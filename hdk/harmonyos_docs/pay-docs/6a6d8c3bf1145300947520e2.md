支付结果获取和处理 **同步返回** Js支付方法调用后，返回结果解析data数据即可，data数据内容格式参见参数说明中的PayResult对象。
    
    var orderInfo = {
            orderInfoStr: obj,
            isDistributed: type
        }
        var that = this
        huaweiPay.sub(ACTION_BUNDLE_NAME, orderInfo, function (data) {
            console.info("orderInfo: " + orderInfo)
            let res = JSON.parse(data)
            huaweiPay.unSub(ACTION_BUNDLE_NAME)
            if (res.returnCode == 0) {
                that.toast(res.data);
            } else {
                that.toast("支付失败")
            }
        })
