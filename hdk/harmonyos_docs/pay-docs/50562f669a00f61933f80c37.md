    StringauthId = DemoKeys.MERC_AUTH_ID;

    StringtraceId = KeyGenUtils.getTraceId();

    payMercAuth.setCallerId(mercNo);
    payMercAuth.setTraceId(traceId);
    payMercAuth.setTime(time);
    payMercAuth.setAuthId(authId);
    StringheaderSourceData = "authId=" + authId + "&bodySign=" + bodySign + "&callerId=" + mercNo + "&time=" + time + "&traceId=" + traceId;
    StringheaderSign = RSAUtils.signByRSAWithPSS(headerSourceData, priKey);
    payMercAuth.setHeaderSign(headerSign);
    payMercAuth.setBodySign(bodySign);
    StringPayMercAuthJsonStr = JSONObject.toJSONString(payMercAuth);
