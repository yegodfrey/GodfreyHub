参数签名示例  
      
    // 1,构造 MercOrderApply 对象
    MercOrder mercOrder = MercOrder.builder()
        .mercOrderNo("PetalCheckoutDemo_" + System.currentTimeMillis())
        .tradeSummary("衣服裤子等商品").mercNo(DemoKeys.MERC_NO)  // 商户号替换成自己的商户号
        .totalAmount(totalAmount)
        .currency("CNY")
        .goodsInfo(goodsInfo)  // goodsInfo字段请自行构建
        // .tradeExpireTime(getTradeExpireTime(30)) // 可控制订单过期时间，可不传，不传默认2小时(单机游戏不传，要使用必须传准确的UTC时间)
        .build();
    MercOrderApply apply = MercOrderApply.builder()
        .mercOrder(mercOrder)
        .allocationType("NO_ALLOCATION")// 添加分账类型
        .payload("衣服裤子等商品") .callbackUrl("https://***.com ")//服务器接收异步通知的回调    .authId("101110000023")  // 请填入正确的证书Id
        .signType("SHA256WithRSA/PSS").build();
    // 2, 进行对象签名
    Map<?, ?> contentMap = JSONObject.parseObject(JSONObject.toJSONString(apply), Map.class);
    String signContent = TradeSignToString.signString(contentMap);
    String privateKey = DemoKeys.MERC_PRIVATE_KEY;
    try {
        // 请参考此处完成订单参数签名
        String signedStr = RSAUtils.signByRSAWithPSS(signContent, privateKey);
        System.out.println("signedStr1=" + signedStr);
        apply.setSign(signedStr);
    } catch (Exception e) {
        e.printStackTrace();
    }
