SecureRandom 使用 示例：Random 创建随机数对象。 代码：
    
    import crypto.crypto.*
    main() {
        let r = SecureRandom()
        for (_ in 0..10) {
            let flip = r.nextBool()
            println(flip)
        }
        return 0
    }
