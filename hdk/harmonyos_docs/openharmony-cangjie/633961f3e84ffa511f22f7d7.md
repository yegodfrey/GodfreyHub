Xml DOM 模式使用
    
    import encoding.xml.*
    
    main() {
        let x: XmlParser = XmlParser()
    
        var ret = x.parse("<myxml>Some data </myxml>")
        match (ret) {
            case Some(root) => println(root.name)
            case None => println("XML Parse error.")
        }
        return 0
    }
