# Drive Kit手工转换指导书

#### 开发准备

您若需要应用支持Drive Kit，必须完成接入准备，请参见[开发准备](https://developer.huawei.com/consumer/cn/doc/development/HMSCore-Guides/config-agc-0000001050185569)。  

#### 登录方式变更

HMS使用华为帐号进行登录，并以此获取鉴权信息、证书等。获取HMS Credential的过程与Google API相差较大，需要手动转换。

下面提供了HMS登录阶段获取unionId、accessToken来生成Credential的示例代码，您可以参考代码进行修改。

Google API：

```
public class DriveQuickstart {
    private static final String APPLICATION_NAME = "Google Drive API Java Quickstart";
    private static final JsonFactory JSON_FACTORY = JacksonFactory.getDefaultInstance();
    private static final String TOKENS_DIRECTORY_PATH = "tokens";
    /**     
      * Global instance of the scopes required by this quickstart.     
      * If modifying these scopes, delete your previously saved tokens/ folder.    
      */
    private static final List<String> SCOPES = Collections.singletonList(DriveScopes.DRIVE_METADATA_READONLY);
    private static final String CREDENTIALS_FILE_PATH = "/credentials.json";
    /**     
      * Creates an authorized Credential object.     
      * @param HTTP_TRANSPORT The network HTTP Transport.     
      * @return An authorized Credential object.     
      * @throws IOException If the credentials.json file cannot be found.     
      */
    private static Credential getCredentials(final NetHttpTransport HTTP_TRANSPORT) throws IOException {
        // Load client secrets.
        InputStream in = DriveQuickstart.class.getResourceAsStream(CREDENTIALS_FILE_PATH);
        if (in == null) {
            throw new FileNotFoundException("Resource not found: " + CREDENTIALS_FILE_PATH);
        }
        GoogleClientSecrets clientSecrets = GoogleClientSecrets.load(JSON_FACTORY, new InputStreamReader(in));
        // Build flow and trigger user authorization request.
        GoogleAuthorizationCodeFlow flow = new GoogleAuthorizationCodeFlow.Builder(
            HTTP_TRANSPORT, JSON_FACTORY, clientSecrets, SCOPES)
            .setDataStoreFactory(new FileDataStoreFactory(new java.io.File(TOKENS_DIRECTORY_PATH)))
            .setAccessType("offline")
            .build();
        LocalServerReceiver receiver = new LocalServerReceiver.Builder().setPort(8888).build();
        return new AuthorizationCodeInstalledApp(flow, receiver).authorize("user");
    }
}

HMS API示例代码：

```
public class LoginActivity extends Activity implements View.OnClickListener {
    private static DriveCredential.AccessMethod refreshAT = new DriveCredential.AccessMethod() {
        @Override
        public String refreshToken() {
            return HmsProxyImpl.getInstance().refreshAccessToken();
        }
    };
	@Override
    public void onClick(View v) {
        if (R.id.account_login_button == v.getId()) {
            HmsProxyImpl.getInstance().signIn(this);
        }
    }
    @Override
    protected void onActivityResult(int requestCode, int resultCode, Intent data) {
        super.onActivityResult(requestCode, resultCode, data);
        switch (requestCode) {
            case HmsProxyImpl.REQUEST_SIGN_IN_LOGIN:
                com.huawei.hmf.tasks.Task<AuthHuaweiId> signInHuaweiIdTask = HuaweiIdAuthManager.parseAuthResultFromIntent(data);
                if (signInHuaweiIdTask.isSuccessful()) {
                    AuthHuaweiId huaweiAccount = signInHuaweiIdTask.getResult();
                    HmsProxyImpl.getInstance().dealSignInResult(huaweiAccount);
                    initDrive();
                } else {
                    finish();
                }
                break;
            default:
                break;
        }
    }
    private void initDrive() {
        final String unionID = HmsProxyImpl.getInstance().getUnionId();
        final String at = HmsProxyImpl.getInstance().getAccessToken();
        TaskManager.getInstance().execute(new Task() {
            @Override
            public void call() {
                int returnCode = CredentialManager.getInstance().init(unionID, at, refreshAT);
                // go to the main activity
                ....
            }
        });
    }
}
public class HmsProxyImpl {
    private HuaweiIdAuthParams signInOptions;
    private static HmsProxyImpl instance = new HmsProxyImpl();
    public static final int REQUEST_SIGN_IN_LOGIN = 1002;
    private ReentrantLock getATLock = new ReentrantLock();
    private HuaweiIdAuthService client;
    private String accessToken;
    private String unionId;
    private Activity activity;
    private HmsProxyImpl() {
        signInOptions = initData();
    }
    public static HmsProxyImpl getInstance() {
        return instance;
    }
    private HuaweiIdAuthParams initData() {
        List<Scope> scopeList = new LinkedList<>(Arrays.asList(
                HuaweiIdAuthAPIManager.HUAWEIID_BASE_SCOPE,
                new Scope(DriveScopes.SCOPE_DRIVE),
                new Scope(DriveScopes.SCOPE_DRIVE_METADATA),
                new Scope(DriveScopes.SCOPE_DRIVE_FILE),
                new Scope(DriveScopes.SCOPE_DRIVE_APPDATA)));
        HuaweiIdAuthParams authParams = new HuaweiIdAuthParamsHelper(HuaweiIdAuthParams.DEFAULT_AUTH_REQUEST_PARAM)
                .setScopeList(scopeList)        
                .setIdToken()        
                .setAccessToken()        
                .createParams();
        return authParams ;
    }
    public void signIn(Activity activity) {
        this.activity = activity;
        if (signInOptions == null) {
            signInOptions = initData();
        }
        client = HuaweiIdAuthManager.getService(activity, signInOptions);
        activity.startActivityForResult(client.getSignInIntent(), REQUEST_SIGN_IN_LOGIN);
    }
    public String refreshAccessToken() {
        try {
            if (client != null) {
                getATLock.lock();
                try {
                    getAT();
                } finally {
                    getATLock.unlock();
                }
            } 
        } catch (Exception e) {
            // deal Exception
        }
        return accessToken;
    }
    private void getAT() {
        for (int retry = 0; retry < 2; retry++) {
            if (signInBackend()) {
                break;
            }
        }
    }
    private boolean signInBackend(){
        clearAccountInfo();
        final CountDownLatch countDownLatch = new CountDownLatch(1);
        com.huawei.hmf.tasks.Task<AuthHuaweiId> task = client.silentSignIn();
        task.addOnSuccessListener(new OnSuccessListener<AuthHuaweiId>() {
            @Override
            public void onSuccess(AuthHuaweiId signInHuaweiId) {
                dealSignInResult(signInHuaweiId);
                countDownLatch.countDown();
            }
        });
        task.addOnFailureListener(new OnFailureListener() {
            @Override
            public void onFailure(Exception e) {
                countDownLatch.countDown();
            }
        });
        try {
            countDownLatch.await(15, TimeUnit.SECONDS);
        }catch (InterruptedException e) {
            countDownLatch.countDown();
        }
        if (TextUtils.isEmpty(getAccessToken())) {
            return false;
        } else {
            return true;
        }
    }
    public void dealSignInResult(AuthHuaweiId huaweiAccount){
        String tempAt = huaweiAccount.getAccessToken();
        if (null == tempAt || tempAt.isEmpty()) {
            return;
        }
        saveAccountInfo(huaweiAccount);
    }
    private void saveAccountInfo(AuthHuaweiId signInHuaweiId) {
        if (signInHuaweiId == null) {
            return;
        }
        unionId = signInHuaweiId.getUnionId();
        accessToken = signInHuaweiId.getAccessToken();
    }
    private void clearAccountInfo() {
        unionId = null;
        accessToken = null;
    }
    public String getAccessToken() {
        return this.accessToken;
    }
    public String getUnionId() {
        return unionId;
    }
}
public class CredentialManager {
    /**
     * Drive credential authorization
     */
    private DriveCredential mCredential;
    private CredentialManager() {
    }
    private static class InnerHolder {
        private static CredentialManager sInstance = new CredentialManager();
    }
    public static CredentialManager getInstance() {
        return InnerHolder.sInstance;
    }
    /**
     * init the drive sdk with access token and HwId Account info,
     * register a AccessMethod to get a new accessToken while accessToken is expired.
     *
     * @param unionID   unionID from HwID
     * @param at        access token
     * @param refreshAT a callback to refresh AT
     */
    public int init(String unionID, String at, DriveCredential.AccessMethod refreshAT) {
        if (StringUtils.isNullOrEmpty(unionID) || StringUtils.isNullOrEmpty(at)) {
            return DriveCode.ERROR;
        }
        DriveCredential.Builder builder = new DriveCredential.Builder(unionID, refreshAT);
        mCredential = builder.build().setAccessToken(at);
        return DriveCode.SUCCESS;
    }
    public DriveCredential getCredential() {
        return mCredential;
    }
}

Drive的构造需要在登录时初始化Credential信息，HMS和Google API初始化Credential的内容请参见[登录方式变更](#section14101144101419)。
