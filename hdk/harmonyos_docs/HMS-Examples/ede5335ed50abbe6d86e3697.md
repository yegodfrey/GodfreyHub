        4. package com.huawei.game.gamekitdemo;
        5. import android.app.Activity;
        6. import android.os.Bundle;
        7. import android.os.Handler;
        8. import android.os.HandlerThread;
        9. import android.os.Message;
        10. import android.view.View;
        11. import android.widget.Button;
        12. import android.widget.Toast;
        13. import com.huawei.game.gamekit.GameManager;
        14. import com.huawei.game.gamekit.c;
        15. import com.huawei.game.gamekitdemo.utils.LogUtil;
        16. import org.json.JSONException;
        17. import org.json.JSONObject;
        18. /**
        19. * DemoMainActivity
        20. * This Demo just show how to use all APIs in GameKit, it cannot be authenticated by the system cause it is not a game application.
        21. *
        22. * @since 2019-09-05
        23. */
        24. public class DemoMainActivity extends Activity implements View.OnClickListener {
        25.     private static final String TAG = "DemoMainActivity";
        26.     private static final int MSG_REGISTER_WITH_CALLBACK = 0;
        27.     private static final int MSG_REGISTER_WITHOUT_CALLBACK = 1;
        28.     private static final int MSG_SEND_DATA = 2;
        29.     private static final int MSG_GET_PHONE_INFO = 3;
        30.     private HandlerThread mWorkThread;
        31.     private Handler mHandler;
        32.     private GameManager.GameCallBack gameSdkCallBack = new GameManager.GameCallBack() {
        33.     @Override
        34.     public void onPhoneInfoUpdated(String info) {
        35.         LogUtil.i(TAG, "onPhoneInfoUpdated: " + info);
        36.         Toast.makeText(DemoMainActivity.this, info, Toast.LENGTH_LONG).show();
        37.         }
        38.     };
        39.     private GameManager mGameKit = GameManager.getGameManager();
        40.     @Override
        41.     protected void onCreate(Bundle savedInstanceState) {
        42.         super.onCreate(savedInstanceState);
        43.         setContentView(R.layout.activity_demo_main);
        44.         initView();
        45.         initThread();
        46.     }
        47.     @Override
        48.     protected void onDestroy() {
        49.         super.onDestroy();
        50.         mWorkThread.quit();
        51.     }
        52.     @Override
        53.     public void onClick(View v) {
        54.         Message msg = mHandler.obtainMessage();
        55.         switch (v.getId()) {
        56.         case R.id.register_sdk_with_callback:
        57.             msg.what = MSG_REGISTER_WITH_CALLBACK;
        58.             mHandler.sendMessage(msg);
        59.         break;
        60.         case R.id.register_sdk_without_callback:
        61.             msg.what = MSG_REGISTER_WITHOUT_CALLBACK;
        62.             mHandler.sendMessage(msg);
        63.         break;
        64.         case R.id.get_phone_info:
        65.             msg.what = MSG_GET_PHONE_INFO;
        66.             mHandler.sendMessage(msg);
        67.         break;
        68.         case R.id.send_data:
        69.             msg.what = MSG_SEND_DATA;
        70.             mHandler.sendMessage(msg);
        71.         break;
        72.         default:
        73.         break;
        74.         }
        75.     }
        76.     private void initView() {
        77.         final Button registerWithCb = findViewById(R.id.register_sdk_with_callback);
        78.         registerWithCb.setOnClickListener(this);
        79.         final Button registerWithoutCb = findViewById(R.id.register_sdk_without_callback);
        80.         registerWithoutCb.setOnClickListener(this);
        81.         Button getPhoneInfo = findViewById(R.id.get_phone_info);
        82.         getPhoneInfo.setOnClickListener(this);
        83.         final Button send = findViewById(R.id.send_data);
        84.         send.setOnClickListener(this);
        85.     }
        86.     private void initThread() {
        87.         mWorkThread = new HandlerThread("WorkThread");
        88.         mWorkThread.start();
        89.         mHandler = new Handler(mWorkThread.getLooper(), new Handler.Callback() {
        90.         @Override
        91.         public boolean handleMessage(Message message) {
        92.             switch (message.what) {
        93.                 case MSG_REGISTER_WITH_CALLBACK:
        94.                     registerKitWithCallback();
        95.                 break;
        96.                 case MSG_REGISTER_WITHOUT_CALLBACK:
        97.                     registerKitWithoutCallback();
        98.                 break;
        99.                 case MSG_SEND_DATA:
        100.                     sendData();
        101.                 break;
        102.                 case MSG_GET_PHONE_INFO:
        103.                     getPhoneInfo();
        104.                 break;
        105.                 default:
        106.                 break;
        107.             }
        108.             return false;
        109.             }
        110.         });
        111.     }
        112.     /**
        113.     * Register Kit with callback.
        114.     * If the API returns a failure, the reason is most likely that this demo is not a game application.
        115.     * You can get info of this device by callback you passed.
        116.     * The result is same with that call {@link com.huawei.game.gamekit.GameManager#getPhoneInfo()}
        117.     */
        118.     private void registerKitWithCallback() {
        119.         boolean isSuccess = mGameKit.registerGame(getPackageName(), gameSdkCallBack);
        120.         Toast.makeText(this, "register result:" + isSuccess, Toast.LENGTH_LONG).show();
        121.     }
        122.     /**
        123.     * Register Kit without callback.
        124.     * If the API returns a failure, the reason is most likely that this demo is not a game application.
        125.     */
        126.     private void registerKitWithoutCallback() {
        127.         boolean isSuccess = mGameKit.registerGame(getPackageName(), null);
        128.         Toast.makeText(this, "register result:" + isSuccess, Toast.LENGTH_LONG).show();
        129.     }
        130.     /**
        131.     * You can get info of the device by this method.
        132.     * The result is same with that callback {@link com.huawei.game.gamekit.GameManager.GameCallBack#onPhoneInfoUpdated(String)}
        133.     */
        134.     private void getPhoneInfo() {
        135.         String phoneInfo = mGameKit.getPhoneInfo();
        136.         Toast.makeText(this, "phoneInfo:" + phoneInfo, Toast.LENGTH_LONG).show();
        137.     }
        138.     /**
        139.     * Call this method,send necessary info to system for scheduling of resources
        140.     */
        141.     private void sendData() {
        142.         String str = getDataString();
        143.         if (str != null) {
        144.             mGameKit.updateGameAppInfo(str);
        145.             Toast.makeText(this, "sendData:" + str, Toast.LENGTH_LONG).show();
        146.         }
        147.     }
        148.     private String getDataString() {
        149.         JSONObject jsonObject = new JSONObject();
        150.         try {
        151.             jsonObject.put("MessageType", 3);
        152.             jsonObject.put("SceneID", 1);
        153.             jsonObject.put("Description", "Game Start");
        154.             jsonObject.put("ImportantLevel", 2);
        155.             jsonObject.put("RecommendFps", 30);
        156.             jsonObject.put("KeyThread", "net|6001");
        157.         } catch (JSONException e) {
        158.             return null;
        159.         }
        160.             return jsonObject.toString();
        161.         }
        162. }
