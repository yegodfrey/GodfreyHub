        54.     /**
        55.      * 普通支付示例 | Common payment Examples
        56.      */
        57.     private void pay() {
        58.         showLog("pay: begin");
        59.         EditText etAmount = (EditText) findViewById(R.id.et_amount);
        60.         float amount = 0;
        61.         if (etAmount != null) {
        62.             String text = etAmount.getText().toString();
        63.             if (!TextUtils.isEmpty(text)) {
        64.                 try {
        65.                     amount = Float.valueOf(text);
        66.                 } catch (NumberFormatException e) {
        67.                     showLog("price format error");
        68.                 }
        69.             }
        70.         }
        71.         PayReq payReq = createPayReq(amount);
        72.         HMSAgent.Pay.pay(payReq, new PayHandler() {
        73.             @Override
        74.             public void onResult(int retCode, PayResultInfo payInfo) {
        75.                 if (retCode == HMSAgent.AgentResultCode.HMSAGENT_SUCCESS && payInfo != null) {
        76.                     boolean checkRst = PaySignUtil.checkSign(payInfo, pay_pub_key);
        77.                     showLog("pay: onResult: pay success and checksign=" + checkRst);
        78.                     if (checkRst) {
        79.                         // 支付成功并且验签成功，发放商品 | Payment successful and successful verification, distribution of goods
        80.                     } else {
        81.                         // 签名失败，需要查询订单状态：对于没有服务器的单机应用，调用查询订单接口查询；其他应用到开发者服务器查询订单状态。| Signature failed, need to query order status: For stand-alone applications without servers, call Query order interface query, other application to the Developer Server query order status.
        82.                     }
        83.                 } else if (retCode == HMSAgent.AgentResultCode.ON_ACTIVITY_RESULT_ERROR
        84.                         || retCode == PayStatusCodes.PAY_STATE_TIME_OUT
        85.                         || retCode == PayStatusCodes.PAY_STATE_NET_ERROR) {
        86.                     // 需要查询订单状态：对于没有服务器的单机应用，调用查询订单接口查询；其他应用到开发者服务器查询订单状态。 | Pay failed, need to query order status: For stand-alone applications without servers, call Query order interface query, other application to the Developer Server query order status.
        87.                 } else {
        88.                     showLog("pay: onResult: pay fail=" + retCode);
        89.                     // 其他错误码意义参照支付api参考 | Other error code meaning reference payment API reference
        90.                 }
        91.             }
        92.         });
        93.         // 将requestid缓存，供查询订单 | RequestID Cache for Query order
        94.         addRequestIdToCache(payReq.getRequestId());
        95.     }
        96.     private void addRequestIdToCache(String requestId) {
        97.         SharedPreferences sp = getSharedPreferences("pay_request_ids", 0);
        98.         sp.edit().putBoolean(requestId, false).commit();
        99.     }
        100.     /**
        101.      * 创建普通支付请求对象 | Create an ordinary Payment request object
        102.      *
        103.      * @param totalAmount 要支付的金额 | Amount to pay
        104.      * @return 普通支付请求对象 | Ordinary Payment Request Object
        105.      */
        106.     private PayReq createPayReq(float totalAmount) {
        107.         PayReq payReq = new PayReq();
        108.         /**
        109.          * 生成requestId | Generate RequestID
        110.          */
        111.         DateFormat format = new java.text.SimpleDateFormat("yyyyMMddhhmmssSSS");
        112.         int random = new SecureRandom().nextInt() % 100000;
        113.         random = random < 0 ? -random : random;
        114.         String requestId = format.format(new Date());
        115.         requestId = String.format("%s%05d", requestId, random);
        116.         /**
        117.          * 生成总金额 | Generate Total Amount
        118.          */
        119.         String amount = String.format("%.2f", totalAmount);
        120.         //商品名称 | Product Name
        121.         payReq.productName = "test product";
        122.         //商品描述 | Product Description
        123.         payReq.productDesc = "test product description";
        124.         // 商户ID，来源于开发者联盟，也叫“支付id” | Merchant ID, from the Developer Alliance, also known as "Payment ID"
        125.         payReq.merchantId = cpId;
        126.         // 应用ID，来源于开发者联盟 | Application ID, from the Developer Alliance
        127.         payReq.applicationID = appId;
        128.         // 支付金额 | Amount paid
        129.         payReq.amount = amount;
        130.         // 支付订单号 | Payment order Number
        131.         payReq.requestId = requestId;
        132.         // 国家码 | Country code
        133.         payReq.country = "CN";
        134.         //币种 | Currency
        135.         payReq.currency = "CNY";
        136.         // 渠道号 | Channel number
        137.         payReq.sdkChannel = 1;
        138.         // 回调接口版本号 | Callback Interface Version number
        139.         payReq.urlVer = "2";
        140.         // 商户名称，必填，不参与签名。会显示在支付结果页面 | Merchant name, must be filled out, do not participate in the signature. will appear on the Pay results page
        141.         payReq.merchantName = "XXX Company (set as your company name here)";
        142.         //分类，必填，不参与签名。该字段会影响风控策略 | Categories, required, do not participate in the signature. This field affects wind control policy
        143.         // X4：主题,X5：应用商店,    X6：游戏,X7：天际通,X8：云空间,X9：电子书,X10：华为学习,X11：音乐,X12 视频, | X4: Theme, X5: App Store, X6: Games, X7: Sky Pass, X8: Cloud Space, X9: ebook, X10: Huawei Learning, X11: Music, X12 video,
        144.         // X31 话费充值,X32 机票/酒店,X33 电影票,X34 团购,X35 手机预购,X36 公共缴费,X39 流量充值 | X31, X32 air tickets/hotels, X33 movie tickets, X34 Group purchase, X35 mobile phone advance, X36 public fees, X39 flow Recharge
        145.         payReq.serviceCatalog = "X6";
        146.         //商户保留信息，选填不参与签名，支付成功后会华为支付平台会原样 回调CP服务端 | The merchant retains the information, chooses not to participate in the signature, the payment will be successful, the Huawei payment platform will be back to the CP service
        147.         payReq.extReserved = "Here to fill in the Merchant reservation information";
        148.         //对单机应用可以直接调用此方法对请求信息签名，非单机应用一定要在服务器端储存签名私钥，并在服务器端进行签名操作。| For stand-alone applications, this method can be called directly to the request information signature, not stand-alone application must store the signature private key on the server side, and sign operation on the server side.
        149.         // 在服务端进行签名的cp可以将getStringForSign返回的待签名字符串传给服务端进行签名 | The CP, signed on the server side, can pass the pending signature string returned by Getstringforsign to the service side for signature
        150.         payReq.sign = PaySignUtil.rsaSign(PaySignUtil.getStringForSign(payReq), pay_priv_key);
        151.         return payReq;
        152.     }

        172.         }
        173.     }
        174. }
