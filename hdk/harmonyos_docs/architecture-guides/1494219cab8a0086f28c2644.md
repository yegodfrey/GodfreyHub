# 二次授权弹窗反复弹出问题总结

## 场景介绍

[权限场景化申请](https://developer.huawei.com/consumer/cn/doc/best-practices/bpta-permission-application)是便捷生活类等应用的高频使用场景之一，如用户首次使用导航、扫码、录音等需要授权的功能时，应用需在合适的时机引导用户完成权限授予。

本示例主要基于[@ohos.abilityAccessCtrl(程序访问控制管理)](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/js-apis-abilityaccessctrl)实现权限的场景化申请、按dialogShownResults分级处理拒绝结果、通过requestPermissionOnSetting二次引导，并结合[@kit.ArkData的用户首选项](https://developer.huawei.com/consumer/cn/doc/harmonyos-guides/data-persistence-by-preferences)持久化"不再提醒"与"系统已禁止弹框"标记，避免反复弹出权限申请弹窗。

## 实现思路

示例围绕"场景化触发+先说明后申请+分级处理拒绝+状态持久化+权限管理页兜底"的完整闭环实现，核心逻辑拆分为三个部分。

1. **权限申请分级封装（PermissionManager）**

   封装requestFromUser方法，在调用[requestPermissionsFromUser](https://developer.huawei.com/consumer/cn/doc/harmonyos-references/js-apis-abilityaccessctrl)后根据authResults与dialogShownResults区分三种结果，避免对"系统已不再弹框"状态重复发起无效调用：

   ```ts
   import { abilityAccessCtrl, bundleManager, common, Permissions } from '@kit.AbilityKit';

   /**
   * 首次申请结果：
   * - granted：用户已授权（或此前已授权）
   * - deniedWithDialog：本次弹了系统弹框，但用户拒绝。此后系统对该权限不再弹框，
   * 与deniedNoDialog等价，应记录SYSTEM_BLOCKED标记，用户再次主动触发时直接二次引导
   * - deniedNoDialog：系统未弹框（用户此前已选"不允许"），当次即走requestOnSetting二次引导
   */
   export type RequestOutcome = 'granted' | 'deniedWithDialog' | 'deniedNoDialog';

   /**
   * 权限管理工具类，封装官方推荐流程：
   * checkAccessToken先检查->requestPermissionsFromUser首次申请
   * ->按dialogShownResults分级处理->requestPermissionOnSetting二次引导
   * 参考：最佳实践《应用权限申请》 https://developer.huawei.com/consumer/cn/doc/best-practices/bpta-permission-application
   */
   export class PermissionManager {
     private static readonly atManager: abilityAccessCtrl.AtManager = abilityAccessCtrl.createAtManager();

   /** 检查一组权限是否已全部授权 */
     static async isGranted(permissions: Permissions[]): Promise<boolean> {
       const bundleInfo = bundleManager.getBundleInfoForSelfSync(
         bundleManager.BundleFlag.GET_BUNDLE_INFO_WITH_APPLICATION);
       const tokenID = bundleInfo.appInfo.accessTokenId;
       for (const permission of permissions) {
         const status = await PermissionManager.atManager.checkAccessToken(tokenID, permission);
         if (status !== abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED) {
           return false;
         }
       }
       return true;
     }

    /** 首次向用户申请授权（仅在用户触发对应功能场景时调用，禁止启动时批量调用） */
     static async requestFromUser(context: common.UIAbilityContext,
       permissions: Permissions[]): Promise<RequestOutcome> {
       const result = await PermissionManager.atManager.requestPermissionsFromUser(context, permissions);
       const allGranted =
         result.authResults.every((item) => item === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED);
       if (allGranted) {
         return 'granted';
       }
       const dialogShown = result.dialogShownResults?.some((shown) => shown) ?? false;
       return dialogShown ? 'deniedWithDialog' : 'deniedNoDialog';
     }

    /**
   * 二次引导：直接拉起权限设置弹框。
   * 注意：仅在系统对该权限不再弹框（已记录SYSTEM_BLOCKED标记）
   * 且用户再次主动点击功能入口时调用。
   */
     static async requestOnSetting(context: common.UIAbilityContext,
       permissions: Permissions[]): Promise<boolean> {
       const statuses = await PermissionManager.atManager.requestPermissionOnSetting(context, permissions);
       return statuses.every((item) => item === abilityAccessCtrl.GrantStatus.PERMISSION_GRANTED);
     }
   }
   ```

2. **用户偏好持久化（PermissionStore）**

   使用[@kit.ArkData的preferences](https://developer.huawei.com/consumer/cn/doc/harmonyos-guides/data-persistence-by-preferences)持久化两类标记：用户勾选的"不再提醒"(MUTED状态)与系统层"已禁止申请弹框"(SYSTEM_BLOCKED状态)，保证重启后状态依然成立：

   ```ts
   import { preferences } from '@kit.ArkData';
   import { common } from '@kit.AbilityKit';

   const STORE_NAME = 'permission_guide_store';

   /**
   * 应用内权限弹窗的用户偏好存储。
   * 只有一种情况不再弹应用内权限弹窗：用户在说明弹窗中勾选了"不再提醒"
   * （可在"权限管理"页的"应用内弹窗提醒"开关中恢复）。
   * "暂不开启"不记录任何状态，下次触发功能时照常弹窗。
   */
   export class PermissionStore {
     private static store: preferences.Preferences | null = null;

     private static getStore(context: common.Context): preferences.Preferences {
       if (!PermissionStore.store) {
         PermissionStore.store = preferences.getPreferencesSync(context, { name: STORE_NAME });
       }
       return PermissionStore.store;
     }

   /** 用户是否勾选了"不再提醒"（权限管理页中的"应用内弹窗提醒"开关为其反值） */
     static isPromptDisabled(context: common.Context, sceneKey: string): boolean {
       return PermissionStore.getStore(context).getSync(`prompt_disabled_${sceneKey}`, false) as boolean;
     }

     static setPromptDisabled(context: common.Context, sceneKey: string, disabled: boolean): void {
       const store = PermissionStore.getStore(context);
       store.putSync(`prompt_disabled_${sceneKey}`, disabled);
       void store.flush();
     }

     /**
   * 系统层是否已禁止该权限的申请弹框（系统弹框被拒绝或dialogShownResults=false，此后系统不再弹框）。
   * 该状态重启后依然成立（系统侧未变），需持久化；授权成功的任意路径应清除。
   */
     static isSystemBlocked(context: common.Context, sceneKey: string): boolean {
       return PermissionStore.getStore(context).getSync(`system_blocked_${sceneKey}`, false) as boolean;
     }

     static setSystemBlocked(context: common.Context, sceneKey: string, blocked: boolean): void {
       const store = PermissionStore.getStore(context);
       store.putSync(`system_blocked_${sceneKey}`, blocked);
       void store.flush();
     }
   }
   ```

3. **场景化触发与状态机调度（Index页面）**

   页面定义权限场景数组（一个权限只绑定一个真实功能场景），用户点击功能按钮时按状态机分发：已授权直接使用->MUTED仅Toast提示->SYSTEM_BLOCKED直接拉起系统授权弹框->其他弹说明弹窗（含"不再提醒"勾选），核心调度逻辑如下：

   ```ts
   import { common, Permissions } from '@kit.AbilityKit';
   import { PermissionManager } from '../utils/PermissionManager';
   import { PermissionStore } from '../utils/PermissionStore';

   interface PermissionScene {
     key: string;
     title: string;
     usage: string;
     benefit: string;
     permissions: Permissions[];
   }

   const SCENES: PermissionScene[] = [
     {
       key: 'location',
       title: '位置信息',
       usage: '用于获取当前位置，提供基于位置的服务（如附近内容、导航）',
       benefit: '开启后可使用基于位置的功能',
       permissions: ['ohos.permission.LOCATION', 'ohos.permission.APPROXIMATELY_LOCATION']
     },
     {
       key: 'camera',
       title: '相机',
       usage: '用于扫码、拍照和上传图像',
       benefit: '开启后可使用扫码和拍照功能',
       permissions: ['ohos.permission.CAMERA']
     },
     {
       key: 'microphone',
       title: '麦克风',
       usage: '用于录制语音和音频',
       benefit: '开启后可使用语音相关功能',
       permissions: ['ohos.permission.MICROPHONE']
     },
     {
       key: 'motion',
       title: '运动健康',
       usage: '用于读取您的运动健康数据，统计运动状态',
       benefit: '开启后可生成运动报告（非必需，拒绝不影响其他功能）',
       permissions: ['ohos.permission.ACTIVITY_MOTION']
     }
   ];
   ```

   申请结果的分级处理在requestPermission方法中完成：granted清除全部标记。deniedWithDialog与deniedNoDialog均持久化SYSTEM_BLOCKED标记，系统弹框被拒绝一次后，系统对该权限不再弹框，两种拒绝结果在状态上等价。其中deniedNoDialog当次即调用requestOnSetting拉起系统授权弹框完成二次引导。已记录SYSTEM_BLOCKED标记的场景，用户再次点击功能入口时不再弹应用内说明弹窗，直接拉起系统授权弹框，减少用户操作步骤。核心处理代码如下。

   ```ts
   // 功能入口点击分发：SYSTEM_BLOCKED状态下直接拉起系统授权弹框
   if (PermissionStore.isSystemBlocked(context, scene.key)) {
     void this.openSystemSetting(scene);
     return;
   }

   private async requestPermission(scene: PermissionScene): Promise<void> {
     const context = this.getContext();
     const outcome = await PermissionManager.requestFromUser(context, scene.permissions);
     if (outcome === 'granted') {
      // 清除全部标记并刷新状态
   // ...
       return;
     }
     // 两种拒绝结果均记录SYSTEM_BLOCKED标记，系统此后不再弹框
     PermissionStore.setSystemBlocked(context, scene.key, true);
     if (outcome === 'deniedNoDialog') {
       // 系统未弹框，当次即拉起系统授权弹框完成二次引导
       await this.openSystemSetting(scene);
     }
   }
   ```
