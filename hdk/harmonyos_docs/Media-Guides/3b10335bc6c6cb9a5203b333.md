# 播放状态回调

播放状态监听器接口[HwAudioStatusListener](https://developer.huawei.com/consumer/cn/doc/development/Media-References/hwaudiostatuslistener-0000001050191645)，可用于歌曲信息、歌曲队列、缓冲进度等发生变化时进行页面刷新。

```screen
// 注册播放状态的监听器。 
mHwAudioManager.addPlayerStatusListener(mPlayListener);
// 移除播放状态的监听器。
mHwAudioManager.removePlayerStatusListener(mPlayListener);

private HwAudioStatusListener mPlayListener = new HwAudioStatusListener() {

    @Override
    public void onSongChange(HwAudioPlayItem song) {
    // 当前音频变化回调。
    }

    @Override
    public void onQueueChanged(List<HwAudioPlayItem> infos) {
    // 队列变化回调。
    }

    @Override 
    public void onBufferProgress(int percent) {
    // 缓冲进度变化回调。
    }

    @Override       
    public void onPlayProgress(long currPos, long duration) {
    // 播放进度变化回调。
    }

    @Override       
    public void onPlayCompleted(boolean isStopped) {
    // 播放完成回调。
    }

    @Override      
    public void onPlayError(int errorCode, boolean isUserForcePlay) {
    // 播放出错回调。
    }

    @Override      
    public void onPlayStateChange(boolean isPlaying, boolean isBuffering) {
    // 播放状态变化回调。
    }
};
```
