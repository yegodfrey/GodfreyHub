package com.lrtest.shareproductpro;

import android.content.Context;
import android.graphics.Point;
import android.util.Log;

import com.huawei.caas.messageservice.HwCaasShareCallBack;
import com.huawei.caas.messageservice.HwCaasShareHandler;
import com.huawei.caas.messageservice.HwCaasShareManager;
import com.huawei.caas.messageservice.HwShareMsgInfo;
import com.huawei.caas.messageservice.HwShareUtils;

public class ShareHelper {
    private static final String TAG = "ShareHelper";

    private HwCaasShareManager hwCaasShareManager;

    private HwCaasShareHandler shareHandler;

    private Context mContext;

    private HwCaasShareCallBack mCallBack = new HwCaasShareCallBack() {
        @Override
        public void initSuccess(HwCaasShareHandler hwCaasShareHandler) {
            if (hwCaasShareHandler == null) {
                Log.e(TAG, "initSuccess error");
                return;
            }
            shareHandler = hwCaasShareHandler;
            Log.i(TAG, "initSuccess");
        }

        @Override
        public void initFail(int retCode) {
            Log.e(TAG, "initFail retCode: " + retCode);
        }

        @Override
        public void releaseSuccess() {
            Log.i(TAG, "releaseSuccess");
            shareHandler = null;
        }

        @Override
        public void sendResult(HwShareUtils.SendResultEnum sendResultEnum) {
            switch (sendResultEnum) {
                case SEND_SUCCESS:
                    Log.i(TAG, "SEND_SUCCESS");
                    break;
                case SEND_FAIL:
                    Log.i(TAG, "SEND_FAIL");
                    break;
                case SEND_CANCEL:
                    Log.i(TAG, "SEND_CANCEL");
                    break;
            }
        }
    };

    public ShareHelper(Context context) {
        mContext = context;
    }

    public void shareKitInit() {
        Log.i(TAG, "shareKitInit");
        hwCaasShareManager = HwCaasShareManager.getInstance();
        hwCaasShareManager.init(mContext, mCallBack);
    }

    public void sendShareMsgInfo(int scene, HwShareMsgInfo info) {
        Log.i(TAG, "sendShareMsgInfo");
        if (shareHandler != null) {
            Log.d(TAG, "sendShareMsgInfo start");
            shareHandler.sendShareMsgInfo(scene, info);
            Log.d(TAG, "sendShareMsgInfo end");
        }
    }

    public Point getSize(HwShareUtils.ShareTypeEnum type) {
        Log.i(TAG, "getSize");
        if (shareHandler != null) {
            Log.d(TAG, "getSize start");
            return shareHandler.getThumbSize(type);
        }
        return null;
    }

    /**
     * release source
     */
    public void release(){
        Log.i(TAG, "release");
        if (hwCaasShareManager != null) {
            Log.d(TAG, "release start");
            hwCaasShareManager.release();
        }
    }
}

/**

package com.lrtest.shareproductpro;

import android.graphics.Bitmap;
import android.graphics.Bitmap.CompressFormat;
import android.util.Log;

import java.io.ByteArrayOutputStream;

public class Util {
    private static final String TAG = "Util";

    public static final int FORMAT_JPEG = 1;
    public static final int FORMAT_WEBP = 2;

    public static int JEPGSIZE = 0;
    public static int WEBPSIZE = 0;

    public static byte[] bmpToByteArray(final Bitmap bmp, final boolean needRecycle, final int type, final int quality) {
        Log.i("HwShareDemo", " bmpToByteArray  origin size :" + bmp.getByteCount() + " origin :" + bmp.getRowBytes());
        ByteArrayOutputStream output = new ByteArrayOutputStream();
        if (type == FORMAT_JPEG) {
            bmp.compress(CompressFormat.JPEG, quality, output);
        } else if (type == FORMAT_WEBP) {
            bmp.compress(CompressFormat.WEBP, quality, output);
        } else {
            bmp.compress(CompressFormat.JPEG, quality, output);
        }

        if (needRecycle) {
            bmp.recycle();
        }

        byte[] result = output.toByteArray();
        try {
            output.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
        Log.i("HwShareDemo", "bmpToByteArray compress size :" + result.length + " type is :" + type + " quailty:" + quality);
        if (type == FORMAT_JPEG) {
            JEPGSIZE = result.length;
        } else if (type == FORMAT_WEBP) {
            WEBPSIZE = result.length;
        }
        return result;
    }
}
