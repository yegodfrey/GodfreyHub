# 元数据管理

文件元数据包含云端文件名、文件大小、文件类型等常用属性，也包括用户自定义的文件属性。文件上传到云端后，您可以设置文件的自定义属性。您也可以在下载文件实体前获取文件的元数据，来决定是否要下载文件。

## 前提条件

* 您已[开通云存储服务](https://developer.huawei.com/consumer/cn/doc/AppGallery-connect-Guides/agc-cloudstorage-enable-service-0000001275330014)。
* 您已[初始化存储实例](https://developer.huawei.com/consumer/cn/doc/AppGallery-connect-Guides/agc-storage-initialize-bucket-java-0000001282104682)。

## 操作步骤

1. 创建文件的引用，并初始化文件的[ObjectMetadata](https://developer.huawei.com/consumer/cn/doc/AppGallery-connect-References/objectmetadata-0000001111669024)实例。
2. 调用[ObjectMetadata.setCustomMetadata](https://developer.huawei.com/consumer/cn/doc/AppGallery-connect-References/objectmetadata-0000001111669024#section1197104774715)设置文件的自定义属性。 说明
   >
   > 为了保证文件上传下载时的数据一致性，上传文件时可以使用setSHA256Hash方法设置文件SHA256值。
3. 调用[StorageReference.updateObjectMetadata](https://developer.huawei.com/consumer/cn/doc/AppGallery-connect-References/storagereference-0000001158228875#section10769172917394)方法将设置的文件元数据覆盖到云端。
4. 当设置元数据成功后，您可以通过调用[StorageReference.getObjectMetadata](https://developer.huawei.com/consumer/cn/doc/AppGallery-connect-References/storagereference-0000001158228875#section6516172214377)方法获取云端文件的元数据。

## 参考代码

```screen
public class StorageMeta {
    private static StorageManagement storageManagement;

    /**
     * Get metadata of a file
     *
     * @throws StorageException storage exception
     */
    public static void getMetadata(String bucketName, String objectName) throws StorageException {
        storageManagement =
            AGCStorageManagement.getInstance(AGCClient.getInstance(), "cn-north-4", bucketName);
        // 创建文件的引用
        StorageReference storageReference = storageManagement.getStorageReference(objectName);
        // 调用StorageReference.getObjectMetadata获取云端文件元数据
        ObjectMetadata objectMetadata = storageReference.getObjectMetadata();
        System.out.println(objectMetadata.getContentType());
        System.out.println(objectMetadata.getCacheControl());
        System.out.println(objectMetadata.getContentEncoding());
        System.out.println(objectMetadata.getContentDisposition());
        System.out.println(objectMetadata.getContentLanguage());
    }
    /**
     * Update metadata of a file
     *
     * @throws StorageException storage exception
     */
    public static void updateMetadata(String bucketName, String objectName) throws StorageException {
        storageManagement =
            AGCStorageManagement.getInstance(AGCClient.getInstance(), "cn-north-4", bucketName);
        StorageReference storageReference = storageManagement.getStorageReference(objectName);
        // 调用StorageReference.updateObjectMetadata将设置的文件元数据覆盖到云端
        ObjectMetadata updatedObjectMetadata = storageReference.updateObjectMetadata(initObjectMetadata());
        System.out.println(updatedObjectMetadata.getContentType());
        System.out.println(updatedObjectMetadata.getCacheControl());
        System.out.println(updatedObjectMetadata.getContentEncoding());
        System.out.println(updatedObjectMetadata.getContentDisposition());
        System.out.println(updatedObjectMetadata.getContentLanguage());
    }
    private static ObjectMetadata initObjectMetadata() {
        // 初始化文件的ObjectMetadata实例
        ObjectMetadata metadata = new ObjectMetadata();
        metadata.setContentType("text/*");
        metadata.setCacheControl("no-cache");
        metadata.setContentEncoding("identity");
        metadata.setContentDisposition("inline");
        metadata.setContentLanguage("en");
        // 调用ObjectMetadata.setCustomMetadata设置文件的自定义属性
        Map<String, String> map = new HashMap<String, String>();
        map.put("key", "value");
        metadata.setCustomMetadata(map);
        return metadata;
    }
    public static void initializeClient(String content) throws AGCException {
        CredentialService credential =
            CredentialParser.toCredentialWithContents(content);
        AGCParameter agcParameter = AGCParameter.builder().setCredential(credential).build();
        AGCClient.initialize(agcParameter);
        System.out.println("init agcParameter Success.");
    }
    public static void main(String[] args) throws AGCException {
        // 存储实例名
        String bucketName = "xxxx";
        // 云端文件路径，支持传入"文件目录/文件名"，或仅传入文件名
        String cloudPath = "images/demo.jpg";
        // 凭据信息，json格式
        String credential = "{\"type\":\"project_client_id\",\"developer_id\":\"xxx\",\"project_id\":\"xxx\",\"client_id\":\"xxx\",\"client_secret\":\"xxx\",\"configuration_version\":\"3.0\",\"region\":\"CN\"}";

        StorageMeta.initializeClient(credential);

        // 更新文件元数据
        StorageMeta.updateMetadata(bucketName, cloudPath);

        // 获取文件元数据
        StorageMeta.getMetadata(bucketName, cloudPath);
    }
}
```
