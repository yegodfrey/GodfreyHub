# 示例代码

## 算法一：FF_PREDICTION_ALGORITHM_HYDRA

此算法仅支持内插方案，示例代码如下：

```screen
#include "frame_flow.h"

predictedFrame = CreateTexture();

FFInstanceCreateInfo createInfo; 
FFInstance m_FF = FrameFlow_Create(&createInfo); // Create instance

FFPerFrameData perFrameData;
perFrameData.flags = FF_PER_FRAME_DATA_DEPTH_BIT | FF_PER_FRAME_DATA_SCENE_COLOR_BIT |
            FF_PER_FRAME_DATA_VIEW_PROJ_BIT | FF_PER_FRAME_DATA_INV_VIEW_PROJ_BIT;
perFrameData. sceneColor= gameSceneFBO.colorTexture;
perFrameData. depthStencil= gameSceneFBO.depthTexture;

FFDrawFrameInfo drawFrameInfo;
drawFrameInfo.target = predictedFrame;
drawFrameInfo.targetIsTexture = FF_TRUE;

FFPredictionAlgorithmInfo algorithmInfo;
algorithmInfo.algorithm = FF_PREDICTION_ALGORITHM_HYDRA;
algorithmInfo.extrapolation = FF_FALSE;
FrameFlow_SetPredictionAlgorithm(m_FF, &algorithmInfo, &algoCaps);  // Select prediction algorithm
FrameFlow_SetResolution(m_FF, &resolution); // Set resolution
FrameFlow_SetStencilRemapInfo(m_FF, semantic); // Set depth semantic
FrameFlow_SetCvvZSemantic(m_FF, cvvZSemantic); // Set Z range semantic
FrameFlow_Activate(m_FF); // Enable predict

while (true) {
    perFrameData_. viewProj = lastViewProj;
    perFrameData_. invViewProj = Invert(&lastViewProj);
    FrameFlow_ProvidePerFrameData(m_FF, &perFrameData); // Set Camera VP Matrix
    FrameFlow_DrawFrame(m_FF, &drawFrameInfo); // Rendering predicted frame
    GameDrawUI();
    eglSwapBuffers();
    BlitToDisplay(gameSceneFBO.colorTexture); 
    GameDrawUI();
    GameDrawScene();
    eglSwapBuffers(); 
} 
```
