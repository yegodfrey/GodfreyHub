# 编译JNI

推理模型编译JNI时，以NPU场景为例，打包APK时需要打包libhiai.so、libhiai_ir.so、 libhiai_model_compatible.so（可选）、libhiai_enhance.so（可选） 和libhiaijni.so，其中libhiaijni.so为Demo的JNI打包二进制。  

#### 编写CMakeLists.txt 以及npu_demo.cmake

以非异构运行在NPU上的场景为例：

CMakeLists.txt 如下：

```
# Copyright (c) Huawei Technologies Co., Ltd. 2022-2022. All rights reserved.
cmake_minimum_required(VERSION 3.6.0)
# please select the cmake file based on the type of device you want to run the APK.
# NPU:npu_demo.cmake
# CPU:cpu_demo.cmake

include(npu_demo.cmake)
#include(cpu_demo.cmake)
```

对应的npu_demo.cmake实现如下：

```
# Copyright (c) Huawei Technologies Co., Ltd. 2022-2022. All rights reserved.

PROJECT(hiaijni LANGUAGES C CXX)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -std=c++11")
set(CMAKE_CXX_STANDARD 11)

include_directories(${CMAKE_CURRENT_SOURCE_DIR}/src/main/jni)
set(jnilibs ${CMAKE_SOURCE_DIR}/libs/${ANDROID_ABI})

#Create libhiai.so
add_library(hiai SHARED IMPORTED)
set_target_properties(hiai PROPERTIES IMPORTED_LOCATION
        ${jnilibs}/libhiai.so)

#Create libhiai_ir.so
add_library(hiai_ir SHARED IMPORTED )
set_target_properties(hiai_ir PROPERTIES IMPORTED_LOCATION
        ${jnilibs}/libhiai_ir.so)

#Create libhiai_ir_build.so
add_library(hiai_ir_build SHARED IMPORTED )
set_target_properties(hiai_ir_build PROPERTIES IMPORTED_LOCATION
        ${jnilibs}/libhiai_ir_build.so)

#Create libhiai_model_compatible.so
add_library(hiai_model_compatible SHARED IMPORTED )
set_target_properties(hiai_model_compatible PROPERTIES IMPORTED_LOCATION
        ${jnilibs}/libhiai_model_compatible.so)

#Create libhiai_enhance.so
add_library(hiai_enhance SHARED IMPORTED )
set_target_properties(hiai_enhance PROPERTIES IMPORTED_LOCATION
        ${jnilibs}/libhiai_enhance.so)

link_directories(${jnilibs})

# hiaijni link library

# npu inference
if (${MODULE} STREQUAL "inference")
    add_library(hiaijni SHARED
            ../${MODULE_NAME}/src/main/jni/ClassifyJni.cpp
            ../${MODULE_NAME}/src/main/jni/Utils.cpp
            )
    target_link_libraries(hiaijni LINK_PUBLIC
            hiai
            hiai_ir
            hiai_model_compatible
            hiai_enhance
            android
            log)
endif()

# npu irmodulebuild
if (${MODULE} STREQUAL "irmodelbuild" )
    add_library(hiaijni SHARED
            ../${MODULE_NAME}/src/main/jni/ClassifyJni.cpp
            ../${MODULE_NAME}/src/main/jni/CreateModel.cpp
            ../${MODULE_NAME}/src/main/jni/Utils.cpp
            )
    target_link_libraries(hiaijni LINK_PUBLIC
            hiai
            hiai_ir
            hiai_ir_build
            hiai_model_compatible
            hiai_enhance
            android
            log)
endif()

set_target_properties(hiaijni PROPERTIES LINKER_LANGUAGE CXX)
```
