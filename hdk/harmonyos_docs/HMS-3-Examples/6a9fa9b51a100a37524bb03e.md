        1. package com.huawei.hmsagentsamplexsj;
        2. import android.graphics.Color;
        3. import android.os.Bundle;
        4. import android.view.View;
        5. import android.widget.Button;
        6. import android.widget.Toast;
        7. import com.huawei.android.hms.agent.HMSAgent;
        8. import com.huawei.android.hms.agent.common.handler.ConnectHandler;
        9. import com.huawei.android.hms.agent.hwid.handler.SignInHandler;
        10. import com.huawei.android.hms.agent.hwid.handler.SignOutHandler;
        11. import com.huawei.hms.support.api.hwid.SignInHuaweiId;
        12. import com.huawei.hms.support.api.hwid.SignOutResult;
        13. public class HwIDActivity extends AgentBaseActivity {
        14.     @Override
        15.     protected void onCreate(Bundle savedInstanceState) {
        16.         super.onCreate(savedInstanceState);
        17.         setContentView(R.layout.activity_hw_id);
        18.         setTabBtnClickListener();
        19.         Button btn = (Button) findViewById(R.id.btn_id);
        20.         if (btn != null) {
        21.             btn.setTextColor(Color.RED);
        22.             btn.setEnabled(false);
        23.         }
        24.         findViewById(R.id.btn_signin).setOnClickListener(this);
        25.         findViewById(R.id.btn_forcesignin).setOnClickListener(this);
        26.         findViewById(R.id.btn_signout).setOnClickListener(this);
        27.         HMSAgent.connect(this, new ConnectHandler() {
        28.             @Override
        29.             public void onConnect(int rst) {
        30.                 showLog("HMS Connect end: " + rst);
        31.             }
        32.         });
        33.     }
        34.     /**
        35.      * 登录授权 | Login Authorization
        36.      *
        37.      * @param forceLogin 如果已经授权登录则直接回调结果，否则：forceLogin为true时会拉起界面，为false时直接回调错误码 | If the login is authorized, the result is directly callback, otherwise: When Forcelogin is true, the activity is pulled and the error code is directly invoked when false
        38.      */
        39.     private void signIn(boolean forceLogin) {
        40.         HMSAgent.Hwid.signIn(forceLogin, new SignInHandler() {
        41.             @Override
        42.             public void onResult(int rtnCode, SignInHuaweiId signInResult) {
        43.                 if (rtnCode == HMSAgent.AgentResultCode.HMSAGENT_SUCCESS && signInResult != null) {
        44.                     //可以获取帐号的 openid，昵称，头像 at信息 | You can get the OpenID, nickname, Avatar, at etc information of Huawei ID
        45.                     showLog("signIn successful=========");
        46.                     showLog("Nickname:" + signInResult.getDisplayName());
        47.                     showLog("openid:" + signInResult.getOpenId());
        48.                     showLog("accessToken:" + signInResult.getAccessToken());
        49.                     showLog("Head pic URL:" + signInResult.getPhotoUrl());
        50.                     showLog("unionID:" + signInResult.getUnionId());
        51.                 } else {
        52.                     showLog("signIn---error: " + rtnCode);
        53.                 }
        54.             }
        55.         });
        56.     }
        57.     /**
        58.      * 退出。此接口调用后，下次再调用signIn会拉起界面，请谨慎调用。如果不确定就不要调用了。 | Exit。 After this method is called, the next time you call signIn will pull the activity, please call carefully. Do not call if you are unsure.
        59.      */
        60.     private void signOut() {
        61.         HMSAgent.Hwid.signOut(new SignOutHandler() {
        62.             @Override
        63.             public void onResult(int rtnCode, SignOutResult signOutResult) {
        64.                 if (rtnCode == HMSAgent.AgentResultCode.HMSAGENT_SUCCESS && signOutResult != null) {
        65.                     showLog("SignOut successful");
        66.                 } else {
        67.                     showLog("SignOut fail:" + rtnCode);
        68.                 }
        69.             }
        70.         });
        71.     }
        72.     /**
        73.      * Called when a view has been clicked.
        74.      *
        75.      * @param v The view that was clicked.
        76.      */
        77.     @Override
        78.     public void onClick(View v) {
        79.         int id = v.getId();
        80.         if (id == R.id.btn_id) {
        81.             // 本页面切换到本页面的按钮事件不处理 | "This page switches to itself" button event does not need to be handled
        82.             return;
        83.         } else if (!onTabBtnClickListener(id)) {
        84.             // 如果不是tab切换按钮则处理业务按钮事件 | Handle Business button events without the TAB toggle button
        85.             switch (id) {
        86.                 case R.id.btn_signin:
        87.                     signIn(false);
        88.                     break;
        89.                 case R.id.btn_forcesignin:
        90.                     signIn(true);
        91.                     break;
        92.                 case R.id.btn_signout:
        93.                     signOut();
        94.                     break;
        95.                 default:
        96.             }
        97.         }
        98.     }
        99. }
