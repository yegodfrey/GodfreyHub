        59.     private static boolean verify(Signature signatureAlgorithm, Jws jws) {
        60.         // Verify cert chain

        63.         try {

        65.                 X509Certificate tmp = readCert(certs.get(i));
        66.                 certificates[i] = tmp;

        69.         } catch (CertificateException | NoSuchAlgorithmException | InvalidKeyException | NoSuchProviderException | SignatureException e) {
