        139.     private static String sign(byte[] data, String privateKey)
        140.     {
        141.         try
        142.         {
        143.             byte[] e = Base64.decodeBase64(privateKey);
        144.             PKCS8EncodedKeySpec pkcs8KeySpec = new PKCS8EncodedKeySpec(e);
        145.             KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        146.             PrivateKey privateK = keyFactory.generatePrivate(pkcs8KeySpec);
        147.             Signature signature = Signature.getInstance("SHA256WithRSA");
        148.             signature.initSign(privateK);
        149.             signature.update(data);
        150.             String signs = Base64.encodeBase64String(signature.sign());
        151.             return signs;
        152.         }
        153.         catch (Exception var)
        154.         {
        155.             System.out.println("SignUtil.sign error." + var);
        156.             return "";
        157.         }
        158.     }
        159.     /**       
        160.      * 根据参数Map构造排序好的参数串       
        161.      *       
        162.      * @param params       
        163.      * @return       
        164.      */   
        165.     private static String format(Map<String, String> params)
        166.     {
        167.         StringBuffer base = new StringBuffer();
        168.         Map<String, String> tempMap = new TreeMap<String, String>(params);
        169.         // 获取计算nsp_key的基础串       
        170.         try
        171.         {
        172.             for (Map.Entry<String, String> entry : tempMap.entrySet())
        173.             {
        174.                 String k = entry.getKey();
        175.                 String v = entry.getValue();
        176.                 base.append(k).append("=").append(URLEncoder.encode(v, "UTF-8")).append("&");
        177.             }
        178.         }
        179.         catch (UnsupportedEncodingException e)
        180.         {
        181.             System.out.println("Encode parameters failed.");
        182.             e.printStackTrace();
        183.         }
        184.         String body = base.toString().substring(0, base.toString().length() - 1);
        185.         // 空格和星号转义     
        186.         body = body.replaceAll("\\+", "%20").replaceAll("\\*", "%2A");
        187.         return body;
        188.     }
        189.     private static String generateCPSign(Map<String,String> requestParams,final String cpAuthKey)
        190.     {
        191.         // 对消息体中查询字符串按字典序排序并且进行URLCode编码       
        192.         String baseStr = format(requestParams);       
        193.         // 用CP侧签名私钥对上述编码后的请求字符串进行签名     
        194.         String cpSign = sign(baseStr.getBytes(Charset.forName("UTF-8")), cpAuthKey);
        195.         return cpSign;
        196.     }
        197. }

        104.                     return null;
        105.                 bt = binr.ReadByte();
        106.                 if (bt != 0x02)
        107.                     return null;
        108.                 twobytes = binr.ReadUInt16();
        109.                 if (twobytes != 0x0001)
        110.                     return null;
        111.                 seq = binr.ReadBytes(15);       //read the Sequence OID
        112.                 if (!CompareBytearrays(seq, SeqOID))    //make sure Sequence for OID is correct
        113.                     return null;

        116.                     return null;

        127.             }

        129.             {

        131.             }

        134.         private static bool CompareBytearrays(byte[] a, byte[] b)
        135.         {
        136.             if (a.Length != b.Length)
        137.                 return false;
        138.             int i = 0;
        139.             foreach (byte c in a)
        140.             {
        141.                 if (c != b[i])
        142.                     return false;
        143.                 i++;
        144.             }
        145.             return true;
        146.         }
        147.         private static RSACryptoServiceProvider DecodeRSAPrivateKey(byte[] privkey)
        148.         {
        149.             byte[] MODULUS, E, D, P, Q, DP, DQ, IQ;
        150.             // ---------  Set up stream to decode the asn.1 encoded RSA private key  ------
        151.             MemoryStream mem = new MemoryStream(privkey);
        152.             BinaryReader binr = new BinaryReader(mem);    //wrap Memory Stream with BinaryReader for easy reading
        153.             byte bt = 0;
        154.             ushort twobytes = 0;
        155.             int elems = 0;
        156.             try
        157.             {
        158.                 twobytes = binr.ReadUInt16();
        159.                 if (twobytes == 0x8130) //data read as little endian order (actual data order for Sequence is 30 81)
        160.                     binr.ReadByte();    //advance 1 byte
        161.                 else if (twobytes == 0x8230)
        162.                     binr.ReadInt16();   //advance 2 bytes
        163.                 else
        164.                    return null;
        165.                 twobytes = binr.ReadUInt16();
        166.                 if (twobytes != 0x0102) //version number
        167.                     return null;
        168.                 bt = binr.ReadByte();
        169.                 if (bt != 0x00)
        170.                     return null;
        171.                 //------  all private key components are Integer sequences ----
        172.                 elems = GetIntegerSize(binr);
        173.                 MODULUS = binr.ReadBytes(elems);
        174.                 elems = GetIntegerSize(binr);
        175.                 E = binr.ReadBytes(elems);
        176.                 elems = GetIntegerSize(binr);
        177.                 D = binr.ReadBytes(elems);
        178.                 elems = GetIntegerSize(binr);
        179.                 P = binr.ReadBytes(elems);
        180.                 elems = GetIntegerSize(binr);
        181.                 Q = binr.ReadBytes(elems);
        182.                 elems = GetIntegerSize(binr);
        183.                 DP = binr.ReadBytes(elems);
        184.                 elems = GetIntegerSize(binr);
        185.                DQ = binr.ReadBytes(elems);
        186.                 elems = GetIntegerSize(binr);
        187.                 IQ = binr.ReadBytes(elems);
        188.                 // ------- create RSACryptoServiceProvider instance and initialize with public key -----
        189.                 RSACryptoServiceProvider RSA = new RSACryptoServiceProvider();
        190.                 RSAParameters RSAparams = new RSAParameters();
        191.                 RSAparams.Modulus = MODULUS;
        192.                 RSAparams.Exponent = E;
        193.                 RSAparams.D = D;
        194.                 RSAparams.P = P;
        195.                 RSAparams.Q = Q;
        196.                 RSAparams.DP = DP;
        197.                 RSAparams.DQ = DQ;
        198.                 RSAparams.InverseQ = IQ;
        199.                RSA.ImportParameters(RSAparams);
        200.                 return RSA;
        201.             }
        202.             catch (Exception)
        203.             {
        204.                 return null;
        205.             }
        206.             finally { binr.Close(); }
        207.         }
        208.         private static int GetIntegerSize(BinaryReader binr)
        209.         {
        210.             byte bt = 0;
        211.             byte lowbyte = 0x00;
        212.             byte highbyte = 0x00;
        213.             int count = 0;
        214.             bt = binr.ReadByte();
        215.             if (bt != 0x02)     //expect integer
        216.                 return 0;
        217.             bt = binr.ReadByte();
        218.             if (bt == 0x81)
        219.                 count = binr.ReadByte();    // data size in next byte
        220.             else
        221.             if (bt == 0x82)
        222.             {
        223.                 highbyte = binr.ReadByte(); // data size in next 2 bytes
        224.                 lowbyte = binr.ReadByte();
        225.                 byte[] modint = { lowbyte, highbyte, 0x00, 0x00 };
        226.                 count = BitConverter.ToInt32(modint, 0);
        227.             }
        228.            else
        229.            {
        230.                count = bt;     // we already have the data size
        231.            }
        232.            while (binr.ReadByte() == 0x00)
        233.            {   //remove high order zeros in data
        234.                count -= 1;
        235.            }
        236.            binr.BaseStream.Seek(-1, SeekOrigin.Current);       //last ReadByte wasn't a removed zero, so back up a byte
        237.            return count;
        238.        }
