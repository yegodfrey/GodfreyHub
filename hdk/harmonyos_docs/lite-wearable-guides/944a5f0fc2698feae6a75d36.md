    /******************************************************************************
     * @brief    CRT0 file used for starting up the rest of the code
     * Copyright (c) 2022 CompanyNameMagicTag
    ******************************************************************************/
    /**
     * @brief  This is the code that gets called when the processor first
     *          starts execution following a reset event. Only the absolutely
     *          necessary set is performed, after which the application
     *          supplied sec_main() routine is called.
     * @param  None
     * @retval : None
    */
            .section .text.entry
            .global data_bss_init
            .option norvc
    data_bss_init:
            /* set data section */
            la      gp, __global_pointer$
            la      t0, __data_begin__
            la      t1, __data_load__
            la      t4, __data_end__
            sub     t2, t4, t0
            beq     t2, x0, end_set_data_loop
    
    set_data_loop:
            lw      t3, (t1)
            sw      t3, (t0)
            addi    t0, t0, 4
            addi    t1, t1, 4
            addi    t2, t2, -4
            blt     x0, t2, set_data_loop
    end_set_data_loop:
    
            /* clear bss section */
            la      t0, __bss_begin__
            la      t1, __bss_end__
            beq     t0, t1, end_clear_bss_loop
            li      t2, 0x00000000
    
    clear_bss_loop:
            sw      t2, (t0)
            addi    t0, t0, 4
            blt     t0, t1, clear_bss_loop
    end_clear_bss_loop:
            ret

    OUTPUT_ARCH( "riscv" )
    ENTRY(__init)
    
    __ICFEDIT_intvec_start__ = 0x0;
    __ICFEDIT_region_ROM_start__ = 0x000000;
    __ICFEDIT_region_ROM_end__   = 0x0FFFF;
    rom_size = __ICFEDIT_region_ROM_end__ - __ICFEDIT_region_ROM_start__;
    __ICFEDIT_region_RAM_start__ = 0x10000;
    __ICFEDIT_region_RAM_end__   = 0x1FFFF;
    ram_size = __ICFEDIT_region_RAM_end__ - __ICFEDIT_region_RAM_start__;
    __ICFEDIT_size_heap__ = 0x1000;
    
    MEMORY
    {
      ROM(rx)   : ORIGIN = __ICFEDIT_region_ROM_start__, LENGTH = rom_size
      RWPI(rw)   : ORIGIN = __ICFEDIT_region_RAM_start__, LENGTH = ram_size
    }
    
    SECTIONS
    {
      .header :
      {
        ROM_start = .;
        *(.header)
      } > ROM
    
      .intvec : ALIGN(4)
      {
         *(.intvec*)
      } > ROM
    
      .text : ALIGN(4)
      {
        *(.text*)
      } > ROM
    
      .rodata : ALIGN(4)
      {
        *(.rodata*)
      } > ROM
    
      .checksum : ALIGN(4)
      {
        *(.checksum*)
      } > ROM
    
      .noinit : ALIGN(8)
      {
        *(.noinit)
      } > ROM
    
      .data : ALIGN(8)
      { 
        RWPI_start = .;
    	__data_begin__ = .;
        __data_load__ = LOADADDR(.data);
        *(.data*)
    	*(.srodata.cst16) *(.srodata.cst8) *(.srodata.cst4) *(.srodata.cst2)
        *(.srodata*)
        *(.sdata .sdata.* .gnu.linkonce.s.*) 
    	__data_end__ = .;
      } > RWPI AT > ROM
      __data_size__ = __data_end__ - __data_begin__;
    
      .bss (NOLOAD): ALIGN(8)
      {
    	__bss_start = .;
        __bss_begin__ = .;
        *(.shbss)
        *(.bss .bss.* .gnu.linkonce.b.*)
    	*(.sbss*)
        *(.scommon)
    	__bss_end__ = . ;
    	RWPI_end = .;
      } > RWPI
      __bss_size__ = __bss_end__ - __bss_begin__;
      __global_pointer$ = __ICFEDIT_region_RAM_start__;
    
      .heap(NOLOAD) : ALIGN(8)
      {
        . += __ICFEDIT_size_heap__;
      } > RWPI
    
      .flash_end : ALIGN(4)
      {
        . = ALIGN(4);
        ROM_end = .;
      } > ROM
    }

    #include "thirdpay_common.h"

    typedef struct {
        /*
         * The payment SDK notifies the Huawei app of the device binding progress.
         * @note If a failure occurs, print the failure cause log in time to help analyze the problem.
         * @param [in] Unique vendor ID, which must be the same as the payment vendor name negotiated in the resource
         * package.
         * @param [in] status Binding status, including binding failure
         * @return void
         */
        void (*notifyBindProcess)(const uint8_t *appName, ThirdPay_BindingStatus status);
    
        /*
         * Payment SDK Bluetooth ble data sending interface
         * @param [in] appName Unique vendor ID, which must be the same as the payment provider name in the resource
         * package.
         * @param [in] bleData Pointer to the data to be sent
         * @param [in] bleDataLen Size of the sent data
         * @return RETCODE_OK The message is sent successfully.
         * Other error codes are returned based on ThirdPay_RetCode.
         * If there is no corresponding error, return RETCODE_FAIL and print logs for fault analysis.
         */
        ThirdPay_RetCode (*notifyBleData)(const uint8_t *appName, const uint8_t *bleData, uint16_t bleDataLen);
    
        /*
         * @note Secure storage stores SDK-defined data in 256-byte partitions. A maximum of 20 partitions are allowed.
         * For security encryption, Huawei APP system implements the following functions:
         * Data stored in the flash memory is encrypted using a secure working key.
         * Data is stored in the memory for as short a time as possible and released immediately after use.
         * Use locks for read/write protection.
         */
    
        /*
         * Secure storage: write
         * @param [in] appName Unique vendor ID, which must be the same as the payment vendor name in the resource package.
         * @param [in] item: data item to be written, which is used to specify the partition. The value ranges from 0 to 19.
         * @param [in] data: data to be written
         * @param [in] dataLen: length of the data to be written. The value ranges from 1 to 256.
         * @return RETCODE_OK is written successfully.
         * Other error codes are returned based on ThirdPay_RetCode.
         * If no corresponding error occurs, return RETCODE_FAIL and print logs for fault analysis.
         */
        ThirdPay_RetCode (*storageWrite)(const uint8_t *appName, uint32_t item, const uint8_t *data, uint32_t dataLen);
    
        /*
         * Secure storage: Read
         * @param [in] appName Unique vendor ID, which must be the same as the payment vendor name in the resource package.
         * @param [in] item: item of the data to be read, which is used to specify the partition. The value ranges from 0
         * to 19.
         * @param [out] data: Pointer to the data to be read. The memory is allocated by the caller.
         * @param [in] dataLen: length of the data to be read. The value ranges from 1 to 256.
         * @return RETCODE_OK is successfully read.
         * Other error codes are returned based on ThirdPay_RetCode.
         * If no corresponding error occurs, return RETCODE_FAIL and print logs for fault analysis.
         */
        ThirdPay_RetCode (*storageRead)(const uint8_t *appName, uint32_t item, uint8_t *data, uint32_t dataLen);
    
        /*
         * Secure storage: delete
         * @param [in] appName Unique vendor ID, which must be the same as the payment provider name in the resource
         * package.
         * @param [in] item: data item to be deleted, which is used to specify the partition. The value ranges from 0 to 19.
         * @return RETCODE_OK Deletion succeeded.
         * Other error codes are returned based on ThirdPay_RetCode.
         * If there is no corresponding error, return RETCODE_FAIL and print logs for fault analysis.
         */
        ThirdPay_RetCode (*storageDelete)(const uint8_t *appName, uint32_t item);
    
        /*
         * Secure storage: specifies whether the item has data.
         * @param [in] appName Unique vendor ID, which must be the same as the payment provider name in the resource
         * package.
         * @param [in] item: data item to be checked, which is used to specify a partition. The value ranges from 0 to 19.
         * @return true: The specified item data exists.
         * false: The specified item does not exist.
         */
        bool (*storageExist)(const uint8_t *appName, uint32_t item);
    
        /*
         *Obtain the UNIX timestamp of the system.
         *@return: UNIX timestamp. The unit is second. If the return value is 0, it indicates that the return fails.
         */
        uint32_t (*getTimestamp)(void);
    
        /*
         * Read the device ID.
         * @param [out] deviceID: Pointer to the obtained device ID. The memory is allocated by the invoker.

         * @return void
         */
        void (*getDeviceID)(uint8_t *deviceID, uint32_t *deviceIdLen);
    
        /*
         * Log Print Interface
         */
        int (*logPrintf)(const char *fmt,...);
    
        /*
         * Memory Malloc Interface
         */
        void *(*osMalloc)(size_t size);
    
        /*
         * Memory Free Interface
         */
        void (*osFree)(void *buf);
    } ThirdPay_FeatureComponent;
    
    #endif  // THIRDPAY_H

    typedef enum {  // Characteristic Properties bit field, spec defined
        // If set, permits broadcasts of the Characteristic Value using Characteristic Configuration Descriptor
        CHARACTER_ATT_BROADCAST = (1 << 0),  // No Broadcast(0)/Broadcast(1),
        // If set, permits reads of the Characteristic Value
        CHARACTER_ATT_READ = (1 << 1),  // No Read(0)/Read(1),
        // If set, permit writes of the Characteristic Value without response
        CHARACTER_ATT_WRITE_NORSP = (1 << 2),  // No Write Without Response(0)/Write Without Response(1),
        // If set, permits writes of the Characteristic Value with response
