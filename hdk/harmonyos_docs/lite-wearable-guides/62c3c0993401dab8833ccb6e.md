# 毕昇编译器

#### 毕昇RWPI&ROPI动态库文件头格式

动态库文件头遵循如下格式：
    
    typedef struct {
    uint32_t magicWords;    // 动态库文件头魔术字，每个不同的动态库可以配置不同的魔术字
    uint32_t romBase;       // 动态库ROM基地址
    uint32_t romLength;     // 动态库bin文件大小，或理解为ROM大小（包含.text代码段，.rodata const数据段，.data_init data数据的初始化信息段等数据信息）
    uint32_t dataLength;    // 动态库数据段大小
    void* init;             // 动态库初始化函数
    void* deinit;           // 动态库去初始化函数
    uint32_t methodsLen;    // 动态库对外提供的接口个数
    void* methods[];        // 动态库对外接口列表
    } LosBinInfo;

**表1** 动态库文件头格式字段 | 大小（字节）  
---|---  
动态库头魔术字 | 4  
ROM起始地址 | 4  
ROM空间大小 | 4  
RAM空间大小 | 4  
动态库init函数地址 | 4  
动态库deinit函数地址 | 4  
动态库对外提供接口个数 | 4  
  
后续为动态库对外提供接口函数地址，动态库header.s内容示例如下：
    
    #   .text
    #   .attribute  4, 16
    #   .attribute  5, "rv32i2p0_m2p0_c2p0_xlinxma1p0_xlinxmb1p0_xlinxmc1p0_xlinxmd1p0"
    #   .file   "test.c"
    .section    .header,"aw",@progbits
    .type   magic_number,@object            # @magic_number
    .globl  magic_number
    .p2align    2
    magic_number:
    .word   0x484d2e62                      # 0x484d2e62
    .size   magic_number, 4
    .type   rom_begin,@object               # @rom_begin
    .globl  rom_begin
    .p2align    2
    rom_begin:
    .word   __ICFEDIT_region_ROM_start__    # ROM起始地址，定义在lds脚本内
    .size   rom_begin, 4
    .type   rom_size,@object                # @rom_size
    .globl  rom_size
    .p2align    2
    rom_size:
    .word   ROM_end - ROM_start             # ROM size的计算，通过lds脚本里面定义的符号ROM_end-ROM_start计算得出
    .size   rom_size, 4
    .type   rwpi_length,@object             # @rwpi_length
    .globl  rwpi_length
    .p2align    2
    rwpi_length:
    .word   RWPI_end - RWPI_start           # RAM size的计算，通过lds脚本里面定义的符号RWPI_start和RWPI_end相减计算得出
    .size   rwpi_length, 4
    .type   in,@object                      # @init
    .globl  in
    .p2align    2
    in:
    .word   __init                          # __init函数
    .size   in, 4
    .type   de,@object                      # @de
    .globl  de
    .p2align    2
    de:
    .word   __deinit                        # __deinit函数
    .size   de, 4
    .type   call_num,@object                # @call_num
    .globl  call_num
    .p2align    2
    call_num:
    .word   5                               # 动态库对外提供接口个数
    .size   call_num, 4
    .type   slot_1,@object                  # 对外提供接口1
    .globl  slot_1
    .p2align    2
    slot_1:
    .word   GetTestValue                    # 对外提供接口GetTestValue
    .size   slot_1, 4
    .type   slot_2,@object                  # 对外提供接口2
    .globl  slot_2
    .p2align    2
    slot_2:
    .word   GetTest1Value                   # 对外提供接口GetTest1Value
    .size   slot_2, 4
    .type   slot_3,@object                  # 对外提供接口3
    .globl  slot_3
    .p2align    2
    slot_3:
    .word   SetTestValue                    # 对外提供接口SetTestValue
    .size   slot_3, 4
    .type   slot_4,@object                  # 对外提供接口4
    .globl  slot_4
    .p2align    2
    slot_4:
    .word   SetTest1Value                   # 对外提供接口SetTest1Value
    .size   slot_4, 4
    .type   slot_5,@object                  # 对外提供接口5
    .globl  slot_5
    .p2align    2
    slot_5:
    .word   ThirdPay_SetInterface           # 对外提供接口ThirdPay_SetInterface
    .size   slot_5, 4
    .ident  "clang version 15.0.4 (BiSheng Mobile MCU 201.1.0.B001 7b810936c2a9)"
    .section    ".note.GNU-stack","",@progbits

#### 数据初始化

毕昇数据初始化，在__init函数内调用__losInitFunc，__losInitFunc调用data_bss_init完成data和bss数据初始化，data_bss_init函数实现在startup.s汇编文件内，data_bss_init实现详情参考[附录章节startip.s](https://developer.huawei.com/consumer/cn/doc/lite-wearable-guides/bin-compilation-tool-appendix)内容。

     * 通过从__data_load__地址空间将数据拷贝到__data_begin__，大小为__data_end__ - __data_begin__，完成data数据的初始化。
     * 通过从__bss_begin__到__bss_end__ 地址空间清零，完成bss数据段的初始化。
     * __data_load__，__data_begin__，__data_end__，__bss_begin__，__bss_end__等符号定义在lds文件内。
    
    void __losInitFunc(void* val)
    {
        data_bss_init();
        return;
    }
    
    int __init(void ** symbols, int symbolLen, void *dataAddr, unsigned int *appVer)
    {
       __losInitFunc(dataAddr);
       return 0;
    }

#### 编译器lds链接脚本实现

     1. 定义ROM和RAM memory region。
            
            __ICFEDIT_intvec_start__ = 0x0;
            __ICFEDIT_region_ROM_start__ = 0x000000;
            __ICFEDIT_region_ROM_end__   = 0x0FFFF;
            rom_size = __ICFEDIT_region_ROM_end__ - __ICFEDIT_region_ROM_start__;
            __ICFEDIT_region_RAM_start__ = 0x10000;
            __ICFEDIT_region_RAM_end__   = 0x1FFFF;
            ram_size = __ICFEDIT_region_RAM_end__ - __ICFEDIT_region_RAM_start__;
            __ICFEDIT_size_heap__ = 0x1000;
            MEMORY
            {
            ROM(rx)   : ORIGIN = __ICFEDIT_region_ROM_start__, LENGTH = rom_size
            RWPI(rw)   : ORIGIN = __ICFEDIT_region_RAM_start__, LENGTH = ram_size
            }

     2. 指定.header、.intvec、.text、.rodata、.checksum、.noinit RO数据到ROM段。
            
            .header :
            {
            ROM_start = .;
            *(.header)
            } > ROM
            .intvec : ALIGN(4)
            {
            *(.intvec*)
            } > ROM
            .text : ALIGN(4)
            {
            *(.text*)
            } > ROM
            .rodata : ALIGN(4)
            {
            *(.rodata*)
            } > ROM
            .checksum : ALIGN(4)
            {
            *(.checksum*)
            } > ROM
            .noinit : ALIGN(8)
            {
            *(.noinit)
            } > ROM

     3. 指定.data、.bss、.heap RW数据到RAM段。
            
            .data : ALIGN(8)
            {
            RWPI_start = .;
            __data_begin__ = .;
            __data_load__ = LOADADDR(.data);
            *(.data*)
            *(.srodata.cst16) *(.srodata.cst8) *(.srodata.cst4) *(.srodata.cst2)
            *(.srodata*)
            *(.sdata .sdata.* .gnu.linkonce.s.*)
            __data_end__ = .;
            } > RWPI AT > ROM
            __data_size__ = __data_end__ - __data_begin__;
            .bss (NOLOAD): ALIGN(8)
            {
            __bss_start = .;
            __bss_begin__ = .;
            *(.shbss)
            *(.bss .bss.* .gnu.linkonce.b.*)
            *(.sbss*)
            *(.scommon)
            __bss_end__ = . ;
            RWPI_end = .;
            } > RWPI
            __bss_size__ = __bss_end__ - __bss_begin__;
            __global_pointer$ = __ICFEDIT_region_RAM_start__;
            .heap(NOLOAD) : ALIGN(8)
            {
            . += __ICFEDIT_size_heap__;
            } > RWPI

     4. 指定ROM结束位置。
            
            .flash_end : ALIGN(4)
            {
            . = ALIGN(4);
            ROM_end = .;
            } > ROM

.flash_end要指定到lds文件最后，确保ROM_end - ROM_start计算出来的大小是bin文件的大小，否则会导致__data_load__ 内容丢失，data数据无法完成初始化。

#### 编译命令

生成sdk.out和sdk.map：
    
    BiSheng_Mobile_203.0.0.B046/BiSheng-llvm-binary-release-musl/bin/clang header.s sdk.c startup.s -frwpi -fropi -O2 --target=riscv32 -T riscv.lds -nostartfiles -Wl,--Map=sdk.map -o sdk.out

生成sdk.bin：
    
    BiSheng_Mobile_203.0.0.B046/BiSheng-llvm-binary-release-musl/bin/llvm-objcopy -O binary sdk.out sdk.bin

生成sdk.asm文件：
    
    BiSheng_Mobile_203.0.0.B046/BiSheng-llvm-binary-release-musl/bin/llvm-objdump -d sdk.out > sdk.asm

**说明**

.data段要求数据不能为0，否则编译出的bin文件头信息与文件实际大小将不符。例如可定义全局变量并在初始化函数中使用防止被编译优化：
    
    uint32_t g_test = 100;
    
    int __init(void ** symbols, int symbolLen, void *dataAddr, unsigned int *appVer)
    {
        __losInitFunc(dataAddr);
        *appVer = 1;
        int32_t test1 = g_test - 1;
        return 0;
    }
