# 标记聚合

华为地图SDK支持聚合标记功能，该功能可以有效地管理在地图不同缩放级别情况下的多个标记。当用户以高缩放级别查看地图时，各个标记会显示在地图上；当用户缩小时，标记会聚集成集群，从而使标记的呈现更有序。

## 添加聚合标记

调用HWMapJsSDK.[HWMarkerCluster](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151)(map, markers, options)添加聚合标记。

```javascript
var map;
var markers = [];
var markerCluster;

var locations =  [
    {lat: 51.5145160, lng: -0.1270060},
    { lat : 51.5064490, lng : -0.1244260 },
    { lat : 51.5097080, lng : -0.1200450 },
    { lat : 51.5090680, lng : -0.1421420 },
    { lat : 51.4976080, lng : -0.1456320 },
    { lat : 51.5061590, lng : -0.140280 },
    { lat : 51.5047420, lng : -0.1470490 },
    { lat : 51.5126760, lng : -0.1189760 },
    { lat : 51.5108480, lng : -0.1208480 }
];

function initMap() {
    var mapOptions = {};
    mapOptions.center = {lat : 51.4976080, lng : -0.1456320 };
    mapOptions.zoom = 13;

    map = new HWMapJsSDK.HWMap(document.getElementById('map'), mapOptions);
    generateMarkers(locations);

    // 初始化聚合点
    markerCluster = new HWMapJsSDK.HWMarkerCluster(map, markers);
}

// 批量初始化marker
function generateMarkers(locations) {
    for (let i = 0; i < locations.length; i++) {
        var opts = {
            position: locations[i]
        };

        markers.push(new HWMapJsSDK.HWMarker(opts));
    }
}
```

## 自定义聚合标记图标

根据标记聚合数量来设置自定义聚合标记图标，请调用HWMapJsSDK.[HWMarkerCluster](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151)(map, markers, options)时，设置[MarkerClusterIconOption](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151#section191218532559)中的quantityUrls属性。

```javascript
markerCluster = new HWMapJsSDK.HWMarkerCluster(map, markers, {
    icon: {
        scale: 1,
        anchor: [1, 1],
        opacity: 1,
        // 设置自定义聚合标记图标
        quantityUrls: [
            // 2-4个标记聚合时，聚合图标为m1.png
            {num:0, url:'https://xxx/m1.png'},
            // 5-14个标记聚合时，聚合图标为m2.png
            {num:5, url:'https://xxx/m2.png'},
            // 大于等于15个标记聚合时，聚合图标为m3.png
            {num:15, url:'https://xxx/m3.png'}
        ]
    }
});
```

![](https://contentcenter-vali-drcn.dbankcdn.cn/pvt_2/DeveloperAlliance_scene_100_1/26/v3/wwQL5qKFREC-MFMJtfktKQ/zh-cn_image_0000001229088795.png?HW-CC-KV=V1&HW-CC-Date=20260922T085431Z&HW-CC-Expire=31536000000&HW-CC-Sign=5F6E468047B9973C6D466360AFE93058B3F256916234155054AB2A7341EA0102 "点击放大")

在使用quantityUrls属性时，如果num不从0开始设置，例如：当num设置为15开始，则2-14个标记数量的聚合图标将使用系统默认图标。

```javascript
const quantityUrls = [
// 此时未设置num为0开始，则2-14个标记聚合时，聚合图标为系统默认图标
// 15-99个标记聚合时，聚合图标为pic2.png
{ num: 15, url: "https://xxx/pic2.png" },
// 大于等于100个标记聚合时，聚合图标为pic3.png
{ num: 100, url: "https://xxx/pic3.png" }
]
```

## 自定义聚合样式

自定义聚合标记样式，如定制聚合后的图标和文本，请调用HWMapJsSDK.[HWMarkerCluster](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151)(map, markers, options)时，自定义[MarkerClusterOptions](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151#section15708155284010)中的renderClusterMarker函数。

```screen
var locations = [
    {
        label: {text:'1', strokeColor:'white', strokeWeight:2},
        position: {lat: 49.871532, lng: 2.331602},
        // 自定义属性
        properties: {symbol: 1}, 
    },
    {
        label: {text:'2', strokeColor:'white', strokeWeight:2},
        position: {lat: 48.8638045, lng: 2.2978623},
        properties: {symbol: 2},
    },
    {
        label: {text:'3', strokeColor:'white', strokeWeight:2},
        position: {lat: 48.80434, lng: 2.12041},
        properties: {symbol: 3},
    },
    {
        label: {text:'4', strokeColor:'white', strokeWeight:2},
        position: {lat: 48.80404, lng: 2.12041},
        properties: {symbol: 4},
    },
    ......
];
let markers = [];
// 批量初始化Marker
function generateMarkers(MarkersNum) {
    for (var i = 0; i < MarkersNum; i++) {
        var opts = {
            position: locations[i].position,
            properties: locations[i].properties,
            label: locations[i].label,
        };
        markers.push(new HWMapJsSDK.HWMarker(opts));
    }
}
// 自定义聚合样式
markerCluster = new HWMapJsSDK.HWMarkerCluster(map, markers, {
    renderClusterMarker(markers) {
        let symbolAccount = 0;
        let icon = {
            url: '',
            anchor: [0.5, 0.5],
            anchorUnit: 'fraction',
        };
        let label = {
            text: '',
            color: '#000',
            strokeColor: 'white',
            strokeWeight: 2
        };
        markers.map(item=>{
            // 获取源Marker的自定义属性
            let symbolVal = item.getProperties().symbol; 
            symbolAccount += (symbolVal&&typeof(symbolVal) === 'number' ? symbolVal : 0);
        })
        label.text = symbolAccount.toString()
            if(symbolAccount <= 10) {
                icon.url = './images/round_marker1.png';
            } else if(symbolAccount < 30 && symbolAccount > 10) {
                icon.url = './images/round_marker2.png';
            } else {
                icon.url = './images/round_marker3.png';
            }
        return {
            label,
            icon
        }
    },
});
```

## 聚合标记事件

[HWMarkerCluster](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151)支持多种事件回调，如下表所示，具体请参见[HWMarkerCluster事件](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151#section582965821917)。

|**事件**|**含义**|
|:--------|:-----|
|click|鼠标点击。|
|mouseOut|鼠标移出。|
|mouseOver|鼠标移入。|

以鼠标点击事件为例，要在地图上设置此侦听器，请调用[HWMarkerCluster](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwmarkercluster-0000001050750151)对象的addListener('click', callback)方法：

```screen
function handleClick() {
      alert("On markerCluster Click!");
}
markerCluster.addListener('click', handleClick);
```
