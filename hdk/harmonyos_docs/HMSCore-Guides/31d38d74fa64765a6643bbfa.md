# 写入篮球运动记录

1. 构造篮球运动记录的特征数据结构[ActivityFeature](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/harmonyos-activity-feature-0000001233313515)。
2. 创建一个跳跃统计数据采集器及数据采集点。
3. 创建一个跳跃详情数据采集器及数据采集点。
4. 基于创建特征值、统计数据集合、paceSummary构建activitySummary。
5. 构建篮球运动记录。
6. 调用addActivityRecord接口添加篮球运动记录。

   <br />

   示例代码如下：

   ```
   import {HMSHealthKit, HMSHealthKitDataType, HMSHealthKitDeviceType, DeviceInfo, DataCollector, Value, SamplePoint, SampleSet, PaceSummary, ActivityFeature, ActivitySummary, ActivityRecord, HMSHealthKitActivityType, ActivityRecordInsertOptions} from '@hw-hmscore/hms-js-health'
   // 步骤1：构造篮球运动记录的特征数据结构ActivityFeature
   let deviceInfo = new DeviceInfo("manufacturer", "modelName", "uuid", HMSHealthKitDeviceType.TYPE_PHONE);
   let dataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_ACTIVITY_FEATURE_BASKETBALL);
   const values = [];
   values.push(new Value("overall_score", "50"));
   values.push(new Value("burst_score", "50"));
   values.push(new Value("jump_score", "50"));
   values.push(new Value("run_score", "50"));
   values.push(new Value("breakthrough_score", "50"));
   values.push(new Value("sport_intensity_score", "50"));
   let samplePoint = new SamplePoint(values, "1624774020000", "1624774320000", "here is customized text");
   let activityFeature = new ActivityFeature(dataCollector, samplePoint);

   // 步骤2：创建一个跳跃统计数据采集器及数据采集点
   let jumpStatisticsDataCollector = new DataCollector(DeviceInfo, "com.health.demo", HMSHealthKitDataType.DT_CONTINUOUS_JUMP_STATISTICS);
   const jumpStatisticsValues = [];
   jumpStatisticsValues.push(new Value("jump_times", "50"));
   jumpStatisticsValues.push(new Value("avg_jump_height", "1.66"));
   jumpStatisticsValues.push(new Value("max_jump_height", "1.66"));
   jumpStatisticsValues.push(new Value("min_jump_height", "1.66"));
   jumpStatisticsValues.push(new Value("avg_passage_duration", "500"));
   jumpStatisticsValues.push(new Value("max_passage_duration", "500"));
   jumpStatisticsValues.push(new Value("min_passage_duration", "500"));
   let jumpStatisticsSamplePoint = new SamplePoint(jumpStatisticsValues, "1624774020000", "1624774320000", "here is customized text");
   const jumpStatisticsSamplePoints = [];
   jumpStatisticsSamplePoints.push(jumpStatisticsSamplePoint);
   let jumpStatisticsSampleSet = new SampleSet(jumpStatisticsDataCollector, jumpStatisticsSamplePoints);

   // 步骤3：创建一个跳跃详情数据采集器及数据采集点
   let jumpDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_CONTINUOUS_JUMP);
   const jumpValues = [];
   jumpValues.push(new Value("jump_height", "0.88f"));
   jumpValues.push(new Value("passage_duration", "1024"));
   let jumpSamplePoint = new SamplePoint(jumpValues, "1624774020000", "1624774320000", "here is customized text");
   const jumpsamplePoints = [];
   jumpsamplePoints.push(jumpSamplePoint);
   let jumpSampleSet = new SampleSet(jumpDataCollector, jumpsamplePoints);

   const sampleSets = [];
   sampleSets.push(jumpSampleSet);

   const statisticsSampleSets = [];
   statisticsSampleSets.push(jumpStatisticsSampleSet);

   // 步骤4：基于创建特征值、统计数据集合、paceSummary构建activitySummary
   let activitySummary = new ActivitySummary(activityFeature, statisticsSampleSets, null);

   // 步骤5：构建篮球运动记录
   let activityRecord = new ActivityRecord("mysession1", "1624774020000", "1624774320000", HMSHealthKitActivityType.BASKETBALL, "here", activitySummary, "+0800", "here is customized text", deviceInfo);
   let activityRecordInsertOptions = new ActivityRecordInsertOptions(activityRecord, sampleSets);

   // 步骤6：调用addActivityRecord接口添加篮球运动记录
   HMSHealthKit.addActivityRecord(activityRecordInsertOptions).then((result) => {
       // 调用成功
       console.info("addActivityRecord success: " + JSON.stringify(result))
   }).catch((error) => {
       // 调用失败
       console.error("addActivityRecord error: " + JSON.stringify(error));
   })
   ```
