# 写入心率告警健康记录

1. 构造健康记录的采集器(以心率过缓健康数据类型为例)，以及构造健康记录结构体。
2. 创建一个心率详情数据采集器及数据采集点。
3. 创建一个心率统计数据采集器及数据采集点。
4. 创建fieldMap的值。
5. 构建健康记录。
6. 调用[addHealthRecord](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/harmonyos-manage-health-records-0000001232134027#section13164153916570)接口添加心率告警健康记录。

   写入心率告警健康记录示例代码如下：

   ```screen
   import {HMSHealthKit, HMSHealthKitDataType, HMSHealthKitDeviceType, DeviceInfo, DataCollector, Value, SamplePoint, SampleSet, HealthRecord, HealthRecordInsertOptions} from '@hw-hmscore/hms-js-health'
   // 步骤1：构造健康记录的采集器(以心率过缓健康数据类型为例)，以及构造健康记录结构体
   // 根据设备生产商、设备模型编号、设备uuid、设备类型构造设备信息
   let deviceInfo = new DeviceInfo("manufacturer", "modelName", "uuid", HMSHealthKitDeviceType.TYPE_PHONE);
   // 根据设备信息、应用包名、数据类型构造数据采集器
   let dataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_HEALTH_RECORD_TACHYCARDIA);

   // 步骤2：创建一个心率详情数据采集器及数据采集点
   // 创建心率明细数据源
   let heartRateDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_INSTANTANEOUS_HEART_RATE);
   // 设置心率明细数据值
   let heartRateValues = [];
   heartRateValues.push(new Value("bpm", 110));
   let heartRateSamplePoints = []
   heartRateSamplePoints.push(new SamplePoint(heartRateValues, 1624774020000, 1624774320000, "here is customized text"));
   let heartRateSampleSets = [];
   heartRateSampleSets.push(new SampleSet(heartRateDataCollector, heartRateSamplePoints));

   // 步骤3：创建一个心率统计数据采集器及数据采集点
   // 创建心率统计数据源
   let heartRateStatisticsDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.POLYMERIZE_CONTINUOUS_HEART_RATE_STATISTICS);
   // 设置心率统计值
   let heartRateStatisticsValues = [];
   heartRateStatisticsValues.push(new Value("avg", 110));
   heartRateStatisticsValues.push(new Value("max", 110));
   heartRateStatisticsValues.push(new Value("min", 110));
   heartRateStatisticsValues.push(new Value("last", 80));
   let heartRateStatisticsSamplePoints = [];
   heartRateStatisticsSamplePoints.push(new SamplePoint(heartRateStatisticsValues, 1624774020000, 1624774320000, "here is customized text"));
   let heartRateStatisticsSampleSets = [];
   heartRateStatisticsSampleSets.push(new SampleSet(heartRateStatisticsDataCollector, heartRateStatisticsSamplePoints));

   // 步骤4：创建fieldMap的值
   const values = []
   values.push(new Value("threshold", "120"));
   values.push(new Value("avg_heart_rate", "120"));
   values.push(new Value("max_heart_rate", "120"));
   values.push(new Value("min_heart_rate", "120"));

   // 步骤5：构建健康记录
   // 基于时间等参数创建健康记录
   let healthrecord = new HealthRecord(1624774020000, 1624774320000, dataCollector, values, "here is customized text", heartRateStatisticsSampleSets, heartRateSampleSets);
   // 创建健康记录等参数创建插入健康记录参数
   let healthRecordInsertOptions = new HealthRecordInsertOptions(healthrecord);

   // 步骤6：调用addHealthRecord接口添加心率告警健康记录
   HMSHealthKit.addHealthRecord(healthRecordInsertOptions).then((result) => {
       // 调用成功
       console.info("addHealthRecord success: " + JSON.stringify(result))
   }).catch((error) => {
       // 调用失败
       console.error("addHealthRecord fail: " + JSON.stringify(error));
   })
   ```
