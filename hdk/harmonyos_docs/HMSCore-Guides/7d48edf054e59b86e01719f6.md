# 示例代码

```javascript
var map; 
var directionsService; 
var directionsRenderer; 
var markerOrigin, markerDestination; 
// 创建路径规划请求体
var DirectionsRequest = {}; 
DirectionsRequest.origin = { 
    lat: 48.86, 
    lng: 2.36 
}; 
DirectionsRequest.destination = { 
    lat: 49.18, 
    lng: 0.39 
}; 
 
function initMap() { 
    directionsService = new HWMapJsSDK.HWDirectionsService(); 
    directionsRenderer = new HWMapJsSDK.HWDirectionsRenderer(); 
 
    var mapOptions = {}; 
    mapOptions.center = {lat: 48.856613, lng: 2.352222}; 
    mapOptions.zoom = 8; 
    mapOptions.language = 'ENG'; 
 
    map = new HWMapJsSDK.HWMap(document.getElementById('map'), mapOptions); 
 
    markerOrigin = new HWMapJsSDK.HWMarker({ 
        position: DirectionsRequest.origin, 
        label: 'Origin' 
    }); 
 
    markerDestination = new HWMapJsSDK.HWMarker({ 
        position: DirectionsRequest.destination, 
        label: 'Destination' 
    }); 
 
    markerOrigin.setMap(map); 
    markerDestination.setMap(map); 
    // 调用路径规划接口 
    directionsService.routeDriving(DirectionsRequest, callback); 
    directionsRenderer.setMap(map); 
    // 通过回调函数获取路径规划结果 
    function callback(DirectionsResult, DirectionsStatus) { 
        if (DirectionsStatus == '0') { 
            // 通过HWDirectionsRenderer对象渲染路径 
            directionsRenderer.setDirections(DirectionsResult); 
        } 
    } 
}
```

## 展示路况信息

通过调用HWMapJsSDK.[HWNaviLine](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwnaviline-0000001171262730)(naviLineOption)，支持在路径规划时展示路况信息：

* 可根据路况设置颜色分段，展示交通拥堵情况。
* 支持显示道路流向方向的箭头，可设置显示或不显示。
* 支持显示不同的样式（主、备选路线）。
* 支持路名显示。
* 支持蚯蚓线从起点到终点的动画显示。
* 支持蚯蚓线点击事件。

### 蚯蚓线生长动画

调用HWMapJsSDK.[HWNaviLine](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwnaviline-0000001171262730)(naviLineOption)时，通过[NaviLineOptions](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwnaviline-0000001171262730#section366163604219)中的animateOption属性设置蚯蚓线生长动画。

```javascript
function initMap() {

    var mapOptions = {};
    mapOptions.center = {lat: 48.856613, lng: 2.352222};
    mapOptions.zoom = 12;
    mapOptions.language = 'en'

    map = new HWMapJsSDK.HWMap(document.getElementById('map'), mapOptions);

    firstNaviLine = new HWMapJsSDK.HWNaviLine({
        map,
        points: points[0],
        strokeColor: '#c7b18c',
        strokeWeight: 1,
        fillColor: '#fce0b0',
        fillWidth: 5,
        textSize: 22,
        textColor: 'hsl(187,100%,42%)',
        collisionEnable: true,
        zIndex: 2,
        NaviLineFillColors: fillColors[0],
        NaviLineTexts: lineTexts[0],
        NaviLineCodes: lineCodes[0],
        animateOption:{
            // 蚯蚓线是否展示动画，默认false
            animate: true,
            // 蚯蚓线生长动画时长，默认1000ms
            timeout: 1000
        }
    })
}
```

### 蚯蚓线事件

调用[HWNaviLine](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/js-hwnaviline-0000001171262730)的addListener(type, callback)、removeListener(type, callback)方法，添加/删除点击事件。

示例代码如下：

```javascript
"JavaScript"
function handler(e){
    let naviline=e.target
    let colors=''
    if(!infoWindow){
        for(let i=0;i<naviline._fillColors.colors.length;i++){
            colors+= '<p>'+naviline._fillColors.colors[i]+'</p>'
        }
        infoWindow = new HWMapJsSDK.HWInfoWindow({
            map,
            position:HWMapJsSDK.HWMapUtils.epsgToLatLng(e.coordinate),
            content: colors
        })
    }else{
        infoWindow.setPosition(HWMapJsSDK.HWMapUtils.epsgToLatLng(e.coordinate))
    }
}

// 添加点击事件
function addListener(){
    firstNaviLine.addListener('click', handler)
}
// 删除点击事件
function removeListener(){
    firstNaviLine.removeListener('click',handler)
}
```
