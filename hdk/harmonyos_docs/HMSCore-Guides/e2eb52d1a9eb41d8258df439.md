# 屏幕状态感知能力

#### 导入接口类

使用屏幕状态感知能力，除需要导入情景感知服务的公共能力类外，还需要导入屏幕状态相关的类。

```
import com.huawei.hmf.tasks.OnFailureListener;
import com.huawei.hmf.tasks.OnSuccessListener;
import com.huawei.hms.kit.awareness.Awareness;
// 导入屏幕快照相关类
import com.huawei.hms.kit.awareness.capture.ScreenStatusResponse;
import com.huawei.hms.kit.awareness.status.ScreenStatus;
// 导入屏幕围栏相关类
import com.huawei.hms.kit.awareness.barrier.AwarenessBarrier;
import com.huawei.hms.kit.awareness.barrier.BarrierStatus;
import com.huawei.hms.kit.awareness.barrier.ScreenBarrier;
import com.huawei.hms.kit.awareness.barrier.BarrierUpdateRequest;
```

#### 能力开发

#### Capture

1. 获取Awareness Kit的"Capture Client"。
2. 通过"Capture Client"调用屏幕状态查询能力接口查询情景状态。
3. 根据情景感知服务返回的结果，进行应用的业务处理。  

   ```
   "Java"
   Awareness.getCaptureClient(context).getScreenStatus()
                   // 执行成功的回调监听
                   .addOnSuccessListener(screenStatusResponse -> {
                       int state = screenStatusResponse.getScreenStatus().getStatus();
                       switch (state) {
                           case ScreenStatus.UNKNOWN:
                               Log.i("TAG","it is unknown");
                           case ScreenStatus.SCREEN_OFF:
                               Log.i("TAG","it is screen off");
                           case ScreenStatus.SCREEN_ON:
                               Log.i("TAG","it is screen on");
                           case ScreenStatus.UNLOCK:
                               Log.i("TAG","it is unlock");
                           default:
                               Log.i("TAG","it is sth wrong");
                       }
                   })
                   // 执行失败的回调监听
                   .addOnFailureListener(e->{
                       Log.e("TAG","get screen state failed");
                       e.printStackTrace();
                   });
   ```

   ```
   "Kotlin"
   Awareness.getCaptureClient(this).screenStatus
       // 执行成功的回调监听
       .addOnSuccessListener { screenStateResponse: ScreenStatusResponse ->
       val state = screenStateResponse.screenStatus.status    
       val screenState: String    
           screenState = when (state) {        
              ScreenStatus.UNKNOWN -> "it is unknown"        
              ScreenStatus.SCREEN_OFF -> "it is screen off"        
              ScreenStatus.SCREEN_ON -> "it is screen on"        
              ScreenStatus.UNLOCK -> "it is unlock"        
              else -> "it is sth wrong"    
           }
       }
       // 执行失败的回调监听
       .addOnFailureListener { e: Exception ->
           Log.e("TAG","get screen state failed") 
           e.printStackTrace()
   }
   ```

#### Barrier

以下以开发屏幕"unlock"的Barrier为例（屏幕解锁后触发Barrier）。

1. 定义Barrier。  

   ```
   "Java"
   AwarenessBarrier screenBarrier = ScreenBarrier.screenUnlock();
   ```

   ```
   "Kotlin"
   var screenBarrier : AwarenessBarrier = ScreenBarrier.screenUnlock()
   ```

2. 定义Barrier状态改变时触发的"PendingIntent"，同时注册一个广播接收器用来接收这个广播。  

   ```
   "Java"
   String actionReceiver = "SCREEN_ACTION";
   ScreenReceiver screenReceiver = new ScreenReceiver();
   Intent intent = new Intent(actionReceiver);
   PendingIntent mPendingIntent = PendingIntent.getBroadcast(this, 1, intent, PendingIntent.FLAG_UPDATE_CURRENT);
   registerReceiver(screenReceiver, new IntentFilter(actionReceiver));
   ```

   ```
   "Kotlin"
   val actionReceiver = "SCREEN_ACTION"
   val screenReceiver = ScreenReceiver()
   val intent = Intent(actionReceiver)
   var mPendingIntent = PendingIntent.getBroadcast(this, 1, intent, PendingIntent.FLAG_UPDATE_CURRENT)
   registerReceiver(screenReceiver, IntentFilter(actionReceiver))
   ```

3. 定义Barrier对应的标签Label，然后添加Barrier。  

   ```
   "Java"
   String barrierKey = "screen barrier key";
   // 定义更新围栏的请求
   BarrierUpdateRequest request =new BarrierUpdateRequest.Builder()
           .addBarrier(barrierKey, screenBarrier, mPendingIntent).build();
   Task<Void> task = Awareness.getBarrierClient(this).updateBarriers(request);
   // 执行成功的回调监听
   task.addOnSuccessListener(
           ret -> {
               Log.d(TAG, "add barrier: success");
           });
   // 执行失败的回调监听
   task.addOnFailureListener(
           e -> {
               Log.d(TAG, "add barrier: failed");
               e.printStackTrace();
           });
   ```

   ```
   "Kotlin"
   val barrierKey = "screen barrier key"
   // 定义更新围栏的请求
   val request = BarrierUpdateRequest.Builder()
           .addBarrier(barrierKey, screenBarrier, mPendingIntent).build()
   val task = Awareness.getBarrierClient(this).updateBarriers(request)
   // 执行成功的回调监听
   task.addOnSuccessListener {
       ret: Void? -> Log.d(TAG, "add barrier: success")
       }
   // 执行失败的回调监听
   task.addOnFailureListener { e: Exception ->
       Log.d(TAG, "add barrier: failed")
       e.printStackTrace()
       }
   ```

4. 定义广播接收器，用于监听Barrier事件，收到事件后进行应用的业务处理。  

   ```
   "Java"
   // 定义广播接收器
   class ScreenReceiver extends BroadcastReceiver{
       @Override
       public void onReceive(Context context, Intent intent) {
           BarrierStatus barrierStatus = BarrierStatus.extract(intent);
           String label =  barrierStatus.getBarrierLabel();
           switch (barrierStatus.getPresentStatus()) {
               case BarrierStatus.TRUE:
                   Log.i(TAG, label + " status:true");
                   break;
               case BarrierStatus.FALSE:
                   Log.i(TAG, label + " status:false");
                   break;
               case BarrierStatus.UNKNOWN:
                   Log.i(TAG, label + " status:unknown");
                   break;
           }
       }
   }
   ```

   ```
   "Kotlin"
   // 定义广播接收器
   internal inner class ScreenReceiver : BroadcastReceiver() {
   override fun onReceive(context: Context, intent: Intent) {
   val barrierStatus = BarrierStatus.extract(intent)
   val label = barrierStatus.barrierLabel
           when (barrierStatus.presentStatus) {
               BarrierStatus.TRUE -> Log.i(TAG, "$label status:true")
               BarrierStatus.FALSE -> Log.i(TAG, "$label status:false")
               BarrierStatus.UNKNOWN -> Log.i(TAG, "$label status:unknown")
           }
       }
   }
   ```

5. 在应用业务处理完成后，根据Barrier对应的标签Label，进行删除Barrier。  

   ```
   "Java"
   String barrierKey = "screen barrier key";
   // 定义更新围栏的请求
   BarrierUpdateRequest request =new BarrierUpdateRequest.Builder()
           .deleteBarrier(barrierKey).build();
   Task<Void> task = Awareness.getBarrierClient(this).updateBarriers(request);
   // 执行成功的回调监听
   task.addOnSuccessListener(
           ret -> {
               Log.d(TAG, "delete barrier: success");
           });
   // 执行失败的回调监听
   task.addOnFailureListener(
           e -> {
               Log.d(TAG, "delete barrier: failed");
               e.printStackTrace();
           });
   ```

   ```
   "Kotlin"
   val barrierKey = "screen barrier key"
   // 定义更新围栏的请求
   val request = BarrierUpdateRequest.Builder()
           .deleteBarrier(barrierKey).build()
   ```
