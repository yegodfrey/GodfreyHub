# 写入跑步运动记录

1. 创建距离统计、速度统计、总步数统计的数据采集器。
2. 创建一个步频详情采集器及数据采集点。
3. 创建activitySummary。
4. 构建运动记录。
5. 调用[addActivityRecord](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/harmonyos-manage-activity-records-0000001186732884#section103608115712)接口添加运动记录。

   写入跑步运动记录示例代码如下：

   ```screen
   import {HMSHealthKit, HMSHealthKitDataType, HMSHealthKitDeviceType, DeviceInfo, DataCollector, Value, SamplePoint, SampleSet, PaceSummary, ActivitySummary, ActivityRecord, HMSHealthKitActivityType, ActivityRecordInsertOptions} from '@hw-hmscore/hms-js-health'
   // 步骤1：创建距离统计、速度统计、总步数统计的数据采集器
   // 根据设备生产商、设备模型编号、设备uuid、设备类型构造设备信息
   let deviceInfo = new DeviceInfo("manufacturer", "modelName", "uuid", HMSHealthKitDeviceType.TYPE_PHONE);

   let distanceTotalDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_CONTINUOUS_DISTANCE_TOTAL);
   const distanceTotalValues = [];
   distanceTotalValues.push(new Value("distance", "50"));
   let distanceTotalSamplePoint = new SamplePoint(distanceTotalValues, "1624774020000", "1624774320000", "here is customized text");
   const distanceTotalSamplePoints = [];
   distanceTotalSamplePoints.push(distanceTotalSamplePoint);
   let distanceTotalSampleSet = new SampleSet(distanceTotalDataCollector, distanceTotalSamplePoints);

   let speedStatisticsDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.POLYMERIZE_CONTINUOUS_SPEED_STATISTICS);
   const speedStatisticsValues = [];
   speedStatisticsValues.push(new Value("avg", "50"));
   speedStatisticsValues.push(new Value("max", "50"));
   speedStatisticsValues.push(new Value("min", "50"));
   let speedStatisticsSamplePoint = new SamplePoint(speedStatisticsValues, "1624774020000", "1624774320000", "here is customized text");
   const speedStatisticsSamplePoints = [];
   speedStatisticsSamplePoints.push(speedStatisticsSamplePoint);
   let speedStatisticsSampleSet = new SampleSet(speedStatisticsDataCollector, speedStatisticsSamplePoints);

   let stepsTotalDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_CONTINUOUS_STEPS_TOTAL);
   const stepsTotalValues = [];
   stepsTotalValues.push(new Value("steps", "50"));
   let stepsTotalSamplePoint = new SamplePoint(stepsTotalValues, "1624774020000", "1624774320000", "here is customized text");
   const stepsTotalSamplePoints = [];
   stepsTotalSamplePoints.push(stepsTotalSamplePoint);
   let stepsTotalSampleSet = new SampleSet(stepsTotalDataCollector, stepsTotalSamplePoints);

   // 步骤2：创建一个步频详情采集器及数据采集点
   let stepDataCollector = new DataCollector(deviceInfo, "com.health.demo", HMSHealthKitDataType.DT_INSTANTANEOUS_STEPS_RATE);
   const stepValues = [];
   stepValues.push(new Value("step_rate", "100"));
   let stepSamplePoint = new SamplePoint(stepValues, "1624774020000", "1624774320000", "here is customized text");
   const stepSamplePoints = [];
   stepSamplePoints.push(stepSamplePoint);
   let stepSampleSet = new SampleSet(stepDataCollector, stepSamplePoints);

   const sampleSets = [];
   sampleSets.push(stepSampleSet);
   const statisticsSampleSets = [];
   statisticsSampleSets.push(distanceTotalSampleSet);
   statisticsSampleSets.push(speedStatisticsSampleSet);
   statisticsSampleSets.push(stepsTotalSampleSet);

   // 步骤3：创建activitySummary
   const partTimeMap = [];
   partTimeMap.push(new Value(1.0, 456));
   const paceMap = [];
   paceMap.push(new Value(1.0, 263));
   // 创建paceSummary
   let paceSummary = new PaceSummary(paceMap, partTimeMap, 247.27626, 212.0);
   // 基于创建特征值、统计数据集合、paceSummary构建activitySummary
   let activitySummary = new ActivitySummary(null, statisticsSampleSets, paceSummary);

   // 步骤4：构建运动记录
   let activityRecord = new ActivityRecord("mysession1", "1624774020000", "1624774320000", HMSHealthKitActivityType.RUNNING, "here", activitySummary, "+0800", "here is customized text", deviceInfo);
   let activityRecordInsertOptions = new ActivityRecordInsertOptions(activityRecord, sampleSets);

   // 步骤5：调用addActivityRecord接口添加运动记录
   HMSHealthKit.addActivityRecord(activityRecordInsertOptions).then((result) => {
       // 调用成功
       console.info("addActivityRecord success: " + JSON.stringify(result))
   }).catch((error) => {
       // 调用失败
       console.error("addActivityRecord error: " + JSON.stringify(error));
   })
   ```
