# 相机动画

HMapView提供了相机动画功能，通过setAnnotationAndLineWithAnimateCamera方法设置，支持相机和标记随路线运动。

```
@interface ViewController ()<HMapViewDelegateAdvance>
@property (nonatomic,strong) HMapView *mapView;
@property (nonatomic,strong) HRouteLine *routeLine;
@property (nonatomic,strong) HAnnotationView *annotationView;
@end
@implementation ViewController

- (void)viewDidLoad {
    [super viewDidLoad];
    HMapView *mapView = [[HMapView alloc] init];
    mapView.frame = UIScreen.mainScreen.bounds;
    [self.view addSubview:mapView];
    self.mapView = mapView;
    
    HRouteLineOptions *hOptions = [[HRouteLineOptions alloc] init];
    hOptions.mPoints = @[
        [[HLatLng alloc]initLatitude:45.002506 withLongitude:2.332645],
        [[HLatLng alloc]initLatitude:45.00490506 withLongitude:2.332645],
        [[HLatLng alloc]initLatitude:45.00490506 withLongitude:2.3334685],
        [[HLatLng alloc]initLatitude:45.002506 withLongitude:2.3334685],
    ].mutableCopy;
    hOptions.color = UIColor.greenColor;
    hOptions.strokeColor = UIColor.redColor;
    hOptions.width = 10;
    
    self.routeLine = [mapView addRouteLine:hOptions];
    self.routeLine.arrowRendered = YES;
    
    HAnnotationOptions *aOptions = [[HAnnotationOptions alloc] init];
    aOptions.position = [[HLatLng alloc] initLatitude:48.8129154 withLongitude:2.3364656];
    HAnnotationView *annotationView = [mapView addAnnotationEx:aOptions];
    
    self.annotationView = annotationView;
}

-(void)moveCamera {
    HMapMessage *msg = [[HMapMessage alloc] initCoordinate:(CLLocationCoordinate2DMake(48.8129154, 2.3364656))];
    HCameraUpdate *update = [[HCameraUpdate alloc] initWithHMapMessage:msg];
    [self.mapView moveCamera:update];
}

-(void)annotationAndLineWithAnimateCamera {
    HMapMessage *msg = [[HMapMessage alloc] initCoordinate:(CLLocationCoordinate2DMake(45.00490506, 2.332645))];
    HCameraUpdate *update = [[HCameraUpdate alloc] initWithHMapMessage:msg];
    [self.mapView setAnnotationAndLineWithAnimateCamera:update duration:3000 annotationView:self.annotationView naviline:self.routeLine index:0];
}
@end
```
