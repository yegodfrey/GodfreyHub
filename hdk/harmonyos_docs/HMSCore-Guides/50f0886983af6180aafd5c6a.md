# 写入自由潜水运动记录

用户可以调用[addActivityRecord](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/activityrecordscontroller-0000001050091295#section1781411218216)方法插入自由潜水活动，使用步骤如下：

1. 使用[自由潜](https://developer.huawei.com/consumer/cn/doc/HMSCore-Guides/freediving-0000001192970763)数据类型，在[ActivitySummary](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/activity-summary-0000001055215648)结构中添加自由潜水活动的特征数据结构ActivityFeature。
2. 构造[ActivityRecord](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/activityrecord_class-0000001050161730)，设置activityType名称为自由潜水，名称取值参见[HiHealthActivities](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/hihealthactivities-0000001050089540)。
3. 构造[ActivityRecordInsertOptions](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/ario_class-0000001050091110)，使用[ActivityRecordsController](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/activityrecordscontroller-0000001050091295).[addActivityRecord](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/activityrecordscontroller-0000001050091295#section1781411218216)插入自由潜水运动记录数据到Health Service Kit。

```screen
// 请注意此处的this为Activity对象
final ActivityRecordsController activityRecordsController = HuaweiHiHealth.getActivityRecordsController(this);

// 1.构造请求对象的时间区间：开始和结束时间
// 请注意此处设置的开始时间和结束时间均需大于2014年1月1号对应的UNIX时间戳
Calendar cal = Calendar.getInstance();
Date now = new Date();
cal.setTime(now);
long endTime = cal.getTimeInMillis();
cal.add(Calendar.HOUR_OF_DAY, -1);
long startTime = cal.getTimeInMillis();

// 2.根据自由潜水活动，构造自由潜水活动的特征数据结构ActivityFeature,并将其添加至ActivitySummary结构中
SamplePoint activityFeature = new SamplePoint.Builder(DataType.DT_ACTIVITY_FEATURE_FREEDIVING).build();
activityFeature.setTimeInterval(startTime + 1L, startTime + 300000L, TimeUnit.MILLISECONDS);
activityFeature.getFieldValue(Field.DIVING_TIME).setIntValue(20);
activityFeature.getFieldValue(Field.DIVING_COUNT).setIntValue(2);
activityFeature.getFieldValue(Field.MAX_DEPTH).setDoubleValue(2d);
activityFeature.getFieldValue(Field.AVG_DEPTH).setDoubleValue(20d);
activityFeature.getFieldValue(Field.MAX_UNDERWATER_TIME).setIntValue(16);
activityFeature.getFieldValue(Field.NO_FLY_TIME).setIntValue(50);
activityFeature.getFieldValue(Field.WATER_TYPE).setIntValue(20);
activityFeature.getFieldValue(Field.SURFACE_TIME).setIntValue(3);
ActivitySummary activitySummary = new ActivitySummary();
activitySummary.setActivityFeature(activityFeature);


// 3.构造统计数据样本点，并添加至DataSummary对象
SamplePoint statisticsSamplePoint = new SamplePoint.Builder(DataType.DT_CONTINUOUS_EXERCISE_HEART_RATE_STATISTICS).build();
statisticsSamplePoint.setTimeInterval(startTime + 1L, startTime + 300000L, TimeUnit.MILLISECONDS);
statisticsSamplePoint.getFieldValue(Field.FIELD_AVG).setDoubleValue(0.12d);
statisticsSamplePoint.getFieldValue(Field.FIELD_MAX).setDoubleValue(0.16d);
statisticsSamplePoint.getFieldValue(Field.FIELD_MIN).setDoubleValue(0.08d);
statisticsSamplePoint.getFieldValue(Field.FIELD_LAST).setDoubleValue(0.16d);

// ActivitySummary中的DataSummary用来承载统计数据
List<SamplePoint> dataSummary = new ArrayList<>();
dataSummary.add(statisticsSamplePoint);
activitySummary.setDataSummary(dataSummary);

// ActivitySummary中的SectionSummary用来承载统计数据
SamplePoint samplePointSectionFreeDivingSamplePoint = new SamplePoint.Builder(DataType.POLYMERIZE_FREEDIVING_STATISTICS).build();
samplePointSectionFreeDivingSamplePoint.getFieldValue(Field.DIVING_TIME).setIntValue(100);
samplePointSectionFreeDivingSamplePoint.getFieldValue(Field.MAX_DEPTH).setDoubleValue(1.5d);
samplePointSectionFreeDivingSamplePoint.getFieldValue(Field.SURFACE_TIME).setIntValue(100);
SampleSection sampleSection = new SampleSection.Builder()
    .setSectionNum(1)
    .setSectionTime(60 * 5L, TimeUnit.SECONDS)
    .setStartTime(startTime + 1L, TimeUnit.MILLISECONDS)
    .setEndTime(startTime + 5 * 60 * 1000L, TimeUnit.MILLISECONDS)
    .setSectionDataList(Collections.singletonList(samplePointSectionFreeDivingSamplePoint))
    .build();
activitySummary.setSectionSummary(Collections.singletonList(sampleSection));

// 4.构造运动记录，设置潜水活动的ActivitySummary，与ActivityTypeId
ActivityRecord activityRecord = new ActivityRecord.Builder()
    .setName("AddActivityRecord")
    .setDesc("This is ActivityRecord add test!")
    .setId("MyAddActivityRecordId")
    .setActivityTypeId(HiHealthActivities.FREE_DIVING)
    .setStartTime(startTime, TimeUnit.MILLISECONDS)
    .setEndTime(endTime, TimeUnit.MILLISECONDS)
    .setActivitySummary(activitySummary)
    .setMetadata("this is a free diving activity test")
    .setTimeZone("+0800")
    .build();

// 5.创建一个运动心率采样明细数据采集器
DataCollector heartDetaildataCollector = new com.huawei.hms.hihealth.data.DataCollector.Builder()
    .setDataType(DataType.DT_INSTANTANEOUS_EXERCISE_HEART_RATE)
    .setDataGenerateType(DataCollector.DATA_TYPE_RAW)
    .setPackageName(getApplicationContext())
    .setDataCollectorName("AddActivityRecord")
    .build();
// 6.根据明细数据采集器，构造采样集
SampleSet sampleSet = SampleSet.create(heartDetaildataCollector);

// 7.构造运动心率采样明细明细（DT_INSTANTANEOUS_EXERCISE_HEART_RATE）采样点对象，添加到明细数据采样集SampleSet
SamplePoint heartDetailSamplePoint = sampleSet.createSamplePoint().setTimeInterval(startTime, endTime, TimeUnit.MILLISECONDS);
heartDetailSamplePoint.getFieldValue(Field.FIELD_BPM).setDoubleValue(0.15d);
sampleSet.addSample(heartDetailSamplePoint);

// 8.构造运动记录请求，设置运动记录activityRecord对象与其关联的明细数据采样集SampleSet
ActivityRecordInsertOptions insertOption = new ActivityRecordInsertOptions.Builder()
    .setActivityRecord(activityRecord)
    .addSampleSet(sampleSet)
    .build();

// 9.调用addActivityRecord方法添加自由潜水运动记录
Task<Void> addTask = activityRecordsController.addActivityRecord(insertOption);
addTask.addOnSuccessListener(new OnSuccessListener<Void>() {
    @Override
    public void onSuccess(Void aVoid) {
        Log.i("ActivityRecords", "ActivityRecord add was successful!");
    }
}).addOnFailureListener(new OnFailureListener() {
    @Override
    public void onFailure(Exception e) {
        Log.i("ActivityRecords", "Failed to get ActivityRecord" + e.getMessage());
    }
});
```
