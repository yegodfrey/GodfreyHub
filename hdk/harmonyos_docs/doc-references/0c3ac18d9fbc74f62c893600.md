    public enum ZSONType

---  
Enumerates JSON data types.  You can use this enum to determine which method in [ZSONReader](https://developer.harmonyos.com/cn/docs/documentation/doc-references/zsonreader-0000001060268813) is used to read data. An example is provided as follows: 
    
    String text = "{\"array\": [1, 2.33, true, \"abc\", null]}";
     try (ZSONReader reader = new ZSONReader(new StringReader(text))) {
         reader.startObject();
         if (reader.peek() == ZSONType.NAME) {
             reader.readName();
         }
         if (reader.peek() == ZSONType.ARRAY) {
             reader.startArray();
             while (reader.hasNext()) {
                 switch (reader.peek()) {
                     case INTEGER: {
                         reader.readInt(); // or reader.readLong()
                         break;
                     }
                     case FLOAT: {
                         reader.readFloat(); // or reader.readDouble()
                         break;
                     }
                     case BOOLEAN: {
                         reader.readBoolean();
                         break;
                     }
                     case STRING: {
                         reader.readString();
                         break;
                     }
                     case NULL: {
                         reader.readNull();
                         break;
                     }
                     default: {
                         throw new ZSONException("error type " + reader.peek());
                     }
                 }
             }
             reader.endArray();
         }
     }
