菜单 在页面范围内关闭通过[bindContextMenu](https://developer.harmonyos.com/cn/docs/documentation/doc-references/ts-universal-attributes-menu-0000001281201186#ZH-CN_TOPIC_0000001281201186__entry1435796836174016p0)属性绑定的菜单。 ![](https://communityfile-drcn.op.dbankcloud.cn/FileServer/getFile/cmtyPub/011/111/111/0000000000011111111.20230804172446.02279662631288145784090192103480:50001231000000:2800:C6937C4605413662743E49CABCA1F5E60555E85E4E02ECAFFE08F1473CEE6C27.png?needInitFileName=true?needInitFileName=true?needInitFileName=true) 从 API Version 8 开始支持。后续版本如有新增内容，则采用上角标单独标记该内容的起始版本。 ContextMenu.closeclose(): void 可以通过该方法在页面范围内关闭通过[bindContextMenu](https://developer.harmonyos.com/cn/docs/documentation/doc-references/ts-universal-attributes-menu-0000001281201186#ZH-CN_TOPIC_0000001281201186__entry1435796836174016p0)给组件绑定的菜单。 示例
    
    // xxx.ets
    @Entry
    @Component
    struct Index {
      @Builder MenuBuilder(){
        Flex({ direction: FlexDirection.Column, alignItems: ItemAlign.Center, justifyContent: FlexAlign.Center }) {
          Text('close')
            .fontSize(30)
            .fontWeight(FontWeight.Bold)
            .onClick(() => {
              ContextMenu.close();
            })
        }.height(400)
        .backgroundColor(Color.Pink)
      }
    
      build() {
        Flex({ direction: FlexDirection.Column, alignItems: ItemAlign.Start, justifyContent: FlexAlign.Start }) {
          Column(){
            Text("Text")
          }.bindContextMenu(this.MenuBuilder, ResponseType.LongPress)
        }
        .width('100%')
        .height('100%')
      }
    }
