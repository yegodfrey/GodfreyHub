    public interface ISensorDataCallback<D extends [SensorData](https://developer.harmonyos.com/cn/docs/documentation/doc-references/sensordata-0000001054838938)<S>,S extends [SensorBase](https://developer.harmonyos.com/cn/docs/documentation/doc-references/sensorbase-0000001054238981)>
