# 科普文章

支持获取文章及调查问卷，提供研究相关的接口。需集成uawei-research-android-common-x.x.x.aar、huawei-research-android-bridge-x.x.x.aar、huawei-research-android-research-x.x.x.aar。

HUAWEI Research平台提供科普文章管理能力，项目管理员在HUAWEI Research平台"研究管理"->"科普文章"中添加文章，如下图所示：

研究APP中通过如下示例代码获取科普文章列表：

```codeblock
import com.huawei.hiresearch.bridge.BridgeManager2; 
import com.huawei.hiresearch.bridge.model.bridge.ArticleInfo; 
import com.huawei.hiresearch.bridge.model.bridge.ArticleInfos; 
import com.huawei.hiresearch.bridge.model.bridge.PresignConstrainedUrlInfo; 
import com.huawei.hiresearch.bridge.provider.CommonProvider; 
import com.huawei.hiresearch.common.model.PageInfo; 
import com.huawei.hiresearch.research.ResearchManager2; 
import com.huawei.hiresearch.research.provider.ArticleProvider; 
import io.reactivex.disposables.CompositeDisposable; 
private final CompositeDisposable mDisposable = new CompositeDisposable(); 
private ArticleProvider articleProvider; 
private CommonProvider commonProvider; 
@Override 
protected void onDestroy() { 
   super.onDestroy(); 
   mDisposable.dispose(); 
} 
/** 
* 获取科普文章列表 
* 
* @param projectCode 申请研究项目时所获取的项目编码 
*/ 
private void getArticleList(String projectCode) { 
   articleProvider = ResearchManager2.getInstance(projectCode).getArticleProvider(); 
   commonProvider = BridgeManager2.getInstance(projectCode).getCommonProvider(); 
   //分页信息，每页20条记录，从第一页开始 
   PageInfo pageInfo = new PageInfo(20, 1); 
   mDisposable.add(articleProvider.getArticleList(pageInfo) 
           .subscribe(resp -> { 
               if (resp.getSuccess()) { 
                   //TODO：科普文章获取成功 
                   ArticleInfos articleInfos = resp.getData(); 
                   if (null != articleInfos && null != articleInfos.getArticleList()) { 
                       for (ArticleInfo articleInfo : articleInfos.getArticleList()) { 
                           //文章ID 
                           String articleId = articleInfo.getArticleId(); 
                           //文章标题 
                           String title = articleInfo.getTitle(); 
                           //文章链接 
                           String url = articleInfo.getUrl(); 
                           //文章作者 
                           String authorname = articleInfo.getAuthorname(); 
                           //文章类型 
                           String type = articleInfo.getType(); 
                           //缩略图云存储id 
                           String thumbnailId = articleInfo.getThumbnailId(); 
                           //Banner图片云存储ID 
                           String bannerId = articleInfo.getBannerId(); 
                           //是否Banner 0: 否 1: 是 
                           boolean isBanner = articleInfo.isBanner(); 
                           //弹框图片云存储ID 
                           String dialogId = articleInfo.getDialogId(); 
                           //是否弹框 0: 否 1: 是 
                           boolean isDialog = articleInfo.isDialog(); 
                           //最后更新时间 
                           String updateTime = articleInfo.getUpdateTime(); 
                           //文章类型名称 
                           String typeName = articleInfo.getTypeName(); 
                       } 
                   } 
               } else { 
                   //TODO：分页获取科普文章列表失败 
                   int statusCode = resp.getStatusCode(); //HTTP状态码 
                   String code = resp.getCode(); //内部错误码 
                   String msg = resp.getMessage(); //错误信息描述 
               } 
           }, err -> { 
               //TODO: 分页获取科普文章列表异常 
           })); 
} 
// 根据图片ID 获取图片下载url地址 
private void getPresignConstrainedObjectURL(String id) { 
   mDisposable.add(commonProvider.getPresignConstrainedObjectURL(id, "thumbnail") 
           .subscribe(resp -> { 
               if (resp.getSuccess()) { 
                   //TODO:登录成功 
                   PresignConstrainedUrlInfo presignConstrainedUrlInfo = resp.getData(); 
                   String imgUrl = presignConstrainedUrlInfo.getUrl(); 
               } else { 
                   //TODO：登录失败 
                   int statusCode = resp.getStatusCode(); //HTTP状态码 
                   String code = resp.getCode(); //内部错误码 
                   String msg = resp.getMessage(); //错误信息描述 
               } 
           }, err -> { 
               //TODO:登录异常 
           })); 
}
```
