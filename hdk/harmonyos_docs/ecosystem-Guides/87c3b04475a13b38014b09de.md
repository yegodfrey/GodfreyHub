在数据表页面创建数据表testStandard关联元数据testStandard。

插入数据至该数据表，脚本如下：

```codeblock
from hiresearchsdk import BridgeClient
from hiresearchsdk.config import BridgeConfig, HttpClientConfig
from hiresearchsdk.model import AuthRequest
from hiresearchsdk.model.table import WriteTableDataRequest
from hiresearchsdk.schemas.basictypes import TimeFrame
from hiresearchsdk.schemas.enums import TemporalRelationshipToPhysicalActivity
# 初始化BridgeConfig类
from hiresearchsdk.schemas.standardfields.health import RRI
from hiresearchsdk.schemas.standardfields.sports import CaloriesBurned, Distance, PhysicalActivity
from hiresearchsdk.schemas.standardfields.unitvalue import LengthUnitValue, KcalUnitValue, RRIUnitValue
from hiresearchsdk.schemas.units import LengthUnit, DurationUnit, KcalUnit
from hiresearchsdk.schemas.enums import ReportedActivityIntensity, DescriptiveStatistic, DescriptiveStatisticDenominator,\
    TemporalRelationshipToMeal
bridgeconfig = BridgeConfig("<yourPojectCode>", "product")
# 连接超时时间，单位s，不设置则默认30s
connect_timeout = 20
# 等待接口返回超时时间，单位s，不设置则默认30s
read_timeout = 20
# 是否失败重试，默认不重试
retry_on_fail = True
# 初始化HttpClientConfig类
httpconfig = HttpClientConfig(connect_timeout, read_timeout, retry_on_fail)
bridgeclient = BridgeClient(bridgeconfig, httpconfig)
request = AuthRequest("<yourAccessKey>", "<yourSecretKey>")
authResponse = bridgeclient.get_authenticate_provider().auth(request)
sessiontoken = authResponse.get_accessToken()
print("sessiontoken:",authResponse.get_accessToken())
# physicalActivity身体活动
timeFrame = TimeFrame(1594361431757, 1594361431800, None)
descriptiveStatistic = DescriptiveStatistic.STANDARD_DEVIATION
descriptiveStatisticDenominator = DescriptiveStatisticDenominator.EPISODE
kcalUnit = KcalUnit.KILOCALORIE
kcalUnitValue = KcalUnitValue(70.52, kcalUnit)
lengthUnit = LengthUnit.KILOMETER
lengthUnitValue = LengthUnitValue(942, lengthUnit)
distance = Distance(lengthUnitValue, descriptiveStatisticDenominator, timeFrame,descriptiveStatistic, "测试用户distance")
caloriesBurned = CaloriesBurned(caloriesBurned=kcalUnitValue,activityName="骑车",descriptiveStatisticDenominator=descriptiveStatisticDenominator,
                                timeFrame=timeFrame,descriptiveStatistic=descriptiveStatistic,userNotes="测试用户caloriesBurned")
reportedActivityIntensity = ReportedActivityIntensity.MODERATE
physicalActivity = PhysicalActivity("运动", timeFrame, distance, caloriesBurned, reportedActivityIntensity, 78, "physicalActivity测试")
# temporalRelationshipToMeal就餐状态，枚举类型
temporalRelationshipToMeal = TemporalRelationshipToMeal.BEFORE_MEAL
# timeFrame时间
timeFrame = TimeFrame(1594361431757, 1594361431757, None)
# rri数组
durationUnit = DurationUnit.SECOND
rRIUnitValue = RRIUnitValue(59.6, durationUnit)
temporalRelationshipToPhysicalActivity = TemporalRelationshipToPhysicalActivity.AFTER_EXCERCISE
timeFrameA= TimeFrame(None, None, 1594361431800)
timeFrameB= TimeFrame(None, None, 1594361431900)
descriptiveStatistic2 = DescriptiveStatistic.MEDIAN
rriA = RRI(rRIUnitValue,10,temporalRelationshipToPhysicalActivity,timeFrameA,descriptiveStatistic2,"rri")
rriB = RRI(rRIUnitValue,20,temporalRelationshipToPhysicalActivity,timeFrameB,descriptiveStatistic2,"rri")
rriList = [rriA.dict(), rriB.dict()]
keys = ["uniqueid","healthid","uploadtime","recordtime","recordschema","externalid","physicalActivity","temporalRelationshipToMeal","timeFrame","rri","comment"]
values =["uniqueidPython002","healthidPython001",1594361431757,1594361431757,"1","externalid001",physicalActivity.dict(),
         temporalRelationshipToMeal.getSchemaValue(),timeFrame.dict(),rriList,"python测试备注"]
dictionary = dict(zip(keys, values))
# 构造写入数据表请求
req = WriteTableDataRequest(sessiontoken, "<yourTableID>", [dictionary])
# 写入数据
bridgeclient.get_bridgedata_provider().write_table_data(req)
```
