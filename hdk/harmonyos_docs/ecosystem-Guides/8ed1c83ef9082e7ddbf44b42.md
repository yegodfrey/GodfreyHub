样例代码：向上图数据表中插入两条数据，代码如下：

```codeblock
import com.google.common.collect.Lists;

import com.huawei.hiresearch.client.model.table.WriteTableDataRequest;

import java.io.IOException;

import java.util.HashMap;

import java.util.List;

import java.util.Map;

public class WriteTable {

    public static void main(String[] args) throws IOException {

        // 获取鉴权
        HiResearchAuth hiResearchDemo = new HiResearchAuth();
        // 获取最新accesstoken
        String accessToken = hiResearchDemo.getAccessToken();
              // 新增写入数据表请求
        WriteTableDataRequest writeTableDataRequest = new WriteTableDataRequest();
        writeTableDataRequest.setSessionToken(accessToken);
        writeTableDataRequest.setTableId("<yourTableID>");
        // 两条数据第一条不传入uniqueid，第二条传入uniqueid，由系统自动生成UUID
              // uniqueid相同的数据，会相互覆盖
        List<Map<String, Object>> dataRows = Lists.newArrayList();
        Map<String, Object> dataRow1 = new HashMap<>();
        dataRow1.put("user_number", "userNumber001");
        dataRow1.put("weight", 55.5);
        dataRow1.put("height", 175);
        dataRow1.put("birthday", 19950101);
        dataRow1.put("gender", 1);
        dataRow1.put("isTrue", true);
        Map<String, Object> dataRow2 = new HashMap<>();
        dataRow2.put("user_number", "userNumber002");
        dataRow2.put("weight", 55.5);
        dataRow2.put("height", 175);
        dataRow2.put("birthday", 19950101);
        dataRow2.put("gender", 1);
        dataRow2.put("isTrue", true);
        dataRow2.put("uniqueid", "uniqueid002");
        dataRows.add(dataRow1);
        dataRows.add(dataRow2);
        writeTableDataRequest.setDataRows(dataRows);
        hiResearchDemo.getClient().getResearchDataProvider().writeTableData(writeTableDataRequest);
    }
}
```
