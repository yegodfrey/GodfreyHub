# 获取调查问卷

HUAWEI Research平台提供调查问卷管理能力，项目管理员在HUAWEI Research平台"研究管理"->"量表问卷"中新增并发布问卷，如下图所示：

研究APP中通过如下示例代码获取调查问卷列表：

```codeblock
import com.huawei.hiresearch.bridge.model.bridge.ChoiceInfo; 
import com.huawei.hiresearch.bridge.model.bridge.QuestionInfo; 
import com.huawei.hiresearch.bridge.model.bridge.QuestionRule; 
import com.huawei.hiresearch.bridge.model.bridge.QuestionnaireInfo; 
import com.huawei.hiresearch.bridge.model.bridge.QuestionnaireInfos; 
import com.huawei.hiresearch.research.ResearchManager2; 
import com.huawei.hiresearch.research.provider.QuestionnaireProvider; 
import java.util.List; 
import io.reactivex.disposables.CompositeDisposable; 
private final CompositeDisposable mDisposable = new CompositeDisposable(); 
private QuestionnaireProvider questionnaireProvider; 
@Override 
protected void onDestroy() { 
   super.onDestroy(); 
   mDisposable.dispose(); 
} 
/** 
* 获取调查问卷 
* 
* @param projectCode 申请研究项目时所获取的项目编码 
*/ 
private void getQuestionnaires(String projectCode) { 
   questionnaireProvider = ResearchManager2.getInstance(projectCode).getQuestionnaireProvider(); 
   mDisposable.add(questionnaireProvider.getQuestionnaires() 
           .subscribe(resp -> { 
               if (resp.getSuccess()) { 
                   //TODO:获取调查问卷成功 
                   QuestionnaireInfos questionnaireInfos = resp.getData(); 
                   if (null != questionnaireInfos && null != questionnaireInfos.getQuestionnaires()) { 
                       for (QuestionnaireInfo questionnaireInfo : questionnaireInfos.getQuestionnaires()) { 
                           //文章id 
                           String id = questionnaireInfo.getId(); 
                           //问卷标题 
                           String title = questionnaireInfo.getTitle(); 
                           //问卷描述信息 
                           String description = questionnaireInfo.getDescription(); 
                           //问卷创建时间 
                           String createTime = questionnaireInfo.getCreateTime(); 
                           //问卷更新时间 
                           String updateTime = questionnaireInfo.getUpdateTime(); 
                           //问卷发布时间 
                           String releaseTime = questionnaireInfo.getReleaseTime(); 
                           //问卷发布策略 
                           int releaseStrategy = questionnaireInfo.getReleaseStrategy(); 
                           //问卷发布策略内容 
                           String strategyContent = questionnaireInfo.getStrategyContent(); 
                           //问卷开始时间 
                           String startTime = questionnaireInfo.getStartTime(); 
                           //问卷结束时间 
                           String endTime = questionnaireInfo.getEndTime(); 
                           //问卷问题 
                           List<QuestionInfo> questions = questionnaireInfo.getQuestions(); 
                           if (null != questions) { 
                               for (QuestionInfo questionInfo : questions) { 
                                   //00 单选；01 多选；02 问答；03 排序；04 打分 
                                   String type = questionInfo.getType(); 
                                   //问题内容 
                                   String content = questionInfo.getContent(); 
                                   //选项 
                                   List<ChoiceInfo> choices = questionInfo.getChoices(); 
                                   //规则 
                                   List<QuestionRule> questionRules = questionInfo.getRules(); 
                               } 
                           } 
                       } 
                   } 
               } else { 
                   //TODO：获取调查问卷失败 
                   int statusCode = resp.getStatusCode(); //HTTP状态码 
                   String code = resp.getCode(); //内部错误码 
                   String msg = resp.getMessage(); //错误信息描述 
               } 
           }, err -> { 
               //TODO:登录异常 
           })); 
}
```
