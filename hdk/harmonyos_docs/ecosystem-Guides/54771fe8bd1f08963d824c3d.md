# 登录HUAWEI Research

研究APP通过华为帐号登录成功后，需将openId，accessToken，authCode等信息传递给HUAWEI Research平台进行二次校验，鉴权通过后并加入研究项目才可以使用SDK中相关能力。全流程示例代码如下：

```codeblock
import com.huawei.hiresearch.bridge.BridgeManager2;
import com.huawei.hiresearch.bridge.model.authentication.HWSignIn;
import com.huawei.hiresearch.bridge.model.bridge.HWJoinInfo;
import com.huawei.hiresearch.bridge.model.consent.InformedConsent;
import com.huawei.hiresearch.bridge.provider.AuthenticationProvider;
import com.huawei.hiresearch.bridge.provider.StudyProjectProvider;
import java.util.List;
import io.reactivex.disposables.CompositeDisposable;
import com.huawei.hms.support.hwid.result.AuthHuaweiId;
 
private final CompositeDisposable mDisposable = new CompositeDisposable();
private AuthenticationProvider authenticationProvider;
private StudyProjectProvider studyProjectProvider;
private String projectCode;
@Override
protected void onDestroy() {
    super.onDestroy();
    mDisposable.dispose();
}
/**
 *  从知情同意、登录鉴权到加入研究项目全流程
 * @param projectCode  申请研究项目时所获取的项目编码
 */
private void init(String projectCode) {
        this.projectCode = projectCode;
        authenticationProvider = BridgeManager2.getInstance(projectCode).getAuthenticationProvider();
        studyProjectProvider = BridgeManager2.getInstance(projectCode).getStudyProjectProvider();
        informedConsent();
        }
//用户知情同意，签署《用户协议》与《隐私声明》
private void informedConsent() {
        mDisposable.add(studyProjectProvider.getInformedConsents()
        .subscribe(resp -> {
        if (resp.getSuccess()) {
        List<InformedConsent> data = resp.getData();
        if (null == data || data.size() < 1) {
        //TODO: 项目管理员在HUAWEI Research 平台未配置用户知情同意相关文件
        } else {
        for (InformedConsent item : data) {
        //发布文件类型： 0是《知情同意》 1是《隐私申明》
        int type = item.getType();
        //《用户协议》或《隐私声明》 发布地址
        String url = item.getUrl();
        }
        //TODO：APP弹出《用户协议》与《隐私声明》 让用户确定签署
        //用户确定后调用华为帐号登录
        signByHuaweiAccount();
        }
        } else {
        //TODO：获取用户知情失败
        int statusCode = resp.getStatusCode(); //HTTP状态码
        String code = resp.getCode(); //内部错误码
        String msg = resp.getMessage(); //错误信息描述
        }
        }, err -> {
        //TODO:获取用户知情异常
        }));
}
//华为帐号登录
private void signByHuaweiAccount() {
   //请参阅华为帐号服务 完成APP华为帐号登录
//华为帐号服务帮助文档 https://developer.huawei.com/consumer/cn/doc/development/HMS-Guides/account-introduction-v4
//华为帐号登录成功后通过回调 可以获取到 AuthHuaweiId 对象
//在登录成功回调中  调用sign2HiResearch 完成HUAWEI Research 二次鉴权校验
 
}
//登录至HUAWEI Research
private void sign2HiResearch(AuthHuaweiId hwAccount) {
 BridgeManager2.getFirstInstance().getBridgeConfig().setAccessToken(hwAccount.getAccessToken());
    HwAccountSignInReq hwSignIn = new HwAccountSignInReq(hwAccount.getOpenId(),projectCode, hwAccount.getAuthorizationCode());
    mDisposable.add(authenticationProvider.signIn(hwSignIn)
            .subscribe(resp -> {
                if (resp.getSuccess()) {
                               // 登录成功后设置token BridgeManager2.getFirstInstance().getBridgeConfig().setAccessToken(resp.getData().getReauthToken());
                    //登录成功加入研究项目
                    joinStudy(hwSignIn);
                } else {
                    //TODO：登录失败
                    int statusCode = resp.getStatusCode(); //HTTP状态码
                    String code = resp.getCode(); //内部错误码
                    String msg = resp.getMessage(); //错误信息描述
                }
            }, err -> {
                //TODO:登录异常
            }));
}
//加入研究项目
private void joinStudy(HwAccountSignInReq hwSignIn) {
        StudyProjectProvider studyProjectProvider = BridgeManager2.getInstance(projectCode).getStudyProjectProvider();
    mDisposable.add(studyProjectProvider.join()
            .subscribe(resp -> {
                if (resp.getSuccess()) {
                    //TODO:加入研究项目成功，跳转研究APP首页
                } else {
                    //TODO:加入研究项目失败
                    int statusCode = resp.getStatusCode(); //HTTP状态码
                    String code = resp.getCode(); //内部错误码
                    String msg = resp.getMessage(); //错误信息描述
                }
            }, err -> {
                //TODO:加入研究项目异常
            }));
}
```
