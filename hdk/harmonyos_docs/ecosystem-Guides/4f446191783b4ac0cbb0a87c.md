# 下载附件字段

本章节介绍如何下载存在附件字段的研究数据表的附件内容，附件字段值形如："attachments/2019/12/11/14/088888cccacc445db9eebba4b102bedd/1576045751653-088888cccacc445db9eebba4b102bedd.zip"
> 注意
>
> APP或者第三方系统上传的元数据中包含附件，如果研究数据表关联了该元数据的附件字段，则可用该API下载。

样例代码：下载uniqueid为64272626766046c19984a449b95cb634的这一条数据的attachment字段的内容。

```codeblock
import com.google.common.collect.Lists;
import com.huawei.hiresearch.client.model.table.FilterCondition;
import com.huawei.hiresearch.client.model.table.SearchTableDataRequest;
import java.io.IOException;
import java.util.ArrayList;
import java.util.List;
import java.util.Map;
public class DownloadAttachment {
  public static void main(String[] args) throws IOException {
    HiResearchAuth hiResearchDemo = new HiResearchAuth();
    // 获取最新accesstoken
    String accessToken = hiResearchDemo.getAccessToken();
    List<Map<String, Object>> resultRows = Lists.newArrayList();
    // 构造过滤条件
    List<FilterCondition> filterConditions = new ArrayList<>();
    // 单个过滤条件: uniqueid为64272626766046c19984a449b95cb634
    FilterCondition simpleCondition = new FilterCondition(FilterCondition.LOGIC_TYPE_AND,
        FilterCondition.OPERATOR_TYPE_GREATER_EQUALS, "uniqueid", "64272626766046c19984a449b95cb634");
    filterConditions.add(simpleCondition);
    // 构造查询请求
    String tableID = "t_heartstudy_f3mgl5ka";
    SearchTableDataRequest searchTableDataRequest = new SearchTableDataRequest(
        accessToken,
        tableID,
        filterConditions,
        null,
        1000,
        Lists.newArrayList("attachment"),
        null);
// 查询并打印结果集
hiResearchDemo.getClient().getResearchDataProvider().queryTableData(searchTableDataRequest,
        (rowsOfCurrentPage, totalCnt) -> {
          // 直接读取研究数据表中附件字段值即可
          String attachmentIdentifierExample = (String) rowsOfCurrentPage.get(0).get("attachment");
          // 根据附件字段下载内容到指定目录
          try {
            List<String> fileNames =
                hiResearchDemo.getClient().getResearchDataProvider().getAttachment(accessToken,
                    attachmentIdentifierExample, "d:/test");
            // 打印附件中的文件名称列表
            System.out.println(fileNames);
          } catch (IOException e) {
            e.printStackTrace();
          }
        });
  }
}
```
