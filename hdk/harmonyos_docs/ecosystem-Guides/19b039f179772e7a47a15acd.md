3. 插入数据至该数据表，脚本如下：

   ```codeblock
   import com.google.common.collect.Lists;
   import com.huawei.hiresearch.client.model.table.WriteTableDataRequest;
   import com.huawei.hiresearch.schemas.basictypes.TimeFrame;
   import com.huawei.hiresearch.schemas.enums.*;
   import com.huawei.hiresearch.schemas.standardfields.datapoints.*;
   import com.huawei.hiresearch.schemas.standardfields.health.RRI;
   import com.huawei.hiresearch.schemas.standardfields.sports.CaloriesBurned;
   import com.huawei.hiresearch.schemas.standardfields.sports.Distance;
   import com.huawei.hiresearch.schemas.standardfields.sports.PhysicalActivity;
   import com.huawei.hiresearch.schemas.units.DurationUnit;
   import com.huawei.hiresearch.schemas.units.KcalUnit;
   import com.huawei.hiresearch.schemas.units.LengthUnit;
   import com.huawei.huangyun.HiResearchAuth;
   import java.io.IOException;
   import java.util.ArrayList;
   import java.util.HashMap;
   import java.util.List;
   import java.util.Map;
   public class TestStandard {
       public static void main(String[] args) throws IOException {
           // 获取最新accesstoken
           HiResearchAuth hiResearchDemo = new HiResearchAuth();
           String accessToken = hiResearchDemo.getAccessToken();
           // 身体活动字段
           KcalUnitValue caloriesBurnedDataPoint = new KcalUnitValue(123.0, KcalUnit.KILOCALORIE);
           CaloriesBurned caloriesBurned = new CaloriesBurned.Builder(caloriesBurnedDataPoint)
                   .setActivityName("test")
                   .setTimeFrame(1594361431757l, 1594461431757l)
                   .setDescriptiveStatistic(DescriptiveStatistic.MEDIAN)
                   .setUserNotes("caloriesBurned-test")
                   .setDescriptiveStatisticDenominator(DescriptiveStatisticDenominator.EPISODE)
                   .build();
           LengthUnitValue lengthDataPoint = LengthUnitValueFactory.newDataPoint(12356.0, LengthUnit.KILOMETER);
           Distance distance = new Distance.Builder(lengthDataPoint)
                   .setTimeFrame(1594361431757l, 1594367431757l)
                   .setDescriptiveStatistic(DescriptiveStatistic.AVERAGE)
                   .setDescriptiveStatisticDenominator(DescriptiveStatisticDenominator.EPISODE)
                   .setUserNotes("distance-test")
                   .build();
           PhysicalActivity physicalActivity = new PhysicalActivity.Builder("activeName")
                   .setCaloriesBurned(caloriesBurned)
                   .setDistance(distance)
                   .setReportedActivityIntensity(ReportedActivityIntensity.LIGHT)
                   .setMetValue(20.0)
                   .setTimeFrame(111l)
                   .setUserNotes("physicalActivity")
                   .build();
           // TemporalRelationshipToMeal 就餐状态，枚举类型
           TemporalRelationshipToMeal temporalRelationshipToMeal = TemporalRelationshipToMeal.AFTER_BREAKFAST;
           // 时间
           TimeFrame timeFrame = new TimeFrame(1594361431757l,1594361431900l);
           // rri数组
           DurationUnit durationUnit = DurationUnit.SECOND;
           DurationUnitValue rriDataPoint = new DurationUnitValue(50052514l, durationUnit);
           RRI rriA = new RRI.Builder(rriDataPoint)
                   .setRelationshipToPhysicalActivity(RelationshipToPhysicalActivity.ACTIVE)
                   .setDescriptiveStatistic(DescriptiveStatistic.MAXIMUM)
                   .setTimeFrame(1594361431757l)
                   .setUserNotes("I feel dizzy")
                   .setSqi(98)
                   .build();
           RRI rriB = new RRI.Builder(rriDataPoint)
                   .setRelationshipToPhysicalActivity(RelationshipToPhysicalActivity.ACTIVE)
                   .setDescriptiveStatistic(DescriptiveStatistic.MAXIMUM)
                   .setTimeFrame(1594361431767l)
                   .setUserNotes("I feel dizzy")
                   .setSqi(198)
                   .build();
           List<RRI> rrilist = new ArrayList<>();
           rrilist.add(rriA);
           rrilist.add(rriB);
           Map<String, Object> datamap = new HashMap<String, Object>();
           datamap.put("uniqueid","javauniqueid001");
           datamap.put("healthid","healthid001");
           datamap.put("uploadtime",1594361431757l);
           datamap.put("recordtime",1594361431757l);
           datamap.put("recordschema","1");
           datamap.put("externalid","externalid001");
           datamap.put("physicalActivity", physicalActivity);
           datamap.put("temporalRelationshipToMeal", temporalRelationshipToMeal);
           datamap.put("timeFrame", timeFrame);
           datamap.put("rri", rrilist);
           datamap.put("comment", "测试备注");
           List<Map<String, Object>> dataRows = Lists.newArrayList();
           dataRows.add(datamap);
           // 新增写入数据表请求
           WriteTableDataRequest writeTableDataRequest = new WriteTableDataRequest();
           writeTableDataRequest.setSessionToken(accessToken);
           writeTableDataRequest.setTableId("<yourTableID>");
           writeTableDataRequest.setDataRows(dataRows);
           hiResearchDemo.getClient().getResearchDataProvider().writeTableData(writeTableDataRequest);
       }
   }
   ```
