# 系统元数据

平台预置系统元数据，支持标准化上传运动健康数据，便于您后续研究分析。您可以登录平台，在"项目管理 < 元数据"查看系统元数据说明。

> 注意
>
> SDK数据项是指您通过数据同步接口，直接获取到的数据结构。具体定义见"3.4 数据说明"。
>
> SDK中平台预置系统元数据定义见huaweiresearch-android-common-x.x.x.aar中com.huawei.hiresearch.common.model.metadata.*下定义。

以上传24小时连续心率数据为例，调用SDK接口从穿戴设备获取到连续心率后转换为系统元数据格式后，上传到Huawei Research平台。样例代码如下：

import com.huawei.hiresearch.sensorprosdk.SensorProManager; 
import com.huawei.hiresearch.sensorprosdk.datatype.hr.HrData; 
import com.huawei.hiresearch.sensorprosdk.provider.callback.SensorProTransFileCallback; 
import java.util.ArrayList; import com.huawei.hiresearch.common.model.metadata.schemas.standardfields.unitvalues.HeartRateUnitValue; 
import com.huawei.hiresearch.common.model.metadata.health.ContinuousHeartRate; 
import java.util.List; import com.huawei.hiresearch.bridge.BridgeManager2; 
import com.huawei.hiresearch.bridge.listeners.OnMetadataUploadProgressChanged;

import com.huawei.hiresearch.bridge.model.dataupload.MetadataCompressResultInfo; 
import com.huawei.hiresearch.bridge.provider.BridgeDataProvider; 
import io.reactivex.disposables.CompositeDisposable;   
private final CompositeDisposable mDisposable = new CompositeDisposable();  
int startTime = (int) ((System.currentTimeMillis() - 48 * 60 * 60 * 1000L) / 1000); 
int endTime = (int) (System.currentTimeMillis() / 1000);  
/**
  * 获取穿戴设备上心率历史数据  *
  * @paramstartTime获取数据的开始时间
  * @paramendTime获取数据的结束时间
  */
 public void getHrDataHistory(int startTime, int endTime) {
     SensorProManager.getInstance().getHealthProvider().getHrDataHistory(startTime, endTime, new SensorProTransFileCallback<List<HrData>>() {
         /**
          * @Paramprogress 文件传输通用回调
          */         @Override
         public void onProgress(Progress progress) {
             // TODO:传输过程中回调
}
          /**
          * @paramerrorCode错误码
          * @paramreturnObject返回信息
          */
         @Override
         public void onResponse(int errorCode, List<HrData> returnObject) {
//系统元数据格式数据上传
            List<ContinuousHeartRate> continuousHeartRatesList = new ArrayList<>();
            HrData hrData = new HrData();
            if (returnObject != null) {
                for (int i = 0; i < returnObject.size(); i++) {
                    hrData.setHrValue(returnObject.get(i).getHrValue());
                    ContinuousHeartRate continuousHeartRate = new ContinuousHeartRate(
                        null,
                        null,
                        new com.huawei.hiresearch.common.model.metadata.schemas.standardfields.health.HeartRate.Builder(
                         new HeartRateUnitValue(hrData.getHrValue()))
                         .setTimeFrame(hrData.getDataStamp())
                         .build());
                   continuousHeartRate.setRecordtime(hrData.getDataStamp());
                   continuousHeartRatesList.add(continuousHeartRate);
                }
            }
            uploadData("yourProjectCode", continuousHeartRatesList); 
}
     });
 }
