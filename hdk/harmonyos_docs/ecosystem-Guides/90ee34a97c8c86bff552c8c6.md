# SDK初始化

1. Huawei Research Android SDK配置初始化

   添加ResearchStack类（继承于HiResearchBridgeStack），参考代码如下：

   ```codeblock
   import android.content.Context;  
   import com.huawei.hiresearch.bridge.BridgeManager2;  
   import com.huawei.hiresearch.bridge.HiResearchBridgeStack;  
   import com.huawei.hiresearch.bridge.config.BridgeConfig;  
   import com.huawei.hiresearch.bridge.config.HttpClientConfig;  
   import com.huawei.hiresearch.research.ResearchManager2;  
   import com.huawei.hiresearch.research.config.ResearchConfig;  
   import com.huawei.hiresearch.sensor.SensorManager;  
   import com.huawei.hiresearch.sensor.config.SensorConfig;  
   import com.huawei.hiresearch.sensorfat.SensorScaleManager;  
   import com.huawei.hiresearch.sensorprosdk.SensorProManager;  
   import java.util.concurrent.TimeUnit;  
     
   public class ResearchStack extends HiResearchBridgeStack {  
       /*** 
        * constructor 
        * @param context 
        */  
       public ResearchStack(Context context, String projectCode) {  
             
           // 初始化bridge（必选）  
           initBridge(context, projectCode);  
     
           // 初始化sensor(可选)  
           initSensor(context);  
     
           // 初始化research(可选)  
           initResearch(context, projectCode);  
     
           // 初始化sensorpro(可选)  
           initSensorPro(context);  
     
           // 初始化sensorpro-scale(可选)  
           initScale(context);  
       }  
       
       /** 
        * 初始化bridge、common（必选） 
        * 模块说明见"3.1.2 HUAWEI Research Android SDK模块说明" 
        * 
        * @param context     上下文 
        * @param projectCode 研究项目编码 
        */  
       private void initBridge(Context context, String projectCode) {  
           BridgeManager2.init(context);  
           // http client 配置  
           HttpClientConfig httpClientConfig = new HttpClientConfig(30, 30, 30, TimeUnit.SECONDS);  
           // bridge配置  
           BridgeConfig bridgeConfig = new BridgeConfig(context);  
           // 设置研究项目id  
           bridgeConfig.setProjectCode(projectCode);  
           BridgeManager2.getInstance().addStudyProject(httpClientConfig, bridgeConfig);  
       }  
     
       /** 
        * 初始化Sensor 
        * 仅在您需要使用huawei-research-android-sensor-x.x.x.aar时初始化 
        * 模块说明见"3.1.2 HUAWEI Research Android SDK模块说明" 
        * 
        * @param context 上下文 
        */  
       private void initSensor(Context context) {  
           // 设置读取设备数据超时时间  
           SensorConfig sensorConfig = new SensorConfig(60000);  
           SensorManager.init(context, sensorConfig);  
       }  
     
     
       /** 
        * 初始化research 
        * 仅在您需要使用huawei-research-android-research-x.x.x.aar时初始化 
        * 模块说明见"3.1.2 HUAWEI Research Android SDK模块说明" 
        * 
        * @param context     上下文 
        * @param projectCode 研究项目编码 
        */  
       private void initResearch(Context context, String projectCode) {  
           ResearchManager2.init(context);  
           ResearchConfig researchConfig = new ResearchConfig(projectCode);  
           ResearchManager2.getInstance().addStudyProject(researchConfig);  
       }  
     
       /** 
        * 初始化sensorpro 
        * 仅在您需要使用huawei-research-android -sensorpro-alg-x.x.aar、 
        * huawei-research-android -sensorpro-common-x.x.aar、 
        * huawei-research-android -sensorpro-wear-x.x.aar时初始化 
        * 模块说明见"3.1.2 HUAWEI Research Android SDK模块说明" 
        * 
        * @param context 上下文 
        */  
       private void initSensorPro(Context context) {  
           SensorProManager.init(context, ""); 
           SensorProCommonManager.init(context); 
     
           // （可选）开启SensorPro日志采集开关。APP需开启存储权限，且APP是debug版本  
           SensorProManager.getInstance().setDebugLog(true);  
       }  
     
       /** 
        * 初始化体脂秤模块 
        * 仅在您需要使用huawei-research-android-scale-x.x.aar时初始化 
        * 模块说明见"3.1.2 HUAWEI Research Android SDK模块说明" 
        * 
        * @param context 上下文 
        */  
       private void initScale(Context context) {  
           SensorScaleManager.init(context);  
       } 
   }
   ```

2. 在MainApplication 的onCreate函数中初始化SensorProManager和HiResearchBridgeStack

   ```codeblock
   import android.app.Application;
   import com.huawei.hiresearch.bridge.HiResearchBridgeStack;
   import com.huawei.hiresearch.sensorfat.SensorScaleManager; 
   public class MainApplication extends Application {
       @Override
       public void onCreate() {
           super.onCreate();
           //SDK初始化
           //"<yourPojectCode>"：HuaweiResearch平台对应研究项目编码
           HiResearchBridgeStack.init(this, new ResearchStack(this, "<yourPojectCode>"));
            
       }
       @Override
       public void onTerminate() {
           super.onTerminate();
           //仅在您初始化了sensor模块时调用
           SensorManager.getInstance().destrory();
           }
   }
   ```
