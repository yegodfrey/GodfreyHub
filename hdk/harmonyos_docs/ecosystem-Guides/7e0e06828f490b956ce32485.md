# 获取鉴权

本章节介绍如何使用JAVA SDK获取鉴权。

获取鉴权成功后，会返回AccessToken，RefreshToken，AccessTokenDurationInMillis（1天）， RefreshTokenDurationInMillis（10天）。其中AccessToken为后续调用SDK api必须的参数，另外AccessTokenDurationInMillis和RefreshTokenDurationInMillis为授权token的有效时间，当AccessToken快过期时，可通过RefreshToken调用刷新鉴权接口获取新的AccessToken。

样例代码：根据项目编码、accessKey、secretKey获取HUAWEI Research鉴权信息，并判断AccessToken是否失效，若失效，根据RefreshToken刷新获取最新鉴权信息。

```codeblock
import com.huawei.hiresearch.client.HiResearchClient;
import com.huawei.hiresearch.client.config.ClientConfig;
import com.huawei.hiresearch.client.config.HttpConfig;
import com.huawei.hiresearch.client.model.authentication.ExchangeForAkReq;
import com.huawei.hiresearch.client.model.authentication.OauthAkResp;
import com.huawei.hiresearch.client.model.authentication.RefreshForAkReq;
import com.huawei.hiresearch.client.provider.AuthenticationProvider;
import java.io.IOException;
import java.util.Date;
import java.util.concurrent.TimeUnit;
/**
 * 获取HUAWEI Research鉴权的样例代码
 */
public class HiResearchAuth {
    public static void main(String[] args) throws IOException {
        HiResearchAuth hiResearchAuth = new HiResearchAuth();
        String accessToken = hiResearchAuth.getAccessToken();
        // TODO 使用accesstoken调用sdk的表操作api
    }
    private HiResearchClient hiResearchClient;
    private OauthAkResp lastestAuthInfo;
    public HiResearchAuth() {
        init();
    }
    /**
     * 获取HUAWEI Research客户端
     *
     * @return
     */
    public HiResearchClient getClient() {
        return this.hiResearchClient;
    }
    /**
     * 获取accesstoken
     *
     * @return
     * @throws IOException
     */
    public String getAccessToken() throws IOException {
        if (lastestAuthInfo == null) {
            // 获取鉴权ak
            lastestAuthInfo = getAccessToken(hiResearchClient);
            return lastestAuthInfo.getAccessToken();
        }
        //  判断accesstoken是否过期，如过期，需要刷新获取最新accesstoken再调用sdk的api
        Date timeWhenTokenIsCreated = new Date(); // token创建时间
        long safePeriod = 5 * 60 * 1000; // 用于确保token在失效前被刷新的安全时间间隔，5min
        Date currentTime = new Date(); // 当前时间
        // 判断accesstoken是否过期，默认1天有效期，单位为毫秒
        if (currentTime.getTime() - timeWhenTokenIsCreated.getTime() - lastestAuthInfo.getAccessTokenDurationInMillis() < safePeriod) {
            // 判断refreshtoken是否过期，默认10天有效期，单位为毫秒
            if (currentTime.getTime() - timeWhenTokenIsCreated.getTime() - lastestAuthInfo.getRefreshTokenDurationInMillis() < safePeriod) {
                // accesstoken与refreshtoken均已失效，需要重新鉴权获取
                lastestAuthInfo = getAccessToken(hiResearchClient);
            } else {
                // accesstoken已失效，refreshtoken未失效，需要刷新鉴权
                lastestAuthInfo = refreshAccessToken(hiResearchClient, lastestAuthInfo.getRefreshToken());
            }
        }
        // 均未失效，直接获取当前accesstoken即可
        return lastestAuthInfo.getAccessToken();
    }
    /**
     * 初始化客户端
     *
     * @return
     */
    private void init() {
        // 初始化HttpConfig类
        HttpConfig httpConfig = new HttpConfig();
        // 设置连接超时,单位:s, 默认30s
        httpConfig.setConnectTimeout(100);
        httpConfig.setConnectTimeoutUnit(TimeUnit.SECONDS);
        // 设置读取超时,单位:s, 默认30s
        httpConfig.setReadTimeout(100);
        httpConfig.setReadTimeoutUnit(TimeUnit.SECONDS);
        // 设置写超时,单位:s, 默认30s
        httpConfig.setWriteTimeout(100);
        httpConfig.setWriteTimeoutUnit(TimeUnit.SECONDS);
        // 设置断连后自动重连,默认不重连
        httpConfig.setRetryOnConnectionFailure(true);
        // 初始化ClientConfig类
        ClientConfig clientConfig = new ClientConfig();
        //  HUAWEI Research平台API地址
        String apiUrl = "https://hiresearch-kit.cbg-app.huawei.com";
        // HUAWEI Research平台对应项目编码
        String projectCode = "<yourPojectCode>";
        clientConfig.setBaseUrl(apiUrl);
        clientConfig.setStudyId(projectCode);
        // 初始化HiResearchClient,其中proxyConfig为代理设置,用户可根据需要设置
        hiResearchClient = new HiResearchClient(clientConfig, httpConfig, null);
    }
 /**
     * 获取accesstoken
     *
     * @param hiResearchClient
     * @return
     * @throws IOException
     */
    private static OauthAkResp getAccessToken(HiResearchClient hiResearchClient) throws IOException {
        // 获取鉴权
        // 初始化AuthenticationProvider
        AuthenticationProvider authenticationProvider = hiResearchClient.getAuthenticationProvider();
        // 构造鉴权请求，<yourAccessKey>、<yourAccessKey>是您在HUAWEI Research "帐号设置>我的凭证"中获取到的个人密钥
        ExchangeForAkReq exchangeForAkReq = new ExchangeForAkReq("<yourAccessKey>", "<yourAccessKey>");
        // 获取鉴权
        OauthAkResp resp = authenticationProvider.exchangeForAccessKey(exchangeForAkReq);
        return resp;
    }
    /**
     * 刷新accesstoken
     *
     * @param hiResearchClient
     * @param refreshToken
     * @return
     * @throws IOException
     */
    private static OauthAkResp refreshAccessToken(HiResearchClient hiResearchClient, String refreshToken) throws IOException {
        // 获取鉴权
        // 初始化AuthenticationProvider
        AuthenticationProvider authenticationProvider = hiResearchClient.getAuthenticationProvider();
        // 构造鉴权请求，<yourAccessKey>、<yourAccessKey>是您在HUAWEI Research "帐号设置>我的凭证"中获取到的个人密钥
        RefreshForAkReq exchangeForAkReq = new RefreshForAkReq("<yourAccessKey>", "<yourSecretKey>",
                refreshToken);
        // 获取新的鉴权
        OauthAkResp resp = authenticationProvider.refreshForAccessKey(exchangeForAkReq);
        return resp;
    }
```
