# 提供3D车控功能

#### 场景说明

车主应用可以在HiCar场景中为用户提供3D车控功能。

手机和车机通过HiCar连接后，将车主应用内容共享至HiCar智慧桌面，更方便实现车控功能。

#### 开发步骤

#### [h2]1.绑定HiCar服务

车主应用车控页UIAbility创建时，判断HiCar已经连接后，绑定HiCar服务用于进程间数据传输。

     1. 在module.json5中配置Car Kit相关权限。
            
            {
              "module": {
                ......
                "requestPermissions": [
                  {
                    "name": "ohos.permission.ACCESS_CAR_DISTRIBUTED_ENGINE"
                  },
                ]
              }
            }

     2. 导入模块。
            
            import { smartMobilityCommon } from '@kit.CarKit';
            import { hilog } from '@kit.PerformanceAnalysisKit';

     3. 注册监听。示例代码如下所示：
            
            try {
              // 获取SmartMobilityAwareness实例
              let awareness = smartMobilityCommon.getSmartMobilityAwareness();
            
              // 指定业务类型（如HiCar）
              let types: smartMobilityCommon.SmartMobilityType[] = [ 
                smartMobilityCommon.SmartMobilityType.HICAR 
              ];
              // 定义回调函数
              const callBack = (info: smartMobilityCommon.SmartMobilityInfo) => {
                hilog.info(0x0000, 'testTag', 'Received info: ' + JSON.stringify(info));
            
                if (info.status === smartMobilityCommon.SmartMobilityStatus.RUNNING) {
                  // HiCar已连接
                } else if (info.status === smartMobilityCommon.SmartMobilityStatus.IDLE) {
                  // HiCar已断开
                }
              };
            
              // 注册监听
              awareness.on('smartMobilityStatus', types, callBack);
            } catch (e) {
              hilog.error(0x0000, 'testTag', `监听失败，错误码: ${e?.code}`);
            }

     4. 注册绑定HiCar服务，通过connectServiceExtensionAbility来进行跨进程通信。示例代码如下所示：
            
            import { common, Want, AbilityConstant } from '@kit.AbilityKit';
            import { bundleManager } from '@kit.AbilityKit';
            import { rpc } from '@kit.IPCKit';
            const HICAR_BUNDLE_NAME = 'com.huawei.hmos.hicar';
            const EXT_SERVICE_DESCRIPTOR = 'HiCarExtOutService';
            private proxy?: rpc.IRemoteObject;
            
            // 1. Hicar服务信息
            let connectWant: Want = {
                bundleName: HICAR_BUNDLE_NAME,
                abilityName: EXT_SERVICE_DESCRIPTOR
            };
            // 2. 设置连接回调
            let connectOptions: common.ConnectOptions = {
              onConnect(elementName: bundleManager.ElementName, remote: rpc.IRemoteObject) {
                console.info('Connected to service!');
                // 保存代理对象用于通信
                this.proxy = remote; 
              },
              onDisconnect(elementName: bundleManager.ElementName) {
                console.error('Service disconnected!');
              },
              onFailed(code: number) {
                console.error(`Connection failed! Code: ${code}`);
              }
            };
            // 3. 在Ability中连接服务
            class EntryAbility extends UIAbility {
              onCreate(want: Want, launchParam: AbilityConstant.LaunchParam) {
                // 获取连接ID
                let connectionId: number = this.context.connectServiceExtensionAbility(connectWant, connectOptions);
              }
            }
            

     5. 车主应用退出时，取消监听**。** 示例代码如下所示：
            
            try {
              let awareness = smartMobilityCommon.getSmartMobilityAwareness();
              let types = [smartMobilityCommon.SmartMobilityType.HICAR];
              // 取消监听hicar状态
              awareness.off('smartMobilityStatus', types, callBack); 
              // 或取消该类型所有监听
              awareness.off('smartMobilityStatus', types); 
            } catch (e) {
              hilog.error(0x0000, 'testTag', `取消监听失败，错误码: ${e?.code}`);
            }
            // 断开hicar服务连接
            function disconnectAbility(context: common.UIAbilityContext) {
              hilog.info(0x00000, 'testTag', 'begin to disconnect Ability');
              if (connectId != undefined) {
                try {
                  context.disconnectServiceExtensionAbility(connectId);
                } catch (err) {
                  let code = (err as BusinessError).code;
                  let message = (err as BusinessError).message;
                  hilog.error(0x0000, 'testTag', 'disconnectService failed, code is ' + code + ', message is ' + message);
                }
              }
            }

    /**
       * 更新车信息给HiCar
       * @param data 车信息数据
       * @returns 传输结果
       */
      public async notifyCarInfo(data: Record<string, Object>): Promise<ResultData> {
        return this.notifyCardDataToHiCar(HICAR_EVENT_ACTION_SEND_CAR_INFO, data);
      }
      /**
       * 更新车状态给HiCar
       * @param data 车状态数据
       * @returns 传输结果
       */
      public async notifyCarState(data: Record<string, Object>): Promise<ResultData> {
        return this.notifyCardDataToHiCar(HICAR_EVENT_ACTION_SEND_CAR_STATE, data);
      }
      private async notifyCardDataToHiCar(action: string, data: Record<string, Object>): Promise<ResultData> {
        const rpcMessage: RpcMessage = {
          action: action,
          extra: data
        }
        try {
          return this.sendMessageToHiCar(rpcMessage);
        } catch (err) {
          console.error(`send msg to HiCar error`);
          return { code: 1002, message: 'send msg to HiCar error' };
        }
      }
      private async sendMessageToHiCar(msg: RpcMessage): Promise<ResultData> {
        return new Promise((resolve, reject) => {
          if (this.proxy === undefined) {
            resolve({ code: 1001, message: 'Service is not init.' });
            return;
          }
          const option: rpc.MessageOption = new rpc.MessageOption();
          const data: rpc.MessageSequence = rpc.MessageSequence.create();
          const reply: rpc.MessageSequence = rpc.MessageSequence.create();
          const ss = JSON.stringify(msg);
          data.writeString(ss);
          console.info(`writeString message: ${ss}`);
          this.proxy.sendMessageRequest(COMMAND_COMMON_SERVICE, data, reply, option).then((result) => {
            if (result === undefined) {
              resolve({code: 1003, message: 'result is empty'});
              return;
            }
            if (result?.errCode === 0) {
              try {
                const replyMsg: ResultData = JSON.parse(reply.readString());
                resolve(replyMsg);
              } catch (err) {
                console.error(`read result error`);
                resolve({code: 1004, message: 'reply parse error'});
              }
            } else {
              resolve({code: result.errCode, message: 'internal error'});
            }
          }).catch((err: BusinessError) => {
            reject(err);
          });
        })
      }
      /**
     * Message from client
     */
    export interface RpcMessage {
      /**
       * request action
       */
      action: string;
      /**
       * extra data
       */
      extra?: Record<string, Object>;
    }

#### [h2]3.从HiCar接收指令

当绑定HiCar服务成功后，车主应用需要向HiCar传递一个回调函数用于控制HiCar卡片的显隐状态。

HiCar卡片的显隐状态（action为vehicle_3D_UI_status），isHiCarCardHide字段说明如下：

     * true：卡片隐藏
     * false：卡片显示（默认值）
    
    {
        **"action"** : **"vehicle_3D_UI_status"** ,
        **"extra"** : {
            **"isHiCarCardHide"** : **false**
        }
    }

示例代码如下所示：
    
    private connectServiceExtensionAbility(context: common.UIAbilityContext): void {
        try {
        this.connectId = context.connectServiceExtensionAbility(HiCarResourceManager.want,
          {
            onConnect: (elementName: bundleManager.ElementName, proxy: rpc.IRemoteObject) => {
    	this.proxy = proxy;
    
    	const data: rpc.MessageSequence = rpc.MessageSequence.create();
    	// writeRemoteObject函数传一个回调函数给HiCar
             data.writeRemoteObject(new HiCarEventCallback('HiCarEventCallback'));
            },
          });
      } catch (err) {
      }
    }

注册回调函数的示例代码如下所示：
    
    export class HiCarEventCallback extends rpc.RemoteObject {
      onRemoteMessageRequest(code: number, data: rpc.MessageSequence, reply: rpc.MessageSequence,
      options: rpc.MessageOption): boolean | Promise<boolean> {
        console.info('HiCarEventCallback:' + code);
        if (code === 2000) {
          const dataStr = data.readString();
          console.info('dataStr:' + dataStr);
        }
        return true;
      }
    }

#### [h2]4.解绑HiCar服务

当退出时，车主应用需要解绑HiCar服务。示例代码如下所示：
    
    public destroy(context: common.UIAbilityContext): void {
        this.stateCallback = undefined;
        if (this.connectId !== 0) {
          try {
            context.disconnectServiceExtensionAbility(this.connectId);
          } catch (err) {
            console.info(`HiCarResourceManager disconnectServiceExtensionAbility err`);
          }
        }
      }
