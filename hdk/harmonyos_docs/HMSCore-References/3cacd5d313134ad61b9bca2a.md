#### Sample Code

```
private void getFile(Drive drive, String fileId){
    try{
        String imagePath = "/storage/emulated/0/DCIM/Camera/";
        String image = "IMG_20190702_223139_NEW.jpg";
        java.io.File f = new java.io.File(imagePath + image);

        // 获取文件元数据
        Drive.Files.Get request = drive.files().get(fileId);
        request.setFields("*");
        File mFile = request.execute();

        // 下载实体文件
        Drive.Files.Get get = drive.files().get(fileId);
        get.setForm("content");
        MediaHttpDownloader downloader = get.getMediaHttpDownloader();
        downloader.setContentRange(0, mFile.size() - 1).setDirectDownloadEnabled(true);
        get.executeContentAndDownloadTo(new FileOutputStream(f));
    } catch (Exception e){
        Log.e(TAG, "getFile error: " + e.getMessage());
    }
}
```
