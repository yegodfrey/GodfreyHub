```screen
// 请注意此处的this为Activity对象

samplePointSpeed.getFieldValue(Field.FIELD_MIN).setDoubleValue(4.0d);
samplePointSpeed.getFieldValue(Field.FIELD_MAX).setDoubleValue(8.0d);

// ActivitySummary中的DataSummary用来承载统计数据
List<SamplePoint> dataSummary = new ArrayList<>();
dataSummary.add(samplePointDistance);
dataSummary.add(samplePointSpeed);
activitySummary.setDataSummary(dataSummary);

// 构建SampleSection，并添加至SectionSummary

samplePointSectionDistanceSamplePoint.getFieldValue(Field.FIELD_DISTANCE).setDoubleValue(400d);

samplePointSectionSpeedSamplePoint.getFieldValue(Field.FIELD_MIN).setDoubleValue(40.0d);
samplePointSectionSpeedSamplePoint.getFieldValue(Field.FIELD_MAX).setDoubleValue(80.0d);
SampleSection sampleSection = new SampleSection.Builder()
    .setSectionNum(1)
    .setSectionTime(60 * 5L, TimeUnit.SECONDS)
    .setStartTime(startTime + 1L, TimeUnit.MILLISECONDS)
    .setEndTime(startTime + 5 * 60 * 1000L, TimeUnit.MILLISECONDS)
    .setSectionDataList(Arrays.asList(samplePointSectionDistanceSamplePoint, samplePointSectionSpeedSamplePoint))
    .build();
activitySummary.setSectionSummary(Collections.singletonList(sampleSection));

// 4.构造运动记录，设置统计数据ActivitySummary对象
ActivityRecord activityRecord = new ActivityRecord.Builder() 
    .setName("AddActivityRecord") 
    .setDesc("This is ActivityRecord add test!") 
    .setId("MyAddActivityRecordId") 
    .setActivityTypeId(HiHealthActivities.RUNNING) 
    .setStartTime(startTime, TimeUnit.MILLISECONDS) 
    .setEndTime(endTime, TimeUnit.MILLISECONDS) 
    .setActivitySummary(activitySummary)
    .setTimeZone("+0800")
    .build(); 
 
// 5.创建一个明细数据采集器 
DataCollector dataCollector = new com.huawei.hms.hihealth.data.DataCollector.Builder() 
    .setDataType(DataType.DT_INSTANTANEOUS_STEPS_RATE) 
    .setDataGenerateType(DataCollector.DATA_TYPE_RAW) 
    .setPackageName(context) 
    .setDataCollectorName("AddActivityRecord") 
    .build(); 
// 6.根据明细数据采集器，构造采样集 
SampleSet sampleSet = SampleSet.create(dataCollector); 
 
// 7.构造步频明细（DT_INSTANTANEOUS_STEPS_RATE）采样点对象，添加到明细数据采样集 
SamplePoint samplePointStepRate = sampleSet.createSamplePoint().setTimeInterval(startTime, endTime, TimeUnit.MILLISECONDS); 
samplePointStepRate.getFieldValue(Field.FIELD_STEP_RATE).setDoubleValue(10.0d);; 
sampleSet.addSample(samplePointStepRate); 
 
// 8.构造运动记录请求，设置运动记录activityRecord对象与其关联的明细数据采样集 
ActivityRecordInsertOptions insertRequest = 
    new ActivityRecordInsertOptions.Builder().setActivityRecord(activityRecord).addSampleSet(sampleSet).build(); 
 
// 9.调用addActivityRecord方法添加运动记录 
Task<Void> addTask = activityRecordsController.addActivityRecord(insertRequest); 
addTask.addOnSuccessListener(new OnSuccessListener<Void>() { 
    @Override 
    public void onSuccess(Void aVoid) { 
        Log.i("ActivityRecords","ActivityRecord added successfully!"); 
    } 
}).addOnFailureListener(new OnFailureListener() { 
    @Override 
    public void onFailure(Exception e) { 
        Log.i("ActivityRecords", "Failed to add ActivityRecord" + e.getMessage());
    } 
});
```

读取运动记录的示例代码如下：

```screen
"Java"
// 请注意此处的this为Activity对象
ActivityRecordsController activityRecordsController = HuaweiHiHealth.getActivityRecordsController(this);

// 1.构造请求对象的时间区间：开始和结束时间
Calendar cal = Calendar.getInstance();
Date now = new Date();
cal.setTime(now);
long endTime = cal.getTimeInMillis();
cal.add(Calendar.DAY_OF_YEAR, -1);
long startTime = cal.getTimeInMillis();

// 2.构造读取运动记录请求体
// 以下示例为读取给定时间段内的跑步运动记录，同时关联读取运动期间的活动心率
// 如果需要读取其他类型的运动期间的详情数据，请开发者自行添加对应的数据类型
List<String> activityTypeList = new ArrayList<>();
activityTypeList.add(HiHealthActivities.RUNNING);
ActivityRecordReadOptions readRequest =
    new ActivityRecordReadOptions.Builder().setTimeInterval(startTime, endTime, TimeUnit.MILLISECONDS)
        .readActivityRecordsFromAllApps()
        .read(DataType.DT_INSTANTANEOUS_EXERCISE_HEART_RATE)
        .setActivityTypeList(activityTypeList)
        .build();

// 3. 调用getActivityRecord方法，根据请求体条件获取Health Service Kit中的运动记录
Task<ActivityRecordReply> getTask = activityRecordsController.getActivityRecord(readRequest);
getTask.addOnSuccessListener(new OnSuccessListener<ActivityRecordReply>() {
    @Override
    public void onSuccess(ActivityRecordReply activityRecordReply) {
        Log.i("ActivityRecords", "Get ActivityRecord was successful!");
        // 打印查询到的运动记录列表
        List<ActivityRecord> activityRecordList = activityRecordReply.getActivityRecords();
        for (ActivityRecord activityRecord : activityRecordList) {
            if (activityRecord == null) {
                continue;
            }
            DateFormat dateFormat = DateFormat.getDateInstance();
            DateFormat timeFormat = DateFormat.getTimeInstance();
            Log.i("ActivityRecordSample",
                "Returned for ActivityRecord: " + activityRecord.getName() + "\n\tActivityRecord Identifier is "
                + activityRecord.getId() + "\n\tActivityRecord created by app is "
                + activityRecord.getPackageName() + "\n\tDescription: " + activityRecord.getDesc()
                + "\n\tStart: " + dateFormat.format(activityRecord.getStartTime(TimeUnit.MILLISECONDS))
                + " " + timeFormat.format(activityRecord.getStartTime(TimeUnit.MILLISECONDS)) + "\n\tEnd: "
                + dateFormat.format(activityRecord.getEndTime(TimeUnit.MILLISECONDS)) + " "
```

### deleteActivityRecord(ActivityRecordDeleteOptions activityRecordDeleteOptions)

|Method|
|:---------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------|
|public Task<Void> deleteActivityRecord([ActivityRecordDeleteOptions](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/ardo_class-0000001182667494) activityRecordDeleteOptions) 根据请求参数中指定的运动记录Id或起止时间，从Health Service Kit中删除数据，根据subDataTypeList删除详情数据。|

> 说明
>
> 1. 删除运动记录接口支持按运动记录id删除，也支持按删除时间段进行删除，若两者都存在时则按运动记录id为准。
> 2. 可以通过设置是否删除关联详情数据subDataTypeList删除详情数据。

**Parameters**

|Name|Description|
|:--------------------------|:------------------------------------------|
|activityRecordDeleteOptions|删除的运动记录查询请求，包括activityRecordIds、删除时间段、数据类型。|

**Return** **s**

|Type|Description|
|:---------|:------------------------|
|Task<Void>|用来检查请求是否成功、失败、已完成的Task对象。|

删除运动记录的示例代码如下：
