```screen
private void updateReplies(Drive drive, File mFile, Comment mComment, Reply mReply) {
    try {
        Reply reply = new Reply();
        reply.setDescription("interface test reply");
        reply.setOperate("reopen");
        mReply = drive.replies().update(mFile.getId(), mComment.getId(), mReply.getId(), reply).setFields("*").execute();
    } catch (Exception ex) {
        Log.e(TAG, "updateReplies error: " + ex.getMessage());
    }
}
```
