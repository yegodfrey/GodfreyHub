            @Override
            public void onFailure(Exception e) {
                Log.i("TAG", "Add dataType failed");
            }
        })
        .addOnCompleteListener(new OnCompleteListener<DataType>() {
            @Override
            public void onComplete(Task<DataType> task) {
                String res = task.isSuccessful() ? "success" : "failed";
                Log.i("TAG", "Add dataType " + res);
                if (task.getException() != null) {
                    Log.i("TAG", "GetException is " + task.getException().toString());
                }
            }
        });
}

#### parseHealthKitAuthResultFromIntent(Intent intent)

|Method|
|:----------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------|
|public [HealthKitAuthResult](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/auth-result-0000001098318552) parseHealthKitAuthResultFromIntent(Intent intent) 从登录授权流程返回的Intent中解析授权结果。|

Parameters  

|Name|Description|
|:-----|:---------------|
|intent|登录授权流程返回的Intent。|

Returns  

|Type|Description|
|:------------------------------------------------------------------------------------------------------------------|:----------------|
|[HealthKitAuthResult](https://developer.huawei.com/consumer/cn/doc/HMSCore-References/auth-result-0000001098318552)|从Intent中解析出的授权结果。|

```
"Java"
// 日志Tag
private static final String TAG = "HealthKitAuthActivity";

// 通过startActivityForResult方式拉起授权流程界面的请求码，具体数值开发者可以自行定义
private static final int REQUEST_AUTH = 1000;

@Override
protected void onActivityResult(int requestCode, int resultCode, @Nullable Intent data) {
    super.onActivityResult(requestCode, resultCode, data);

    // 只处理授权流程的响应
    if (requestCode == REQUEST_AUTH) {
        // 从intent中获取授权响应结果
        // 请注意此处的this为Activity对象
        HealthKitAuthResult result = HuaweiHiHealth.getSettingController(this).parseHealthKitAuthResultFromIntent(data);
        if (result == null) {
            Log.w(TAG, "authorization fail");
            return;
        }

        if (result.isSuccess()) {
            Log.i(TAG, "authorization success");
        } else {
            Log.w(TAG, "authorization fail, errorCode:" + result.getErrorCode());
        }
    }
}
```
