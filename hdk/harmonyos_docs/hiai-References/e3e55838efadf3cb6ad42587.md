// 创建文本翻译器。

ner.asyncEntityExtract(input).addOnSuccessListener(new OnSuccessListener<RemoteNerResultItem[]>() {

                        }

    @Override
    public void onFailure(Exception e) {
        // 识别失败,获取相关异常信息。     
        try {         
            MLException mlException = (MLException) e; 
            // 获取错误码，开发者可以对错误码进行处理，根据错误码进行差异化的页面提示。

        } 
 
    }
});
  
  // 释放资源
  if (ner != null) {
        ner.stop();
}
