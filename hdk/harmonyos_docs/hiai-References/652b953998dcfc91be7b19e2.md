# 使用示例

![](https://alliance-communityfile-drcn.dbankcdn.com/FileServer/getFile/cmtyPub/011/111/111/0000000000011111111.20260104150629.94521416166228852019260728634511:50001231000000:2800:33FF3A3271256B65B162A678B047F0A4E427B5517931D3C75991187BD26CBEBC.png)  
此类接口从DDK 5.0.1.0版本起废弃，请使用模型管家V1接口[使用示例](https://developer.huawei.com/consumer/cn/doc/hiai-References/usage-example-0000001052249426)替代。

假定当前有一个模型，训练时采用的训练集为BRG888的图片，使能了动态AIPP之后，可以接收YUYV类型的图片作为模型推理的输入；当用于模型推理的图片尺寸与训练集图片的尺寸不一致时，还可以使用AIPP的裁剪、缩放和Padding功能，改变输入图片尺寸。以下示例代码基于CANN API，使能了AIPP的裁剪功能，将一张YUV420格式的图片预处理为224x224的输入。

```
#include "model_manager/model_manager.h"
#include "model_manager/model_builder.h"
#include "tensor/image_format.h"
#include "tensor/image_tensor_buffer.h"
#include "tensor/image_process_config_types.h"
#include "tensor/image_config_tensor_util.h"

std::shared_ptr<hiai::IBuiltModel> builtModel = nullptr;
std::string modelName1 = "mymodel.om";
std::string modelPath1 = "/data/local/tmp/mymodel.om";
printf("modelName = %s\n", modelName1.c_str());
printf("modelPath = %s\n", modelPath1.c_str());
// 编译离线模型
auto retcode = builder->Build(buildoptions, modelName1, modelPath1, builtModel);
if (retcode != SUCCESS || builtModel == nullptr) {
    printf("Build failed\n");
    return;
}
printf("Build success\n");

// 加载模型
ModelInitOptions options;
std::shared_ptr<IModelManager> manager = CreateModelManager();
ret = manager->Init(options, builtModel, nullptr);
if (ret != SUCCESS) {
    printf("Init failed\n");
    return;
}
printf("Init builtModel succ\n");

// 创建图像输入tensor
std::vector<NDTensorDesc> inputDesc = builtModel->GetInputTensorDescs();
std::vector<std::shared_ptr<INDTensorBuffer>> input;
std::shared_ptr<IImageTensorBuffer> imageBuffer =
CreateImageTensorBuffer(1, 480, 480, ImageFormat::YUV420SP, ImageColorSpace::JPEG, 0);
input.push_back(dynamic_pointer_cast<INDTensorBuffer>(imageBuffer));

// 创建crop参数 裁剪输出成400*400
CropPara crop;
crop.cropStartPosW = 0;
crop.cropStartPosH = 0;
crop.cropSizeW = 400;
crop.cropSizeH = 400;
std::shared_ptr<INDTensorBuffer> cropParaTesnor = CreateImageConfigTensor(crop);
input.push_back(cropParaTesnor);

// 创建channelswp参数 uv通道交换功能打开
ChannelSwapPara channelswp;
channelswp.rbuvSwapSwitch = true;
channelswp.axSwapSwitch = false;
std::shared_ptr<INDTensorBuffer> channelswpParaTesnor = CreateImageConfigTensor(channelswp);
input.push_back(channelswpParaTesnor);

// 创建csc参数 转换成BGR888格式
CscPara csc;
csc.outputFormat = ImageFormat::BGR888;
csc.imageColorSpace = ImageColorSpace::JPEG;
std::shared_ptr<INDTensorBuffer> cscParaTesnor = CreateImageConfigTensor(csc);
input.push_back(cscParaTesnor);

// 创建resize参数 缩放到222*222
ResizePara resize;
resize.resizeOutputSizeW = 222;
resize.resizeOutputSizeH = 222;
std::shared_ptr<INDTensorBuffer> resizetesnor = CreateImageConfigTensor(resize);
input.push_back(resizetesnor);

// 创建dtcPara参数 用于减均值,减最小值,乘方差
DtcPara dtcPara;
dtcPara.pixelMeanChn0 = 0;
dtcPara.pixelMeanChn1 = 1;
dtcPara.pixelMeanChn2 = 2;
dtcPara.pixelMeanChn3 = 3;
dtcPara.pixelMinChn0 = 4.0f;
dtcPara.pixelMinChn1 = 5.0f;
dtcPara.pixelMinChn2 = 6.0f;
dtcPara.pixelMinChn3 = 7.0f;
dtcPara.pixelVarReciChn0 = 8.0f;
dtcPara.pixelVarReciChn1 = 9.0f;
dtcPara.pixelVarReciChn2 = 10.0f;
dtcPara.pixelVarReciChn3 = 11.0f;
std::shared_ptr<INDTensorBuffer> dtctesnor = CreateImageConfigTensor(dtcPara);
input.push_back(dtctesnor);

// 创建PadPara参数 上下左右各补1 到得输出大小为224*224的图像, 并且在通道0 上padding 的值为1.0, 其余没有设置的通道默认padding的值为0
PadPara paddingPara;
paddingPara.paddingSizeTop = 1;
paddingPara.paddingSizeBottom = 1;
paddingPara.paddingSizeLeft = 1;
paddingPara.paddingSizeRight = 1;
paddingPara.paddingValueChn0 = 1.0;
std::shared_ptr<INDTensorBuffer> padTensor = CreateImageConfigTensor(paddingPara);
input.push_back(padTensor);


std::vector<NDTensorDesc> outputDesc = builtModel->GetOutputTensorDescs();
std::vector<std::shared_ptr<INDTensorBuffer>> output;
for (size_t i = 0; i < outputDesc.size(); i++) {
   std::shared_ptr<INDTensorBuffer> outputtensorbuffer = CreateNDTensorBuffer(outputDesc[i]);
   output.push_back(outputtensorbuffer);
}

struct timeval tv_start1, tv_end1;
gettimeofday(&tv_start1, NULL);
ret = manager->Run(input, output);
if (ret != SUCCESS) {
   printf("run failed\n");
   return;
}
gettimeofday(&tv_end1, NULL);
printf("TestCL onProcessDone manager Time is: %f\n",
   (tv_end1.tv_sec - tv_start1.tv_sec) * 1000 + (tv_end1.tv_usec - tv_start1.tv_usec) * 1.0 / 1000);
printf("DeInit");
manager->DeInit();
printf("\n");
```
