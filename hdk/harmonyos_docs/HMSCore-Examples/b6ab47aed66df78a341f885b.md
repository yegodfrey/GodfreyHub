    package com.huawei.petalpay.checkout.demo.impl;
    
    import android.app.Activity;
    import android.os.Handler;
    import android.widget.Toast;
    
    import com.huawei.hms.aggrpay.AggrPayClient;
    import com.huawei.hms.aggrpay.Pay;
    import com.huawei.hms.aggrpay.entity.PayResult;
    import com.huawei.petalpay.checkout.demo.common.LogC;
    import com.huawei.petalpay.checkout.demo.impl.base.BasePetalPayImpl;
    
    /**
     * 文 件 名: PetalPayOmniImpl.java
     * 版    权: Copyright Huawei Tech.Co.Ltd. All Rights Reserved.
     * 描    述: 基础支付接入实现
     */
    public class PetalPayOmniImpl extends BasePetalPayImpl {
        private final static String TAG = "PetalPayOmniImpl";
        @Override
        public void payApi(final Activity activity, Handler handler) {
            context = activity;
            Runnable runnable = () -> {
                AggrPayClient client = Pay.getPayClient(activity);
                PayResult result = client.pay(getOrderStr());
                handler.post(() -> dealOminPayResult(result));
            };
            Thread payThread = new Thread(runnable);
            payThread.start();
        }
        /**
         * 支付结果处理
         *
         * @param payResult
         */
        public void dealOminPayResult(PayResult payResult) {
            if (payResult == null) {
                LogC.e(TAG, "支付失败", false);
                return;
            }
            String returnCode = payResult.getReturnCode();
            String returnMsg = payResult.getReturnMsg();
            // 如果returnCode="0"，表示成功，注意：为了支付安全，请以服务器的通知或查询结果为准。
            String payTips = createPayTips(returnCode, returnMsg);
            Toast.makeText(context, payTips, Toast.LENGTH_LONG).show();
        }
    }
