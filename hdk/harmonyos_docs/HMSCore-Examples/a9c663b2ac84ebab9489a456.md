    // 基础支付
    async pay() {
           // orderStr云测调用预下单接口获取prePayID后生成，具体请查接入文档和服务器demo看，此处仅做格式参考
          const orderStr = "{\"merc_no\":\"***\",\"prepay_id\":\"***\",\"timestamp\":\"1680490672345\",\"noncestr\":\"b5318bc2b712b47056506badac6d912a\",\"sign\":\"*****\",\"auth_id\":\"****\"}";
          // 快应用appId
          const appId = "****";
          HMSPAYMENT.pay(orderStr, appId).then(res => {
            console.log("res: "+ JSON.stringify(res));
            if(res.errCode === 0) {
              const result = JSON.parse(JSON.stringify(res.errMsg));
              const payResult =JSON.parse(result);
              const { returnCode } = JSON.parse(payResult.PayResult);
              console.log("returnCode: "+ returnCode);
              // 如果returnCode="0"，表示成功，注意：为了支付安全，请以服务器的通知或查询结果为准。
              if(returnCode === "0") {
                console.log("支付成功");
              } else {
                console.log("支付失败");
              }
            }
          }).catch(err => {
            console.log("err: "+ JSON.stringify(err));
          })
        },
