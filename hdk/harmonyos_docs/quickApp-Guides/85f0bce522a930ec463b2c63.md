# 接入定位服务

## 简介

华为定位服务采用GPS、WiFi、基站等多途径的混合定位模式进行定位，赋予应用开发者快速、精准地获取用户位置信息的能力，构建全球定位服务能力，助力开发者发展全球业务。
> 说明
>
> 在使用华为定位服务前，请务必确认同意相关协议，详细内容参见华为定位服务的[隐私声明](https://developer.huawei.com/consumer/cn/doc/HMSCore-Guides/introduction-0000001050706106)。

## 准入条件

接入华为定位服务前，需要完成如下准备工作：

* 在华为开发者联盟上注册成为开发者，并完成实名认证，详细参见"[账号注册认证](https://developer.huawei.com/consumer/cn/doc/start/registration-and-verification-0000001053628148)"。
* 在AppGallery Connect上完成快应用的创建，详细参见"[创建快应用](https://developer.huawei.com/consumer/cn/doc/quickApp-Guides/quickapp-create-quickapp-0000001079835824)"。

## 接入指引

华为定位服务可以在全球多个国家和地区使用，详细内容请参见"[华为定位服务支持的地区](https://developer.huawei.com/consumer/cn/doc/HMSCore-Guides-V5/supported-regions-0000001050437657-V5)"，在中国大陆以外的国家和地区，仅支持国际通用坐标系 wgs84。

调用 [地理位置](https://developer.huawei.com/consumer/cn/doc/quickApp-References/quickapp-api-geolocation-0000001074421453) 接口实现定位服务提供的相关功能。

为提高调试定位服务的准确率，请确保调试时满足如下条件：

* 已安装 HMS Core 3.0.0.300及以上版本。
* 打开HMS Core的定位权限。在"设置"下的"应用管理"中找到HMS Core，点击"权限"进入，确保位置权限处于开启状态。
* 开启待测试应用的定位权限。
* 开启系统的定位服务。即下拉通知栏，确保位置信息开关处于开启状态。
* 建议使用移动网络，并在网络信号较强的地方测试，以便华为定位服务可以提供更精确的位置信息。

<script>
  import geolocation from "@system.geolocation";
  import prompt from "@system.prompt";
  module.exports = {
    data: {
      componentData: {},
      platitude: "",
      plongitude: "",
      maplatitude: "",
      maplongitude: "",
      maddress: "",
      mpoiname: "",
      caddress: "选择起点位置",
      cname: "",
      aaddress: "点击输入目的地位置",
      aname: "",
      wcity: "",
      waddress: "",
      wlatitude: "",
      wlongitude: "",
      citydisplay: false,
      tmarkers: [
        {
          id: 1,
          latitude: 39.906901,
          longitude: 116.397972,
          //coordtype: "gcj02",
          coordType: "wgs84",
          title: "marker1",
          iconPath: "/Common/maker.jpg",
          width: "40px",
          height: "40px"
        }
      ]
    },
    onInit() {
      this.$page.setTitleBar({
        text: "华为快应用定位服务示例demo",
        textColor: "#ffffff",
        backgroundColor: "#007DFF",
        backgroundOpacity: 0.5,
        menu: true
      });
    },
    onShow(options) {
      "// Do something when show.";
      this.getcurrentposition();
    },
    getcurrentposition() {
      var that = this;
      //调用geolocation.subscribe方法来获取经纬度
      geolocation.subscribe({
        callback: function (data) {
          that.platitude = data.latitude;
          that.plongitude = data.longitude;
          //将callback中返回的经纬度传给map组件在地图中把当前所在位置显示出来
          that.maplatitude = data.latitude;
          that.maplongitude = data.longitude;
          console.log("handling success: longitude=" + data.longitude + ", latitude=" + data.latitude);
          that.getposition();
          //将map组件的中心点移动到所在的位置
          that.$element("mapid").moveToMyLocation();
        },
        fail: function (data, code) {
          console.log("handling fail, code=" + code);
        }
      });
    },
    getposition() {
      var that = this;
      //根据subscribe接口返回的经纬度获取位置对应的省份、城市等信息
      geolocation.reverseGeocodeQuery({
        latitude: that.platitude,
        longitude: that.plongitude,
        includePoiInfo: true,
        //coordType: "gcj02",
        coordType: "wgs84",
        success: function (ret) {
          //将返回的地址和城市等信息在页面的最上方展示出来
          that.maddress = ret.address;
          that.mpoiname = ret.poiInfoList[0].poiName;
          console.info(`### geolocation.reverseGeocodeQuery ###` + JSON.stringify(ret));
        },
        fail: function (errormsg, errorcode) {
          console.info(`### geolocation.reverseGeocodeQuery ### ${errorcode}: ${errormsg}`);
        }
      });
    },
    //选择起点位置
    chooseposition() {
      var that = this;
      geolocation.getLocation({
        success: function (data) {
          that.platitude = data.latitude;
          that.plongitude = data.longitude;
          that.maplatitude = data.latitude;
          that.maplongitude = data.longitude;
          console.log("handling success: longitude = " + data.longitude + ", latitude = " + data.latitude + ", accuracy = " + data.accuracy + ", time = " + data.time);
        },
        fail: function (data, code) {
          console.log("handling fail, code = " + code);
        },
        complete: function () {
          console.log("geolocation complete----------");
          that.getposition();
          that.$element("mapid").moveToMyLocation();
          //调用geolocation.chooseLocation接口打开快应用内置地图选择位置
          geolocation.chooseLocation({
            latitude: that.platitude,
            longitude: that.plongitude,
            //coordType: "gcj02",
            coordType: "wgs84",
            success: function (data) {
              console.log("chooseLocation success ： " + JSON.stringify(data));
              //将返回的地址信息作为起点位置在页面中展示出来
              that.cname = data.name;
              that.caddress = data.address;
            }
          });
        }
      });
    },
    //将选择终点位置的弹框显示出来
    choose() {
      this.citydisplay = true;
    },
    //获取输入的城市
    getcity(e) {
      this.wcity = e.value;
    },
    //获取输入的地址
    getaddress(e) {
      this.waddress = e.value;
    },
    //点击弹窗中的确定按钮调用geolocation.geocodeQuery和geolocation.chooseLocation并将返回的地址信息作为终点位置在页面中展示出来
    choosewanttoposition() {
      var that = this;
      //geolocation.geocodeQuery调用获取地理位置对应的经纬度
      geolocation.geocodeQuery({
        city: that.wcity,
        address: that.waddress,
        success: function (ret) {
          that.platitude = ret.latitude;
          that.plongitude = ret.longitude;
          console.info("### geolocation.geocodeQuery ### ", JSON.stringify(ret));
        },
        fail: function (errormsg, errorcode) {
          console.info("### geolocation.geocodeQuery ### ${errorcode}: ${errormsg}");
        },
        complete: function () {
          //调用geolocation.chooseLocation接口打开快应用内置地图选择位置
          geolocation.chooseLocation({
            //将geolocation.geocodeQuery获取的经纬度传入
            latitude: that.platitude,
            longitude: that.plongitude,
            //coordType: "gcj02",
            coordType: "wgs84",
            success: function (data) {
              console.log("chooseLocation success ： " + JSON.stringify(data));
              //将返回的地址信息作为终点位置在页面中展示出来
              that.aname = data.name;
              that.aaddress = data.address;
              that.issdisplay = true;
              //将返回的经纬度作为后面geolocation.openLocation所需的经纬度坐标，然后来进行导航
              that.wlatitude = data.latitude;
              that.wlongitude = data.longitude;
              //将终点位置的经纬度在地图上显示出来
              that.maplongitude = data.longitude;
              that.maplatitude = data.latitude;
              var mapcom = that.$element("mapid");
              mapcom.translateMarker({
                markerId: 1,
                destination: { latitude: that.maplatitude, longitude: that.maplongitude }
              });
            },
            fail: function (erromsg, errocode) {
              console.log("chooseLocation fail : " + errocode + ": " + erromsg);
            },
            complete: function () {
              console.log("chooseLocation complete.");
              setTimeout(() => {
                that.citydisplay = false; //关闭弹窗
              }, 500);
            }
          });
        }
      });
    },
    //判断起点和终点里是否有正确填写地址，没有就弹出相应提示，有就调用geolocation.openLocation，选择位置，拉起地图app进行导航
    toposition() {
      if (this.aname === "" || this.cname === "") {
        if (this.aname === "" && this.cname === "") {
          prompt.showToast({
            message: "请输入起点位置和目的地位置后再试"
          });
        } else if (this.aname === "" && this.cname !== "") {
          prompt.showToast({
            message: "请输入目的地位置后再试"
          });
        } else {
          prompt.showToast({
            message: "请输入起点位置后再试"
          });
        }
      } else {
        geolocation.openLocation({
          //将chooseLocation返回的经纬度作为参数填入
          latitude: this.wlatitude,
          longitude: this.wlongitude,
          //coordType: "gcj02",
          coordType: "wgs84",
          name: this.aname,
          address: this.aaddress,
          scale: 18,
          success: function () {
            console.log("openLocation success . ");
          },
          fail: function (erromsg, errocode) {
            console.log("geolocation.openLocation----------" + errocode + ": " + erromsg);
          },
          complete: function () {
            console.log("openLocation complete.");
          }
        });
      }
    },
    cancel() {
      this.citydisplay = false;
    }
  };
</script>
